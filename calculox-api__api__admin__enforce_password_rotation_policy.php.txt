<?php
// File: calcpro-api/api/admin/enforce_password_rotation_policy.php

require_once '../../core/config.php';
require_once '../../core/response.php';
require_once '../../core/auth.php';
require_once '../../core/SecurityGuard.php';

// 1. Initialize Security & Authenticate
$securityGuard = new SecurityGuard($conn);
$ipStatus = $securityGuard->checkIpStatus();
if (!$ipStatus['allowed']) {
    sendResponse(false, null, $ipStatus['message'], 403, $ipStatus['code']);
}

$adminData = verifyAdminToken();
$adminId = $adminData->adminId;

if ($_SERVER['REQUEST_METHOD'] !== 'POST') {
    sendResponse(false, null, 'Invalid request method.', 405);
}

$input = json_decode(file_get_contents('php://input'), true);
$currentPassword = $input['current_password'] ?? '';
$newPassword = $input['new_password'] ?? '';
$policyDays = isset($input['policy_days']) ? (int)$input['policy_days'] : 0;

// 2. Strict Input Validation (16 to 64 Characters)
if (empty($currentPassword) || empty($newPassword) || $policyDays <= 0) {
    sendResponse(false, null, 'Missing required security parameters.', 400, 'MISSING_PARAMETERS');
}

if (strlen($newPassword) < 16 || strlen($newPassword) > 64) {
    sendResponse(false, null, 'New password must be between 16 and 64 characters long.', 400, 'PASSWORD_LENGTH_INVALID');
}

if ($currentPassword === $newPassword) {
    sendResponse(false, null, 'New password cannot be the same as the current password.', 400, 'PASSWORD_REUSED');
}

try {
    // 3. Initiate ACID Transaction Lockbox
    $conn->beginTransaction();

    // 4. Verify Current Identity
    $stmt = $conn->prepare("SELECT password FROM admins WHERE id = :id LIMIT 1 FOR UPDATE");
    $stmt->execute([':id' => $adminId]);
    $admin = $stmt->fetch(PDO::FETCH_ASSOC);

    if (!$admin || !password_verify($currentPassword, $admin['password'])) {
        // Rollback immediately on authorization failure
        $conn->rollBack();
        
        // Log the failure in the Brute Force Manager
        $adminEmailStmt = $conn->prepare("SELECT email FROM admins WHERE id = :id");
        $adminEmailStmt->execute([':id' => $adminId]);
        $emailRow = $adminEmailStmt->fetch(PDO::FETCH_ASSOC);
        $securityGuard->logFailure($emailRow['email'] ?? 'unknown', $adminId, 'password');

        sendResponse(false, null, 'The current password provided is incorrect.', 401, 'AUTH_FAILED');
    }

    // 5. Hash New Credential
    $newPasswordHash = password_hash($newPassword, PASSWORD_DEFAULT);

    // 6. Execute Atomic Updates
    // A. Update Password & Force Session Expiration (via updated timestamp)
    $updateAdminStmt = $conn->prepare("UPDATE admins SET password = :password, password_last_updated = UTC_TIMESTAMP() WHERE id = :id");
    $updateAdminStmt->execute([
        ':password' => $newPasswordHash,
        ':id' => $adminId
    ]);

    // B. Update the Global Rotation Policy
    $updateSettingsStmt = $conn->prepare("UPDATE settings SET setting_value = :val, updated_at = UTC_TIMESTAMP() WHERE setting_key = 'password_expiry_days'");
    $updateSettingsStmt->execute([':val' => (string)$policyDays]);

    // 7. Commit Transaction (Lockbox successfully stored)
    $conn->commit();

    sendResponse(true, null, 'Security policy enforced and credentials updated successfully.');

} catch (Exception $e) {
    // 8. Failsafe: Rollback on any internal system error
    if ($conn->inTransaction()) {
        $conn->rollBack();
    }
    
    if (function_exists('logInternalSystemError')) {
        logInternalSystemError('critical', 'ACID Transaction Failed: ' . $e->getMessage(), __FILE__, __LINE__);
    }
    sendResponse(false, null, 'An internal error occurred while processing the security update.', 500, 'SERVER_ERROR');
}
?>
