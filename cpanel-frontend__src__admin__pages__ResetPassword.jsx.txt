// cPanel_Frontend/src/admin/pages/ResetPassword.jsx
import React, { useState, useEffect } from "react";
import { useParams, Link, useNavigate } from "react-router-dom";
import { motion, AnimatePresence } from "framer-motion";
import toast from "react-hot-toast";
import SafeIcon from "../../common/SafeIcon";
import { useLayout } from "../context/LayoutContext";
import { useTheme } from "../context/ThemeContext";

import * as IoIcons from "react-icons/io5";
import * as HiIcons from "react-icons/hi";
import * as BiIcons from "react-icons/bi";
import * as FiIcons from "react-icons/fi"; // Added for shield icons

const ResetPassword = () => {
  const { token } = useParams();
  const navigate = useNavigate();

  const { setBackgroundClass } = useLayout();
  const { isDark } = useTheme(); 

  const [uiState, setUiState] = useState(token ? "verifyingToken" : "requestForm");

  const [email, setEmail] = useState("");
  const [password, setPassword] = useState("");
  const [confirmPassword, setConfirmPassword] = useState("");
  const [showPassword, setShowPassword] = useState(false);
  const [showConfirmPassword, setShowConfirmPassword] = useState(false);
  const [isSubmitting, setIsSubmitting] = useState(false);
  const [error, setError] = useState("");
  const [siteName, setSiteName] = useState("Calculox");

  const [passwordStrength, setPasswordStrength] = useState(0);

  useEffect(() => {
    const fetchPublicSiteIdentity = async () => {
      try {
        const envUrl = import.meta.env.VITE_API_BASE_URL || '';
        const cleanBase = envUrl.replace(/\/+$/, '');
        const response = await fetch(`${cleanBase}/api/public/get_public_bootstrap.php`);
        const result = await response.json();
        if (result && result.success) {
          const fetchedName = result.data?.settings?.siteName || result.data?.siteName || result.data?.settings?.site_name;
          if (fetchedName) {
            setSiteName(fetchedName);
          }
        }
      } catch (error) {
        console.error("Could not fetch public site identity.");
      }
    };
    fetchPublicSiteIdentity();
  }, []);

  useEffect(() => {
    setBackgroundClass("bg-[#020617]");
    return () => setBackgroundClass("bg-gray-50 dark:bg-gray-900");
  }, [setBackgroundClass]);

  useEffect(() => {
    if (token) {
      const verifyToken = async () => {
        try {
          const envUrl = import.meta.env.VITE_API_BASE_URL || '';
          const cleanBase = envUrl.replace(/\/+$/, '');
          const response = await fetch(`${cleanBase}/api/admin/verify_reset_token.php?token=${encodeURIComponent(token)}`);
          const data = await response.json();

          if (response.ok && data.success) {
            setUiState("resetForm");
          } else {
            setUiState("invalidToken");
            setError(data.message || "This password reset link is invalid or has expired.");
          }
        } catch (err) {
          setUiState("invalidToken");
          setError("Network error occurred while verifying the secure link.");
        }
      };
      setTimeout(verifyToken, 600); 
    }
  }, [token]);

  const calculateStrength = (pass) => {
    let strength = 0;
    if (pass.length >= 16 && pass.length <= 64) strength += 1; 
    if (pass.match(/[A-Z]/)) strength += 1;
    if (pass.match(/[a-z]/)) strength += 1;
    if (pass.match(/[0-9]/)) strength += 1;
    if (pass.match(/[^A-Za-z0-9]/)) strength += 1;
    return strength;
  };

  const getStrengthColor = (strength) => {
      if (strength <= 2) return 'bg-rose-500';
      if (strength === 3) return 'bg-amber-500';
      if (strength === 4) return 'bg-blue-500';
      return 'bg-emerald-500';
  };

  const getStrengthLabel = (strength) => {
      if (strength === 0) return '';
      if (strength <= 2) return 'Weak - Add uppercase, numbers, or symbols';
      if (strength === 3) return 'Fair - Could be stronger';
      if (strength === 4) return 'Good - Almost there';
      return 'Strong - Excellent password';
  };

  const getStrengthTextColor = (strength) => {
      if (strength <= 2) return 'text-rose-400';
      if (strength === 3) return 'text-amber-400';
      if (strength === 4) return 'text-blue-400';
      return 'text-emerald-400';
  };

  const handlePasswordChange = (value, setter, isNewPwd = false) => {
    let finalValue = value;
    if (value.length > 64) {
      finalValue = value.slice(0, 64);
    }
    setter(finalValue);
    if (isNewPwd) {
        setPasswordStrength(calculateStrength(finalValue));
    }
  };

  const handleRequestSubmit = async (e) => {
    e.preventDefault();
    setError("");
    setIsSubmitting(true);

    try {
      const envUrl = import.meta.env.VITE_API_BASE_URL || '';
      const cleanBase = envUrl.replace(/\/+$/, '');
      const response = await fetch(`${cleanBase}/api/admin/request_admin_password_reset.php`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ email }),
      });

      const data = await response.json();

      if (response.ok && data.success) {
        setUiState("emailSent");
        toast.success(data.message || "Reset link sent!");
      } else {
        setError(data.message || "Failed to send reset link.");
        toast.error("Failed to process request.");
      }
    } catch (err) {
      setError("Network error. Please try again.");
    } finally {
      setIsSubmitting(false);
    }
  };

  const isPasswordValid = password.length >= 16 && password.length <= 64;
  const hasStartedTyping = password.length > 0;
  const confirmHasStarted = confirmPassword.length > 0;
  const isMatch = password === confirmPassword;

  const handleResetSubmit = async (e) => {
    e.preventDefault();
    setError("");

    if (!isPasswordValid) {
      setError("Password must be between 16 and 64 characters long.");
      return;
    }
    if (passwordStrength < 3) {
      setError("Please choose a stronger password to proceed.");
      return;
    }
    if (!isMatch) {
      setError("Passwords do not match.");
      return;
    }

    setIsSubmitting(true);

    try {
      const envUrl = import.meta.env.VITE_API_BASE_URL || '';
      const cleanBase = envUrl.replace(/\/+$/, '');
      const response = await fetch(`${cleanBase}/api/admin/reset_admin_password.php`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
          token: token,
          password: password,
        }),
      });

      const data = await response.json();

      if (response.ok && data.success) {
        // [NEW] Transition to Labor Gap instead of immediate success
        setUiState("securing");
        setTimeout(() => {
            setUiState("success");
            toast.success(data.message || "Password updated successfully!");
            setIsSubmitting(false);
        }, 2500);
      } else {
        setError(data.message || "Failed to reset password.");
        setIsSubmitting(false);
      }
    } catch (err) {
      setError("Network error. Please try again.");
      setIsSubmitting(false);
    }
  };

  const renderBackgroundOrbs = () => (
    <div className="absolute top-0 left-0 w-full h-full overflow-hidden z-0 pointer-events-none">
      <div className="absolute top-[-10%] left-[-10%] w-[500px] h-[500px] bg-indigo-500/10 rounded-full blur-[120px] mix-blend-screen animate-pulse-slow"></div>
      <div className="absolute bottom-[-10%] right-[-10%] w-[600px] h-[600px] bg-slate-600/10 rounded-full blur-[150px] mix-blend-screen"></div>
      <div className="absolute top-[40%] right-[20%] w-[300px] h-[300px] bg-blue-900/10 rounded-full blur-[100px] mix-blend-screen"></div>
    </div>
  );

  return (
    <div className="flex flex-col items-center justify-center min-h-screen p-4 relative font-inter overflow-hidden bg-[#020617]">
      {renderBackgroundOrbs()}

      <motion.div
        initial={{ opacity: 0, y: 40 }}
        animate={{ opacity: 1, y: 0 }}
        transition={{ duration: 0.7, ease: [0.16, 1, 0.3, 1] }}
        className="w-full max-w-[440px] z-10"
      >
        <div className="bg-[#0B1221]/80 backdrop-blur-3xl rounded-[2.5rem] shadow-[0_20px_40px_rgb(0,0,0,0.5)] border border-slate-800/60 relative overflow-hidden p-5 sm:p-7">
          
          <div className="absolute top-0 left-0 right-0 h-[1px] bg-gradient-to-r from-transparent via-white/10 to-transparent"></div>

          <div className="text-center mb-6 mt-1">
            <div className="w-14 h-14 bg-gradient-to-br from-slate-800 to-[#020617] rounded-2xl mx-auto mb-4 flex items-center justify-center shadow-[0_8px_20px_rgba(0,0,0,0.4)] ring-1 ring-slate-700/60 relative group">
                <div className="absolute inset-0 bg-blue-500/10 rounded-2xl transition-colors duration-300 group-hover:bg-blue-500/20"></div>
                <SafeIcon icon={IoIcons.IoGridOutline} className="w-7 h-7 text-blue-400 relative z-10" />
            </div>
            
            <h1 className="text-2xl font-bold text-white mb-2 font-montserrat tracking-tight">
              {siteName}
            </h1>
            <p className="text-sm text-slate-400 font-medium uppercase tracking-widest">
              Recovery Portal
            </p>
          </div>

          <AnimatePresence>
            {error && uiState !== "invalidToken" && (
              <motion.div
                key="error"
                initial={{ opacity: 0, y: -10 }}
                animate={{ opacity: 1, y: 0 }}
                exit={{ opacity: 0, y: -10 }}
                className="mb-6 p-3 rounded-2xl bg-rose-900/10 border border-rose-500/30 flex items-start gap-3 shadow-[inset_0_1px_4px_rgba(0,0,0,0.2)]"
              >
                <SafeIcon icon={IoIcons.IoAlertCircleOutline} className="w-5 h-5 text-rose-400 shrink-0 mt-0.5" />
                <p className="text-[13px] text-rose-300 font-medium">{error}</p>
              </motion.div>
            )}
          </AnimatePresence>

          <AnimatePresence mode="wait">
            
            {uiState === "verifyingToken" ? (
              <motion.div
                key="verifying"
                initial={{ opacity: 0, scale: 0.95 }}
                animate={{ opacity: 1, scale: 1 }}
                exit={{ opacity: 0, scale: 0.95 }}
                transition={{ duration: 0.3 }}
                className="flex flex-col items-center justify-center py-8 space-y-4"
              >
                <span className="w-10 h-10 border-4 border-blue-500/20 border-t-blue-500 rounded-full animate-spin" />
                <p className="text-sm font-semibold text-slate-400">Verifying secure link...</p>
              </motion.div>

            ) : uiState === "invalidToken" ? (
              <motion.div
                key="invalid"
                initial={{ opacity: 0, scale: 0.95 }}
                animate={{ opacity: 1, scale: 1 }}
                exit={{ opacity: 0, scale: 0.95 }}
                transition={{ duration: 0.3 }}
                className="text-center py-4"
              >
                <div className="w-16 h-16 bg-rose-900/20 rounded-full flex items-center justify-center mb-5 mx-auto border border-rose-500/30 shadow-[inset_0_1px_4px_rgba(0,0,0,0.3)]">
                  <SafeIcon icon={IoIcons.IoCloseCircle} className="w-8 h-8 text-rose-400" />
                </div>
                <h2 className="text-xl font-bold text-white mb-2">Link Invalid or Expired</h2>
                <p className="text-sm text-slate-400 mb-8 max-w-[280px] mx-auto font-medium">
                  {error || "For your security, password reset links expire after a short time. Please request a new one."}
                </p>
                <button
                  onClick={() => {
                    navigate('/reset-password');
                    setUiState("requestForm");
                    setError("");
                  }}
                  className="w-full bg-white/5 hover:bg-white/10 text-white font-semibold py-3 px-4 rounded-2xl transition-all duration-300 border border-white/10 shadow-[inset_0_1px_4px_rgba(0,0,0,0.1)] text-sm"
                >
                  Request New Reset Link
                </button>
              </motion.div>

            ) : uiState === "requestForm" ? (
              <motion.div
                key="request"
                initial={{ opacity: 0, filter: "blur(4px)", scale: 0.98 }}
                animate={{ opacity: 1, filter: "blur(0px)", scale: 1 }}
                exit={{ opacity: 0, filter: "blur(4px)", scale: 0.98 }}
                transition={{ duration: 0.3 }}
              >
                <div className="text-center mb-6">
                  <p className="text-sm text-slate-400 font-medium">
                    Enter your email address and we'll send you instructions to reset your password.
                  </p>
                </div>
                <form onSubmit={handleRequestSubmit} className="space-y-4">
                  <div className="space-y-2">
                    <label className="block text-[11px] font-bold text-slate-400 uppercase tracking-widest ml-1">
                      Email Address
                    </label>
                    <div className="relative group">
                      <SafeIcon icon={HiIcons.HiOutlineMail} className="absolute left-4 top-1/2 -translate-y-1/2 w-5 h-5 text-slate-400 group-focus-within:text-blue-400 transition-colors z-10" />
                      <input
                        type="email"
                        value={email}
                        onChange={(e) => setEmail(e.target.value)}
                        className="w-full pl-12 pr-4 py-3.5 border-none rounded-2xl focus:ring-2 focus:ring-blue-500/40 bg-white/5 text-white placeholder-slate-500 transition-all outline-none text-sm shadow-[inset_0_1px_4px_rgba(0,0,0,0.3)]"
                        placeholder="admin@calculox.com"
                        required
                        disabled={isSubmitting}
                      />
                    </div>
                  </div>

                  <motion.button
                    whileHover={{ scale: 1.01 }}
                    whileTap={{ scale: 0.99 }}
                    type="submit"
                    disabled={isSubmitting || !email}
                    className="w-full mt-3 bg-blue-600 hover:bg-blue-500 text-white font-semibold py-3.5 px-4 rounded-2xl transition-all duration-300 shadow-lg shadow-blue-900/50 disabled:opacity-50 disabled:cursor-not-allowed disabled:shadow-none flex items-center justify-center space-x-2 text-sm border border-blue-500/30"
                  >
                    <AnimatePresence mode="wait">
                      {isSubmitting ? (
                        <motion.div
                          key="loading"
                          initial={{ opacity: 0 }}
                          animate={{ opacity: 1 }}
                          exit={{ opacity: 0 }}
                          className="flex items-center justify-center gap-2"
                        >
                          <span className="w-4 h-4 border-2 border-white/80 border-t-transparent rounded-full animate-spin" />
                          <span className="tracking-wide">Sending...</span>
                        </motion.div>
                      ) : (
                        <motion.div
                          key="text"
                          initial={{ opacity: 0 }}
                          animate={{ opacity: 1 }}
                          exit={{ opacity: 0 }}
                          className="flex items-center justify-center gap-2"
                        >
                          <SafeIcon icon={BiIcons.BiSend} className="w-5 h-5" />
                          <span className="tracking-wide">Send Reset Link</span>
                        </motion.div>
                      )}
                    </AnimatePresence>
                  </motion.button>
                </form>
              </motion.div>

            ) : uiState === "resetForm" ? (
              <motion.div
                key="reset"
                initial={{ opacity: 0, filter: "blur(4px)", scale: 0.98 }}
                animate={{ opacity: 1, filter: "blur(0px)", scale: 1 }}
                exit={{ opacity: 0, filter: "blur(4px)", scale: 0.98 }}
                transition={{ duration: 0.3 }}
              >
                <div className="text-center mb-6">
                  <p className="text-sm text-slate-400 font-medium">
                    Your new password must meet the enterprise security requirements below.
                  </p>
                </div>
                <form onSubmit={handleResetSubmit} className="space-y-4">
                  <div className="space-y-2">
                    <div className="flex items-center justify-between ml-1 mb-2">
                        <label className="block text-[11px] font-bold text-slate-400 uppercase tracking-widest">
                            New Password
                        </label>
                        <span className="text-[10px] font-semibold text-slate-500 uppercase tracking-wider">16 - 64 Chars</span>
                    </div>
                    
                    <div className="relative group">
                      <SafeIcon icon={HiIcons.HiOutlineLockClosed} className="absolute left-4 top-1/2 -translate-y-1/2 w-5 h-5 text-slate-400 group-focus-within:text-blue-400 transition-colors z-10" />
                      <input
                        type={showPassword ? "text" : "password"}
                        value={password}
                        onChange={(e) => handlePasswordChange(e.target.value, setPassword, true)}
                        minLength={16}
                        maxLength={64}
                        className={`w-full pl-12 pr-12 py-3.5 rounded-2xl transition-all outline-none text-sm shadow-[inset_0_1px_4px_rgba(0,0,0,0.3)] ${
                          !hasStartedTyping 
                              ? "border-none bg-white/5 text-white placeholder-slate-500 focus:ring-2 focus:ring-blue-500/40"
                              : isPasswordValid
                                  ? "border border-emerald-500/50 bg-emerald-900/10 text-emerald-300 focus:ring-2 focus:ring-emerald-500/20"
                                  : "border border-rose-500/50 bg-rose-900/10 text-rose-300 focus:ring-2 focus:ring-rose-500/20"
                        }`}
                        placeholder="••••••••••••"
                        required
                        disabled={isSubmitting}
                      />
                      <button
                        type="button"
                        onClick={() => setShowPassword(!showPassword)}
                        className="absolute right-3 top-1/2 -translate-y-1/2 text-slate-400 hover:text-slate-300 transition-colors p-1.5 focus:outline-none"
                        tabIndex="-1"
                      >
                        <SafeIcon icon={showPassword ? HiIcons.HiOutlineEyeOff : HiIcons.HiOutlineEye} className="w-5 h-5" />
                      </button>
                    </div>

                    {password && (
                        <div className="mt-2 ml-1 space-y-1.5">
                            <div className="flex gap-1 h-1.5 mr-1">
                                {[1, 2, 3, 4, 5].map((level) => (
                                    <div 
                                        key={level} 
                                        className={`flex-1 rounded-full transition-colors duration-300 ${
                                            passwordStrength >= level 
                                                ? getStrengthColor(passwordStrength) 
                                                : 'bg-white/10'
                                        }`} 
                                    />
                                ))}
                            </div>
                            <p className={`text-[11px] font-medium transition-colors ${getStrengthTextColor(passwordStrength)}`}>
                                {getStrengthLabel(passwordStrength)}
                            </p>
                        </div>
                    )}

                    <AnimatePresence>
                      {password.length === 64 && (
                        <motion.div
                          initial={{ opacity: 0, height: 0 }}
                          animate={{ opacity: 1, height: "auto" }}
                          exit={{ opacity: 0, height: 0 }}
                        >
                          <p className="text-amber-400 text-[11px] font-medium mt-1.5 ml-1 flex items-center gap-1">
                            <SafeIcon icon={IoIcons.IoWarningOutline} className="w-3.5 h-3.5" />
                            Maximum password length reached (64/64).
                          </p>
                        </motion.div>
                      )}
                    </AnimatePresence>
                  </div>

                  <div className="space-y-2 mt-2">
                    <label className="block text-[11px] font-bold text-slate-400 uppercase tracking-widest ml-1">
                      Confirm Password
                    </label>
                    <div className="relative group">
                      <SafeIcon icon={HiIcons.HiOutlineLockClosed} className="absolute left-4 top-1/2 -translate-y-1/2 w-5 h-5 text-slate-400 group-focus-within:text-blue-400 transition-colors z-10" />
                      <input
                        type={showConfirmPassword ? "text" : "password"}
                        value={confirmPassword}
                        onChange={(e) => handlePasswordChange(e.target.value, setConfirmPassword)}
                        minLength={16}
                        maxLength={64}
                        className={`w-full pl-12 pr-12 py-3.5 rounded-2xl transition-all outline-none text-sm shadow-[inset_0_1px_4px_rgba(0,0,0,0.3)] ${
                          !confirmHasStarted 
                            ? "border-none bg-white/5 text-white placeholder-slate-500 focus:ring-2 focus:ring-blue-500/40"
                            : isMatch
                              ? "border border-emerald-500/50 bg-emerald-900/10 text-emerald-300 focus:ring-2 focus:ring-emerald-500/20"
                              : "border border-rose-500/50 bg-rose-900/10 text-rose-300 focus:ring-2 focus:ring-rose-500/20"
                        }`}
                        placeholder="••••••••••••"
                        required
                        disabled={isSubmitting}
                      />
                      <button
                        type="button"
                        onClick={() => setShowConfirmPassword(!showConfirmPassword)}
                        className="absolute right-3 top-1/2 -translate-y-1/2 text-slate-400 hover:text-slate-300 transition-colors p-1.5 focus:outline-none"
                        tabIndex="-1"
                      >
                        <SafeIcon icon={showConfirmPassword ? HiIcons.HiOutlineEyeOff : HiIcons.HiOutlineEye} className="w-5 h-5" />
                      </button>
                    </div>
                  </div>

                  <motion.button
                    whileHover={{ scale: 1.01 }}
                    whileTap={{ scale: 0.99 }}
                    type="submit"
                    disabled={isSubmitting || !isPasswordValid || !isMatch || passwordStrength < 3}
                    className="w-full mt-5 bg-blue-600 hover:bg-blue-500 text-white font-semibold py-3.5 px-4 rounded-2xl transition-all duration-300 shadow-lg shadow-blue-900/50 disabled:opacity-50 disabled:cursor-not-allowed disabled:shadow-none flex items-center justify-center space-x-2 text-sm border border-blue-500/30"
                  >
                    <AnimatePresence mode="wait">
                      {isSubmitting ? (
                        <motion.div
                          key="loading"
                          initial={{ opacity: 0 }}
                          animate={{ opacity: 1 }}
                          exit={{ opacity: 0 }}
                          className="flex items-center justify-center gap-2"
                        >
                          <span className="w-4 h-4 border-2 border-white/80 border-t-transparent rounded-full animate-spin" />
                          <span className="tracking-wide">Updating Password...</span>
                        </motion.div>
                      ) : (
                        <motion.div
                          key="text"
                          initial={{ opacity: 0 }}
                          animate={{ opacity: 1 }}
                          exit={{ opacity: 0 }}
                          className="flex items-center justify-center gap-2"
                        >
                          <SafeIcon icon={IoIcons.IoLockClosedOutline} className="w-5 h-5" />
                          <span className="tracking-wide">Secure & Save Password</span>
                        </motion.div>
                      )}
                    </AnimatePresence>
                  </motion.button>
                </form>
              </motion.div>

            ) : uiState === "securing" ? (
              // [NEW] The Enterprise Labor Gap State (Dark UI match)
              <motion.div
                  key="securing"
                  initial={{ opacity: 0, filter: "blur(4px)", scale: 0.98 }}
                  animate={{ opacity: 1, filter: "blur(0px)", scale: 1 }}
                  exit={{ opacity: 0, filter: "blur(4px)", scale: 0.98 }}
                  transition={{ duration: 0.3 }}
                  className="text-center py-6"
              >
                  <div className="flex flex-col items-center justify-center space-y-6">
                      <div className="relative">
                          <div className="w-20 h-20 bg-emerald-900/20 rounded-full flex items-center justify-center text-emerald-400 mb-2 border border-emerald-500/30 shadow-[inset_0_1px_4px_rgba(0,0,0,0.3)]">
                              <SafeIcon icon={FiIcons.FiShield} className="w-10 h-10" />
                          </div>
                          <motion.div 
                              initial={{ scale: 0 }}
                              animate={{ scale: 1 }}
                              transition={{ delay: 0.3, type: "spring" }}
                              className="absolute -bottom-1 -right-1 w-8 h-8 bg-emerald-500 rounded-full border-4 border-[#0B1221] flex items-center justify-center text-white"
                          >
                              <SafeIcon icon={FiIcons.FiCheck} className="w-4 h-4" />
                          </motion.div>
                      </div>

                      <div>
                          <h3 className="text-xl font-bold text-white">Credentials Secured</h3>
                          <p className="text-sm text-slate-400 mt-2 font-medium">
                              Validating keys and securing session...
                          </p>
                      </div>

                      <div className="w-full max-w-[200px] h-1.5 bg-slate-800 rounded-full overflow-hidden mt-4 mx-auto shadow-inner">
                          <motion.div 
                              initial={{ width: "0%" }}
                              animate={{ width: "100%" }}
                              transition={{ duration: 2.2, ease: "easeInOut" }}
                              className="h-full bg-emerald-500 rounded-full"
                          />
                      </div>
                  </div>
              </motion.div>

            ) : uiState === "success" || uiState === "emailSent" ? (
              <motion.div
                key="success"
                initial={{ opacity: 0, filter: "blur(4px)", scale: 0.98 }}
                animate={{ opacity: 1, filter: "blur(0px)", scale: 1 }}
                exit={{ opacity: 0, filter: "blur(4px)", scale: 0.98 }}
                transition={{ duration: 0.3 }}
                className="text-center py-6"
              >
                <div className="flex flex-col items-center justify-center space-y-4">
                  <div className="w-16 h-16 bg-emerald-900/20 rounded-full flex items-center justify-center mb-2 border border-emerald-500/30 shadow-[inset_0_1px_4px_rgba(0,0,0,0.3)]">
                     <SafeIcon icon={IoIcons.IoCheckmarkCircleOutline} className="w-8 h-8 text-emerald-400" />
                  </div>
                  <h1 className="text-xl font-bold text-white tracking-tight mb-2">
                    {uiState === "emailSent" ? "Check Your Email" : "Password Updated"}
                   </h1>
                  <p className="text-sm text-slate-400 max-w-[260px] mx-auto font-medium leading-relaxed">
                    {uiState === "emailSent"
                      ? `We've sent password reset instructions to ${email}.`
                      : "Your account is secure. You can now log in with your new password."}
                  </p>
                </div>
              </motion.div>
             ) : null}
          </AnimatePresence>

          {/* Footer Notice (Matched to Login.jsx) */}
          <div className="mt-6 pt-5 border-t border-slate-700/40 text-center">
             <Link
              to="/login"
              className="text-[11px] font-bold uppercase tracking-widest text-slate-400 hover:text-slate-200 transition-colors flex items-center justify-center gap-2 mx-auto group mb-4"
            >
              <SafeIcon icon={HiIcons.HiArrowLeft} className="w-3.5 h-3.5 transition-transform duration-300 group-hover:-translate-x-1.5" />
              Back to login
            </Link>
            <p className="text-[10px] text-slate-600">
              &copy; {new Date().getFullYear()} {siteName} &mdash; Administrator Control Panel
            </p>
          </div>
          
        </div>
      </motion.div>
    </div>
  );
};

export default ResetPassword;
