<?php
// File: calcpro-api/api/admin/reset_admin_password.php

// Use robust, absolute paths
require_once dirname(__DIR__, 2) . '/core/config.php';
require_once dirname(__DIR__, 2) . '/core/response.php';

try {
    // --- 1. Get Incoming Data ---
    $json = file_get_contents('php://input');
    $data = json_decode($json);

    // --- 2. Validation ---
    if (!isset($data->token) || empty($data->token)) {
        sendResponse(false, null, 'A valid reset token is required.', 400, 'INVALID_INPUT');
    }
    if (!isset($data->password) || empty($data->password)) {
        sendResponse(false, null, 'A new password is required.', 400, 'INVALID_INPUT');
    }
    
    // Explicitly cast to string
    $new_password = (string)$data->password;

    // [FIX #3] THE OVERWORKED SERVER: Prevent CPU Exhaustion (Denial of Service)
    if (strlen($new_password) > 64) {
        sendResponse(false, null, 'Password is too long. Maximum allowed length is 64 characters.', 400, 'PASSWORD_TOO_LONG');
    }

    // --- 3. ENTERPRISE FIX: Inline Password Strength Validation ---
    // [UPDATED] Strict 16-character minimum (NIST Guidelines - Length over Complexity)
    if (strlen($new_password) < 16) {
        sendResponse(false, null, 'Password must be at least 16 characters long.', 400, 'WEAK_PASSWORD');
    }

    // --- 4. Sanitize and Prepare Data ---
    $token = (string)$data->token;
    $token_hash = hash('sha256', $token);

    // --- 5. Find User by Token in the Rate-Limiting Table ---
    $sql_find = "SELECT admin_id, expires_at FROM admin_password_resets WHERE token_hash = :token_hash LIMIT 1";
    $stmt_find = $conn->prepare($sql_find);
    $stmt_find->execute([':token_hash' => $token_hash]);
    $reset_record = $stmt_find->fetch(PDO::FETCH_ASSOC);
    
    if (!$reset_record) {
        // Token doesn't exist, was deleted, or was already used
        sendResponse(false, null, 'This password reset link is invalid. Please try again.', 400, 'INVALID_TOKEN');
    }

    // Strictly enforce UTC timezone evaluation to prevent expiration drift bugs
    $utcZone = new DateTimeZone('UTC');
    $expiresAt = new DateTime($reset_record['expires_at'], $utcZone);
    $now = new DateTime('now', $utcZone);
    
    if ($expiresAt < $now) {
        sendResponse(false, null, 'This password reset link has expired. Please request a new one.', 400, 'EXPIRED_TOKEN');
    }

    $admin_id = $reset_record['admin_id'];

    // --- 6. Transaction: Update Password AND Hard Kill Token ---
    $new_hashed_password = password_hash($new_password, PASSWORD_BCRYPT);
    $conn->beginTransaction();

    try {
        // A. Update the admin's password in the main admins table AND trigger the "Changed Locks"
        // ENTERPRISE FIX: Nullify current_session_jti to instantly kill active sessions
        $sql_update_pwd = "UPDATE admins SET password = :password, password_last_updated = CURRENT_TIMESTAMP, current_session_jti = NULL WHERE id = :id";
        $stmt_update_pwd = $conn->prepare($sql_update_pwd);
        $stmt_update_pwd->execute([
            ':password' => $new_hashed_password,
            ':id' => $admin_id
        ]);

        // B. THE HARD KILL: Destroy all pending reset tokens for this admin completely.
        $sql_hard_kill = "DELETE FROM admin_password_resets WHERE admin_id = :admin_id";
        $stmt_hard_kill = $conn->prepare($sql_hard_kill);
        $stmt_hard_kill->execute([':admin_id' => $admin_id]);

        // C. ORPHANED TRUST TOKEN PURGE: Destroy all trusted devices to enforce 2FA globally
        $sql_purge_trust = "DELETE FROM admin_trusted_devices WHERE admin_id = :admin_id";
        $stmt_purge_trust = $conn->prepare($sql_purge_trust);
        $stmt_purge_trust->execute([':admin_id' => $admin_id]);

        $conn->commit();
        sendResponse(true, null, 'Password has been reset successfully. You can now log in.');

    } catch (Exception $e) {
        $conn->rollBack();
        throw new Exception("Database password update failed: " . $e->getMessage());
    }

} catch (Exception $e) {
    error_log("Error in reset_admin_password.php: " . $e->getMessage());
    sendResponse(false, null, 'An internal server error occurred.', 500, 'SERVER_ERROR');
}

$conn = null;
?>
