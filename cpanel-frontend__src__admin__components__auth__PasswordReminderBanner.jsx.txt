// cPanel_Frontend/src/admin/components/auth/PasswordReminderBanner.jsx
import React, { useState } from 'react';
import { useAuth } from "../../context/AuthContext";
import SafeIcon from "../../../common/SafeIcon";
import * as FiIcons from "react-icons/fi";
import { motion, AnimatePresence } from 'framer-motion';

const PasswordReminderBanner = ({ onUpdateClick }) => {
    const { passwordSecurity } = useAuth();
    const [isDismissed, setIsDismissed] = useState(false);

    // FIX 1 & 3: Determine visibility BEFORE JSX so AnimatePresence can track mount/unmount.
    // Also removed `passwordSecurity.isExpired` from the bail-out — when expired, we show
    // an urgent (red) variant instead of hiding the banner entirely.
    const shouldShow =
        passwordSecurity &&
        passwordSecurity.showWarning &&
        !isDismissed;

    const isExpired = passwordSecurity?.isExpired;
    const days = passwordSecurity?.daysRemaining;

    // Colour tokens: amber for warning, red for expired
    const colors = isExpired
        ? {
            border:       'border-red-500/20',
            shadow:       'shadow-[0_4px_30px_rgba(239,68,68,0.08)]',
            ping:         'bg-red-400/20',
            iconBg:       'bg-red-100/80 dark:bg-red-500/20',
            iconBorder:   'border-red-200/50 dark:border-red-500/30',
            icon:         'text-red-600 dark:text-red-400',
            label:        'text-red-600 dark:text-red-400',
            btnBorder:    'border-red-200 dark:border-red-700/50',
            btnText:      'text-red-900 dark:text-red-300',
            btnBg:        'from-red-100 to-red-200 hover:from-red-200 hover:to-red-300 dark:from-red-500/20 dark:to-red-600/20 dark:hover:from-red-500/30 dark:hover:to-red-600/30',
          }
        : {
            border:       'border-amber-500/20',
            shadow:       'shadow-[0_4px_30px_rgba(245,158,11,0.05)]',
            ping:         'bg-amber-400/20',
            iconBg:       'bg-amber-100/80 dark:bg-amber-500/20',
            iconBorder:   'border-amber-200/50 dark:border-amber-500/30',
            icon:         'text-amber-600 dark:text-amber-400',
            label:        'text-amber-600 dark:text-amber-400',
            btnBorder:    'border-amber-200 dark:border-amber-700/50',
            btnText:      'text-amber-900 dark:text-amber-300',
            btnBg:        'from-amber-100 to-amber-200 hover:from-amber-200 hover:to-amber-300 dark:from-amber-500/20 dark:to-amber-600/20 dark:hover:from-amber-500/30 dark:hover:to-amber-600/30',
          };

    const shortMsg  = isExpired
        ? 'Your credentials have expired.'
        : `Credentials expire in ${days} ${days === 1 ? 'day' : 'days'}.`;

    const longLabel = isExpired ? 'Immediate Action Required:' : 'Action Recommended:';
    const longMsg   = isExpired
        ? 'Your administrator credentials have expired. Please rotate them immediately to restore access.'
        : `Your administrator credentials expire in ${days} ${days === 1 ? 'day' : 'days'}. Proactive rotation is advised.`;

    return (
        // FIX 1: AnimatePresence now wraps the conditional render in JSX,
        // so Framer Motion can observe the child mounting/unmounting and
        // correctly trigger enter/exit animations.
        <AnimatePresence>
            {shouldShow && (
                <motion.div
                    initial={{ height: 0, opacity: 0 }}
                    animate={{ height: 'auto', opacity: 1 }}
                    exit={{ height: 0, opacity: 0 }}
                    transition={{ duration: 0.25, ease: 'easeInOut' }}
                    // FIX 2: `overflow-hidden` removed from this element.
                    // It was clipping content while height animated from 0,
                    // making the banner invisible during the enter transition.
                    // A wrapper div below handles overflow clipping safely.
                    className={`bg-white/60 dark:bg-[#0B1221]/60 backdrop-blur-xl border-b ${colors.border} ${colors.shadow} relative z-50`}
                >
                    {/* This inner wrapper is the correct place for overflow-hidden */}
                    <div className="overflow-hidden">
                        {/* Subtle animated background gradient */}
                        <div className="absolute inset-0 bg-gradient-to-r from-transparent via-amber-500/5 to-transparent animate-[shimmer_8s_infinite_linear] pointer-events-none" />

                        <div className="max-w-[1400px] mx-auto py-3 px-4 sm:px-6 lg:px-8 relative">
                            <div className="flex items-center justify-between flex-wrap gap-3">
                                <div className="flex flex-1 items-center gap-3.5 min-w-0">
                                    <div className="relative flex items-center justify-center shrink-0">
                                        <span className={`absolute inset-0 rounded-full ${colors.ping} animate-ping`} />
                                        <span className={`relative flex p-1.5 rounded-lg ${colors.iconBg} border ${colors.iconBorder} shadow-sm`}>
                                            <SafeIcon
                                                icon={isExpired ? FiIcons.FiAlertOctagon : FiIcons.FiAlertTriangle}
                                                className={`h-4 w-4 ${colors.icon}`}
                                            />
                                        </span>
                                    </div>
                                    <p className="font-bold text-slate-800 dark:text-slate-200 truncate text-[13px] tracking-wide">
                                        <span className="md:hidden">{shortMsg}</span>
                                        <span className="hidden md:inline">
                                            <span className={`${colors.label} mr-1.5`}>{longLabel}</span>
                                            {longMsg}
                                        </span>
                                    </p>
                                </div>
                                <div className="order-3 mt-2 flex-shrink-0 w-full sm:order-2 sm:mt-0 sm:w-auto flex items-center gap-3">
                                    <button
                                        onClick={onUpdateClick}
                                        className={`flex items-center justify-center px-4 py-1.5 border ${colors.btnBorder} rounded-lg shadow-sm text-[12px] font-bold ${colors.btnText} bg-gradient-to-b ${colors.btnBg} transition-all`}
                                    >
                                        Rotate Credentials
                                    </button>
                                    <button
                                        onClick={() => setIsDismissed(true)}
                                        className="p-1.5 rounded-md text-slate-400 hover:text-slate-600 hover:bg-slate-100/50 dark:hover:text-slate-200 dark:hover:bg-slate-800/50 transition-colors focus:outline-none"
                                        aria-label="Dismiss"
                                    >
                                        <SafeIcon icon={FiIcons.FiX} className="h-4 w-4" />
                                    </button>
                                </div>
                            </div>
                        </div>
                    </div>
                </motion.div>
            )}
        </AnimatePresence>
    );
};

export default PasswordReminderBanner;
