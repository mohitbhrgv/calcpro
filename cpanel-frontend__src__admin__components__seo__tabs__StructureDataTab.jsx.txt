import React, { useState, useEffect } from "react";
import SafeIcon from "../../../../common/SafeIcon";
import toast from "react-hot-toast";
import * as FiIcons from "react-icons/fi";

const { FiLayout, FiFileText, FiCode, FiCheckCircle, FiAlertTriangle, FiClipboard, FiInfo } = FiIcons;

const StructureDataTab = ({ settings, handleCustomChange, pagesList }) => {
  const [jsonInput, setJsonInput] = useState("");
  const [isValid, setIsValid] = useState(true);
  const [errorMessage, setErrorMessage] = useState("");

  // Initialize from settings or Default Template
  useEffect(() => {
    if (settings.seo_website_json_ld) {
      setJsonInput(settings.seo_website_json_ld);
      validateJson(settings.seo_website_json_ld);
    } else {
      // Default Template matching the Competitor's Clean Structure
      const defaultTemplate = JSON.stringify({
        "@context": "https://schema.org",
        "@type": "WebSite",
        "name": "%sitename%",
        "url": "%siteurl%",
        "description": "%sitedesc%"
      }, null, 2);
      
      setJsonInput(defaultTemplate);
    }
  }, [settings.seo_website_json_ld]);

  const validateJson = (value) => {
    if (!value || value.trim() === "") {
        setIsValid(true);
        setErrorMessage("");
        return true;
    }
    try {
      JSON.parse(value);
      setIsValid(true);
      setErrorMessage("");
      return true;
    } catch (e) {
      setIsValid(false);
      const shortError = e.message.split(',')[0]; 
      setErrorMessage(shortError);
      return false;
    }
  };

  const handleChange = (e) => {
    let val = e.target.value;

    // --- SMART CLEANER: Auto-strip <script> tags ---
    if (val.includes("<script")) {
        const cleaned = val.replace(/<script[^>]*>([\s\S]*?)<\/script>/gi, "$1").trim();
        if (cleaned !== val) {
            val = cleaned;
            toast.success("Removed <script> tags automatically.", { icon: '🧹' });
        }
    }

    setJsonInput(val);
    validateJson(val);
    
    // Update Global Settings
    handleCustomChange("seo_website_json_ld", val);
  };

  const copyVariable = (text) => {
    navigator.clipboard.writeText(text);
    toast.success(`Copied ${text}`, { id: 'copy-variable' });
  };

  const variables = [
    { label: "Site Name", code: "%sitename%", desc: "Global Site Name" },
    { label: "Site URL", code: "%siteurl%", desc: "Canonical Frontend URL" },
    { label: "Site Description", code: "%sitedesc%", desc: "Global Meta Description" }
  ];

  return (
    <div className="animate-fade-in max-w-5xl">
      {/* --- Header Card --- */}
      <div className="flex items-start gap-4 p-4 mb-8 rounded-lg bg-indigo-50 dark:bg-indigo-900/20 border border-indigo-100 dark:border-indigo-800/50">
        <div className="p-2 bg-indigo-100 dark:bg-indigo-800 rounded-lg text-indigo-600 dark:text-indigo-300 shrink-0">
          <SafeIcon icon={FiLayout} className="w-5 h-5" />
        </div>
        <div>
          {/* UPDATED HEADER TEXT */}
          <h4 className="font-bold text-gray-900 dark:text-white">Identity & Structure</h4>
          <p className="text-sm text-gray-600 dark:text-gray-300 mt-1 leading-relaxed">
            Define your website's global identity (<strong>WebSite</strong> schema) and semantically map core pages 
            (<strong>AboutPage</strong>, <strong>ContactPage</strong>). This helps search engines understand your site's hierarchy, purpose, and official brand assets.
          </p>
        </div>
      </div>

      <div className="space-y-8">
        
        {/* --- Section 1: Semantic Page Mapping --- */}
        <div className="bg-white dark:bg-gray-800 rounded-xl border border-gray-200 dark:border-gray-700 overflow-hidden shadow-sm">
            <div className="px-6 py-4 border-b border-gray-100 dark:border-gray-700 bg-gray-50 dark:bg-gray-700/50 flex justify-between items-center">
                <div className="flex items-center gap-2">
                    <SafeIcon icon={FiFileText} className="text-blue-500" />
                    <h3 className="text-sm font-bold text-gray-900 dark:text-white uppercase tracking-wide">Semantic Page Mapping</h3>
                </div>
            </div>
            
            <div className="p-6 space-y-6">
                {/* About Page */}
                <div className="flex flex-col md:flex-row md:gap-8">
                    <div className="w-full md:w-1/3">
                        <label className="block text-sm font-semibold text-gray-800 dark:text-gray-200 mb-1">
                           Select About Page
                        </label>
                        <p className="text-xs text-gray-500 dark:text-gray-400">
                            Adds <code>@type: AboutPage</code> schema.
                        </p>
                    </div>
                    <div className="w-full md:w-2/3">
                        <select 
                            value={settings.seo_page_for_about || ''}
                            onChange={(e) => handleCustomChange('seo_page_for_about', e.target.value)}
                            className="w-full px-4 py-2 border border-gray-300 dark:border-gray-600 rounded-lg dark:bg-gray-700 dark:text-white focus:ring-2 focus:ring-blue-500 outline-none"
                        >
                            <option value="">-- None --</option>
                            {pagesList.map(page => (
                                <option key={page.id} value={page.slug}>{page.title} (/{page.slug})</option>
                            ))}
                        </select>
                    </div>
                </div>

                <hr className="border-gray-100 dark:border-gray-700" />

                {/* Contact Page */}
                <div className="flex flex-col md:flex-row md:gap-8">
                    <div className="w-full md:w-1/3">
                        <label className="block text-sm font-semibold text-gray-800 dark:text-gray-200 mb-1">
                           Select Contact Page
                        </label>
                        <p className="text-xs text-gray-500 dark:text-gray-400">
                            Adds <code>@type: ContactPage</code> schema.
                        </p>
                    </div>
                    <div className="w-full md:w-2/3">
                        <select 
                            value={settings.seo_page_for_contact || ''}
                            onChange={(e) => handleCustomChange('seo_page_for_contact', e.target.value)}
                            className="w-full px-4 py-2 border border-gray-300 dark:border-gray-600 rounded-lg dark:bg-gray-700 dark:text-white focus:ring-2 focus:ring-blue-500 outline-none"
                        >
                            <option value="">-- None --</option>
                            {pagesList.map(page => (
                                <option key={page.id} value={page.slug}>{page.title} (/{page.slug})</option>
                            ))}
                        </select>
                    </div>
                </div>
            </div>
        </div>

        {/* --- Section 2: Website Schema Editor (Standardized UI) --- */}
        <div className="bg-white dark:bg-gray-800 rounded-xl border border-gray-200 dark:border-gray-700 overflow-hidden shadow-sm">
            {/* UPDATED HEADER: Matches "Semantic Page Mapping" style exactly 
                - Same background, padding, border-b, and uppercase text style
            */}
            <div className="px-6 py-4 border-b border-gray-100 dark:border-gray-700 bg-gray-50 dark:bg-gray-700/50 flex justify-between items-center">
                <div className="flex items-center gap-2">
                    <SafeIcon icon={FiCode} className="text-blue-500" />
                    <h3 className="text-sm font-bold text-gray-900 dark:text-white uppercase tracking-wide">Website Schema Editor</h3>
                </div>
            </div>

            <div className="p-6">
                {/* UPDATED DESCRIPTION TEXT */}
                <p className="text-sm text-gray-500 dark:text-gray-400 mb-6 leading-relaxed max-w-4xl">
                   Configure the official <strong>WebSite</strong> schema to establish authoritative brand identity in Knowledge Graphs and search results. This modern implementation replaces deprecated Sitelinks logic with a robust, standard-compliant identity definition.
                </p>

                <div className="grid grid-cols-1 lg:grid-cols-3 gap-8">
                    
                    {/* Left: Editor */}
                    <div className="lg:col-span-2 space-y-4">
                        <div className={`relative rounded-xl border transition-colors bg-white dark:bg-gray-800 ${isValid ? 'border-gray-300 dark:border-gray-600 focus-within:border-blue-500' : 'border-red-500 focus-within:border-red-500'}`}>
                            {/* Inner Header for Editor Status */}
                            <div className="flex justify-between items-center px-4 py-2 border-b border-gray-100 dark:border-gray-700 bg-gray-50 dark:bg-gray-700/50 rounded-t-xl">
                                <span className="text-xs font-bold text-gray-500 dark:text-gray-400 uppercase tracking-wider">JSON-LD Source</span>
                                {isValid ? (
                                    <span className="text-xs text-green-600 dark:text-green-400 flex items-center gap-1">
                                        <SafeIcon icon={FiCheckCircle} className="w-3 h-3" /> Valid
                                    </span>
                                ) : (
                                    <span className="text-xs text-red-600 dark:text-red-400 flex items-center gap-1">
                                        <SafeIcon icon={FiAlertTriangle} className="w-3 h-3" /> Invalid
                                    </span>
                                )}
                            </div>
                            <textarea 
                                value={jsonInput}
                                onChange={handleChange}
                                className="w-full h-80 p-4 font-mono text-sm bg-transparent border-none outline-none resize-y text-gray-800 dark:text-gray-200 placeholder-gray-400 dark:placeholder-gray-500"
                                spellCheck="false"
                                placeholder="{ ... }"
                            />
                        </div>
                        
                        {!isValid && (
                            <div className="p-3 bg-red-50 dark:bg-red-900/20 text-red-700 dark:text-red-200 text-xs rounded-lg font-mono flex items-start gap-2 border border-red-100 dark:border-red-800">
                                <SafeIcon icon={FiAlertTriangle} className="w-4 h-4 shrink-0 mt-0.5" />
                                <div>
                                    <strong>Syntax Error:</strong> {errorMessage}.
                                    <br/>Ensure you are pasting <em>only</em> the JSON object.
                                </div>
                            </div>
                        )}
                    </div>

                    {/* Right: Variables Helper */}
                    <div className="space-y-6">
                        <div className="bg-white dark:bg-gray-800 rounded-xl border border-gray-200 dark:border-gray-700 overflow-hidden shadow-sm">
                            <div className="p-4 border-b border-gray-100 dark:border-gray-700 bg-gray-50 dark:bg-gray-700/50">
                                <h5 className="font-bold text-gray-900 dark:text-white flex items-center gap-2 text-sm">
                                    <SafeIcon icon={FiInfo} className="w-4 h-4 text-blue-500" />
                                    Dynamic Variables
                                </h5>
                            </div>
                            <div className="p-2">
                                {variables.map((v) => (
                                    <button
                                        key={v.code}
                                        onClick={() => copyVariable(v.code)}
                                        className="w-full flex items-center justify-between p-3 hover:bg-gray-50 dark:hover:bg-gray-700/50 rounded-lg group transition-colors text-left"
                                    >
                                        <div>
                                            <code className="text-xs font-bold text-blue-600 dark:text-blue-400 bg-blue-50 dark:bg-blue-900/30 px-1.5 py-0.5 rounded">
                                                {v.code}
                                            </code>
                                            <p className="text-xs text-gray-500 dark:text-gray-400 mt-1">
                                                {v.desc}
                                            </p>
                                        </div>
                                        <SafeIcon icon={FiClipboard} className="w-4 h-4 text-gray-300 dark:text-gray-500 group-hover:text-blue-500 transition-colors" />
                                    </button>
                                ))}
                            </div>
                            <div className="p-4 bg-gray-50 dark:bg-gray-700/30 border-t border-gray-100 dark:border-gray-700 text-xs text-gray-500 dark:text-gray-400 leading-relaxed">
                            This schema is injected on the <strong>Homepage</strong>. These placeholders are replaced with your live settings on render.
                            </div>
                        </div>
                    </div>

                </div>
            </div>
        </div>

      </div>
    </div>
  );
};

export default StructureDataTab;
