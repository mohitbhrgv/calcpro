<?php
// Core utilities
require_once '../../core/config.php';
require_once '../../core/response.php';
require_once '../../core/auth.php';

// --- SECURITY CHECK ---
$admin_data = verifyAdminToken();

try {
    // --- Get Incoming Data ---
    $json = file_get_contents('php://input');
    $data = json_decode($json);

    // --- Validation ---
    if (!isset($data->log_type) || !isset($data->id) || !is_numeric($data->id)) {
        sendResponse(false, null, 'A log type and a numeric ID are required.', 400, 'INVALID_INPUT');
    }

    $log_type = $data->log_type;
    $log_id = (int)$data->id;

    // --- Whitelist of allowed tables ---
    $allowed_tables = [
        'login_attempts'       => 'Failed Email Log',
        'admin_login_failures' => 'Failed Password Log',
        'temporary_blocks'     => 'Temporary Block Log'
    ];

    if (!array_key_exists($log_type, $allowed_tables)) {
        sendResponse(false, null, 'Invalid log type specified.', 400, 'INVALID_LOG_TYPE');
    }
    
    $table_name = $log_type;
    $log_name_friendly = $allowed_tables[$log_type];

    // --- Begin Transaction for Data Integrity ---
    $conn->beginTransaction();

    // --- Fetch the IP address from the log entry being deleted ---
    $ip_address = null;
    // We can safely interpolate $table_name here because it was explicitly whitelisted above
    $stmt_get_ip = $conn->prepare("SELECT ip_address FROM `{$table_name}` WHERE id = :id");
    $stmt_get_ip->execute([':id' => $log_id]);
    
    if ($row = $stmt_get_ip->fetch(PDO::FETCH_ASSOC)) {
        $ip_address = $row['ip_address'];
    }

    // --- Main Deletion ---
    $stmt_delete = $conn->prepare("DELETE FROM `{$table_name}` WHERE id = :id");
    $stmt_delete->execute([':id' => $log_id]);
    $affected_rows = $stmt_delete->rowCount();

    if ($affected_rows > 0) {
        // --- CASCADING LOGIC ---
        // If we delete a temporary block, we clear their recent attempts so they can try again.
        // If we delete an attempt, we don't automatically unblock them, we just clean up the secondary records.
        if ($ip_address) {
            if ($log_type === 'temporary_blocks') {
                $stmtDelLogin = $conn->prepare("DELETE FROM login_attempts WHERE ip_address = :ip");
                $stmtDelLogin->execute([':ip' => $ip_address]);
                
                $stmtDelAdmin = $conn->prepare("DELETE FROM admin_login_failures WHERE ip_address = :ip");
                $stmtDelAdmin->execute([':ip' => $ip_address]);
            } else {
                $stmtDelTemp = $conn->prepare("DELETE FROM temporary_blocks WHERE ip_address = :ip");
                $stmtDelTemp->execute([':ip' => $ip_address]);
            }
        }
        $conn->commit();
        
        sendResponse(true, null, "{$log_name_friendly} entry deleted successfully.", 200);
    } else {
        $conn->rollBack();
        sendResponse(false, null, "{$log_name_friendly} entry not found or already deleted.", 404, 'NOT_FOUND');
    }

} catch (Exception $e) {
    if ($conn->inTransaction()) {
        $conn->rollBack();
    }
    error_log("Error in delete_security_log.php: " . $e->getMessage());
    sendResponse(false, null, 'An internal server error occurred.', 500, 'SERVER_ERROR');
}

$conn = null;
?>
