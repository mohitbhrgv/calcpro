// cPanel_Frontend/src/admin/components/auth/ForcePasswordUpdateModal.jsx
import React, { useState, useEffect, useRef } from "react";
import toast from "react-hot-toast";
import SafeIcon from "../../../common/SafeIcon";
import * as FiIcons from "react-icons/fi";
import { authenticatedFetch } from "../../utils/apiService";
import { motion, AnimatePresence } from "framer-motion";

const VERIFICATION_MESSAGES = [
    "Authenticating secure session...",
    "Evaluating cryptographic lifecycle...",
    "Rotation protocol required."
];

const ForcePasswordUpdateModal = ({ isOpen, onSuccess, onLogout, timeLeft }) => {
    const [currentPassword, setCurrentPassword] = useState("");
    const [newPassword, setNewPassword] = useState("");
    const [confirmPassword, setConfirmPassword] = useState("");
    
    const [isVerifying, setIsVerifying] = useState(true);
    const [verificationStep, setVerificationStep] = useState(0);
    const [loading, setLoading] = useState(false);
    const [statusText, setStatusText] = useState("");
    const [error, setError] = useState(null);

    // Synchronized State for Enterprise UX
    const [passwordStrength, setPasswordStrength] = useState(0);
    const [isSuccessState, setIsSuccessState] = useState(false);

    const abortControllerRef = useRef(null);
    const firstInputRef = useRef(null);
    const modalRef = useRef(null); 
    const timersRef = useRef(new Set());

    useEffect(() => {
        if (!isOpen) {
            setCurrentPassword("");
            setNewPassword("");
            setConfirmPassword("");
            setError(null);
            setLoading(false);
            setIsVerifying(true);
            setVerificationStep(0);
            setStatusText("");
            setPasswordStrength(0);
            setIsSuccessState(false);
            
            if (abortControllerRef.current) abortControllerRef.current.abort();
            
            timersRef.current.forEach(clearTimeout);
            timersRef.current.clear();
        } else {
            toast.dismiss(); 
            setIsVerifying(true);
            setVerificationStep(0);
            setStatusText("");
        }
    }, [isOpen]);

    useEffect(() => {
        if (isOpen && isVerifying) {
            const t1 = setTimeout(() => setVerificationStep(1), 800);
            const t2 = setTimeout(() => setVerificationStep(2), 1600);
            const t3 = setTimeout(() => setIsVerifying(false), 2400);

            return () => {
                clearTimeout(t1);
                clearTimeout(t2);
                clearTimeout(t3);
            };
        }
    }, [isOpen, isVerifying]);

    useEffect(() => {
        if (isOpen && !isVerifying && !isSuccessState) {
            const tId = setTimeout(() => {
                if (firstInputRef.current) firstInputRef.current.focus();
            }, 100);
            return () => clearTimeout(tId);
        }
    }, [isOpen, isVerifying, isSuccessState]);

    useEffect(() => {
        const handleKeyDown = (e) => {
            if (!isOpen || isVerifying || e.key !== 'Tab' || !modalRef.current) return;

            const focusableElements = modalRef.current.querySelectorAll(
                'button:not([disabled]), [href], input:not([disabled]), select:not([disabled]), textarea:not([disabled]), [tabindex]:not([tabindex="-1"])'
            );
            
            if (focusableElements.length === 0) return;

            const firstElement = focusableElements[0];
            const lastElement = focusableElements[focusableElements.length - 1];

            if (e.shiftKey) {
                if (document.activeElement === firstElement || document.activeElement === document.body) {
                    e.preventDefault();
                    lastElement.focus();
                }
            } else {
                if (document.activeElement === lastElement || document.activeElement === document.body) {
                    e.preventDefault();
                    firstElement.focus();
                }
            }
        };
        document.addEventListener('keydown', handleKeyDown);
        return () => document.removeEventListener('keydown', handleKeyDown);
    }, [isOpen, isVerifying]);

    useEffect(() => {
        return () => {
            if (abortControllerRef.current) abortControllerRef.current.abort();
            timersRef.current.forEach(clearTimeout);
            timersRef.current.clear();
        };
    }, []);

    if (!isOpen) return null;

    const calculateStrength = (pass) => {
        let strength = 0;
        if (pass.length >= 16 && pass.length <= 64) strength += 1; 
        if (pass.match(/[A-Z]/)) strength += 1;
        if (pass.match(/[a-z]/)) strength += 1;
        if (pass.match(/[0-9]/)) strength += 1;
        if (pass.match(/[^A-Za-z0-9]/)) strength += 1;
        return strength;
    };

    const getStrengthColor = (strength) => {
        if (strength <= 2) return 'bg-red-500';
        if (strength === 3) return 'bg-amber-500';
        if (strength === 4) return 'bg-blue-500';
        return 'bg-emerald-500';
    };

    const getStrengthLabel = (strength) => {
        if (strength === 0) return '';
        if (strength <= 2) return 'Weak - Add uppercase, numbers, or symbols';
        if (strength === 3) return 'Fair - Could be stronger';
        if (strength === 4) return 'Good - Almost there';
        return 'Strong - Excellent password';
    };

    const getStrengthTextColor = (strength) => {
        if (strength <= 2) return 'text-red-600 dark:text-red-400';
        if (strength === 3) return 'text-amber-600 dark:text-amber-400';
        if (strength === 4) return 'text-blue-600 dark:text-blue-400';
        return 'text-emerald-600 dark:text-emerald-400';
    };

    const handlePasswordChange = (value, setter, isNewPwd = false) => {
        let finalValue = value;
        if (value.length > 64) {
            finalValue = value.slice(0, 64);
        }
        setter(finalValue);
        if (isNewPwd) {
            setPasswordStrength(calculateStrength(finalValue));
        }
    };

    const hasStartedTyping = newPassword.length > 0;
    const isPasswordValid = newPassword.length >= 16 && newPassword.length <= 64;
    
    const confirmHasStarted = confirmPassword.length > 0;
    const isMatch = newPassword === confirmPassword;

    const handleSubmit = async (e) => {
        e.preventDefault();
        setError(null);

        if (currentPassword === newPassword) {
            setError("New password must be different from your current password.");
            return;
        }

        if (!isMatch) {
            setError("New passwords do not match.");
            return;
        }

        if (!isPasswordValid) {
            setError("Password must be between 16 and 64 characters long.");
            return;
        }

        if (passwordStrength < 3) {
            setError("Please choose a stronger password to proceed.");
            return;
        }

        setLoading(true);
        setStatusText("Encrypting payload...");

        abortControllerRef.current = new AbortController();
        const uxTimer = setTimeout(() => {
            if (abortControllerRef.current && !abortControllerRef.current.signal.aborted) {
                setStatusText("Securing network session...");
            }
        }, 600);
        timersRef.current.add(uxTimer);

        try {
            const response = await authenticatedFetch("/api/admin/update_admin_password.php", {
                method: "POST",
                body: JSON.stringify({ currentPassword, newPassword }),
                signal: abortControllerRef.current.signal
            });

            if (abortControllerRef.current.signal.aborted) return;

            if (response.success) {
                // Detach the success state from the red emergency border
                setIsSuccessState(true);
                const successTimer = setTimeout(() => {
                    if (isOpen) onSuccess(response.message || "Password updated securely! Please log in again.");
                }, 2500);
                timersRef.current.add(successTimer);
            } else {
                setError(response.message || "Failed to update password.");
                setLoading(false);
            }
        } catch (err) {
            if (err.name === 'AbortError' || abortControllerRef.current.signal.aborted) return;
            console.error("Password Update Error (Network/Auth):", err);
            
            const isAuthError = err.message?.includes('401') || err.errorCode === 'AUTH_TOKEN_REVOKED' || err.errorCode === 'AUTH_TOKEN_INVALID';
            if (isAuthError) {
                setError("Session expired. Please securely terminate and log in again.");
            } else {
                setError("An unexpected network error occurred. Please try again.");
            }
            setLoading(false);
        }
    };

    return (
        <div 
            ref={modalRef} 
            role="dialog" 
            aria-modal="true" 
            aria-labelledby="modal-title"
            aria-describedby="modal-description"
            className="fixed inset-0 z-[100] flex items-center justify-center p-4 bg-slate-900/60 backdrop-blur-xl"
         >
            <AnimatePresence mode="wait">
                {isVerifying ? (
                    <motion.div
                        key="verifying"
                        initial={{ opacity: 0, scale: 0.95 }}
                        animate={{ opacity: 1, scale: 1 }}
                        exit={{ opacity: 0, scale: 1.05 }}
                        transition={{ duration: 0.4, ease: "easeOut" }}
                        className="flex flex-col items-center justify-center space-y-6 w-full"
                    >
                        <div className="relative flex items-center justify-center w-16 h-16">
                            <div className="absolute inset-0 rounded-full border-2 border-slate-700/50"></div>
                            <div className="absolute inset-0 rounded-full border-2 border-transparent border-t-blue-500 animate-[spin_1.5s_linear_infinite]"></div>
                            <SafeIcon icon={FiIcons.FiShield} className="w-6 h-6 text-blue-500" />
                        </div>
    
                        <div className="flex flex-col items-start space-y-3">
                            {VERIFICATION_MESSAGES.map((msg, idx) => (
                                <div key={`verification-step-${idx}`} className={`flex items-center space-x-3 text-sm font-medium transition-all duration-500 ${idx <= verificationStep ? 'text-white opacity-100 translate-y-0' : 'text-slate-500 opacity-0 translate-y-2'}`}>
                                    {idx < verificationStep ? (
                                        <SafeIcon icon={FiIcons.FiCheckCircle} className="w-4 h-4 text-emerald-400" />
                                    ) : idx === verificationStep ? (
                                        <span className="w-4 h-4 rounded-full border-2 border-blue-500 border-t-transparent animate-spin"></span>
                                    ) : (
                                         <span className="w-4 h-4 rounded-full border-2 border-slate-700"></span>
                                    )}
                                    <span>{msg}</span>
                                </div>
                            ))}
                        </div>
                    </motion.div>
                ) : isSuccessState ? (
                    // [NEW] The Standalone Clean Success Modal (Matches Enforcer)
                    <motion.div
                        key="success-modal"
                        initial={{ opacity: 0, scale: 0.95, y: 20 }}
                        animate={{ opacity: 1, scale: 1, y: 0 }}
                        exit={{ opacity: 0, scale: 0.95, y: 20 }}
                        className="bg-white dark:bg-[#0B1221] rounded-2xl shadow-2xl w-full max-w-md overflow-hidden border border-gray-200/50 dark:border-slate-800/60 relative"
                    >
                        <div className="p-10 flex flex-col items-center justify-center text-center space-y-6">
                            <div className="relative">
                                <div className="w-20 h-20 bg-emerald-100 dark:bg-emerald-900/30 rounded-full flex items-center justify-center text-emerald-600 dark:text-emerald-400 mb-2">
                                    <SafeIcon icon={FiIcons.FiShield} className="w-10 h-10" />
                                </div>
                                <motion.div 
                                    initial={{ scale: 0 }}
                                    animate={{ scale: 1 }}
                                    transition={{ delay: 0.3, type: "spring" }}
                                    className="absolute -bottom-1 -right-1 w-8 h-8 bg-emerald-500 rounded-full border-4 border-white dark:border-[#0B1221] flex items-center justify-center text-white"
                                >
                                    <SafeIcon icon={FiIcons.FiCheck} className="w-4 h-4" />
                                </motion.div>
                            </div>

                            <div>
                                <h3 className="text-xl font-bold text-gray-900 dark:text-white">Credentials Secured</h3>
                                <p className="text-sm text-gray-500 dark:text-gray-400 mt-2">
                                    Validating keys and securing session...
                                </p>
                            </div>

                            <div className="w-full max-w-[200px] h-1.5 bg-gray-100 dark:bg-slate-800 rounded-full overflow-hidden mt-4">
                                <motion.div 
                                    initial={{ width: "0%" }}
                                    animate={{ width: "100%" }}
                                    transition={{ duration: 2.2, ease: "easeInOut" }}
                                    className="h-full bg-emerald-500 rounded-full"
                                />
                            </div>
                        </div>
                    </motion.div>
                ) : (
                    // The Original Red Glowing Warning Modal
                    <motion.div
                        key="main-modal"
                        initial={{ opacity: 0, y: 20 }}
                        animate={{ opacity: 1, y: 0 }}
                        transition={{ duration: 0.5, ease: "easeOut" }}
                        className="p-[1px] rounded-3xl bg-gradient-to-b from-red-500/40 via-slate-800/40 to-slate-800/40 shadow-[0_0_40px_rgba(220,38,38,0.15)] w-full max-w-md"
                     >
                        <div className="bg-white/95 dark:bg-[#0B1221]/95 backdrop-blur-3xl rounded-[23px] overflow-hidden relative">
                            
                            {timeLeft !== undefined && (
                                <div className="absolute top-5 right-5 flex items-center gap-1.5 px-3 py-1.5 bg-amber-100 dark:bg-amber-500/10 border border-amber-200 dark:border-amber-500/20 rounded-full shadow-sm z-10">
                                    <SafeIcon icon={FiIcons.FiClock} className="w-3.5 h-3.5 text-amber-600 dark:text-amber-500 animate-pulse" />
                                    <span className="text-[11px] font-mono font-bold text-amber-700 dark:text-amber-400 tracking-wider">
                                        {Math.floor(timeLeft / 60)}:{(timeLeft % 60).toString().padStart(2, '0')}
                                    </span>
                                </div>
                            )}

                            <div className="p-8">
                                <div className="flex items-center justify-center w-14 h-14 rounded-2xl bg-gradient-to-br from-red-500/20 to-orange-500/10 mx-auto mb-6 border border-red-500/20 shadow-inner mt-2">
                                    <SafeIcon icon={FiIcons.FiLock} className="w-7 h-7 text-red-500" />
                                </div>
                                <h2 id="modal-title" className="text-2xl font-extrabold text-center text-slate-900 dark:text-white mb-2 tracking-tight">Security Update Required</h2>
                                <p id="modal-description" className="text-[13px] leading-relaxed text-center text-slate-500 dark:text-slate-400 mb-6 px-2">
                                    Your cryptographic credentials have reached their maximum lifecycle. Rotate your password to maintain compliance.
                                </p>

                                {error && (
                                    <motion.div 
                                        role="alert" 
                                        aria-live="assertive"
                                        initial={{ opacity: 0, height: 0 }} 
                                        animate={{ opacity: 1, height: 'auto' }} 
                                        className="mb-6 p-3.5 bg-red-50/80 dark:bg-red-500/10 border border-red-200 dark:border-red-500/20 rounded-xl text-sm font-semibold text-red-600 dark:text-red-400 text-center flex items-center justify-center gap-2 shadow-sm"
                                    >
                                        <SafeIcon icon={FiIcons.FiAlertCircle} className="w-4 h-4 shrink-0" />
                                        {error}
                                    </motion.div>
                                )}

                                <form onSubmit={handleSubmit} noValidate className="space-y-5">
                                    <div className="space-y-1.5">
                                        <label htmlFor="currentPwd" className="block text-[13px] font-bold text-slate-700 dark:text-slate-300 ml-1">Current Password</label>
                                        <div className="relative flex items-center">
                                            <SafeIcon icon={FiIcons.FiKey} className="absolute left-4 w-4 h-4 text-slate-400" />
                                            <input 
                                                id="currentPwd" 
                                                type="password" 
                                                required 
                                                disabled={loading} 
                                                ref={firstInputRef} 
                                                autoComplete="current-password" 
                                                className="w-full pl-11 pr-4 py-3 bg-slate-50/50 dark:bg-slate-900/50 border border-slate-200 dark:border-slate-800 rounded-xl focus:border-blue-500 focus:ring-4 focus:ring-blue-500/10 dark:text-white outline-none transition-all disabled:opacity-50 text-sm shadow-sm" 
                                                placeholder="Enter current password" 
                                                value={currentPassword} 
                                                onChange={e => handlePasswordChange(e.target.value, setCurrentPassword)} 
                                            />
                                        </div>
                                    </div>

                                    <div className="space-y-1.5">
                                        <div className="flex items-center justify-between ml-1">
                                            <label htmlFor="newPwd" className="block text-[13px] font-bold text-slate-700 dark:text-slate-300">New Password</label>
                                            <span className="text-[11px] font-semibold text-slate-400 uppercase tracking-wider">16 - 64 Chars</span>
                                        </div>
                                        <div className="relative flex items-center">
                                            <SafeIcon icon={FiIcons.FiShield} className="absolute left-4 w-4 h-4 text-slate-400" />
                                            <input 
                                                id="newPwd" 
                                                type="password" 
                                                required 
                                                disabled={loading} 
                                                autoComplete="new-password" 
                                                minLength={16}
                                                maxLength={64}
                                                className={`w-full pl-11 pr-4 py-3 bg-slate-50/50 dark:bg-slate-900/50 border rounded-xl focus:outline-none focus:ring-4 transition-all disabled:opacity-50 text-sm shadow-sm ${
                                                    !hasStartedTyping
                                                        ? "border-slate-200 dark:border-slate-800 focus:border-blue-500 focus:ring-blue-500/10 dark:text-white"
                                                        : isPasswordValid
                                                            ? "border-green-500/50 focus:border-green-500 focus:ring-green-500/20 text-green-900 dark:text-green-100"
                                                            : "border-red-500/50 focus:border-red-500 focus:ring-red-500/20 text-red-900 dark:text-red-100"
                                                }`}
                                                placeholder="Create new password" 
                                                value={newPassword} 
                                                onChange={e => handlePasswordChange(e.target.value, setNewPassword, true)} 
                                            />
                                        </div>

                                        {newPassword && (
                                            <div className="mt-2 ml-1 space-y-1.5">
                                                <div className="flex gap-1 h-1.5 mr-1">
                                                    {[1, 2, 3, 4, 5].map((level) => (
                                                        <div 
                                                            key={level} 
                                                            className={`flex-1 rounded-full transition-colors duration-300 ${
                                                                passwordStrength >= level 
                                                                    ? getStrengthColor(passwordStrength) 
                                                                    : 'bg-slate-200 dark:bg-slate-700'
                                                            }`} 
                                                        />
                                                    ))}
                                                </div>
                                                <p className={`text-[11px] font-medium transition-colors ${getStrengthTextColor(passwordStrength)}`}>
                                                    {getStrengthLabel(passwordStrength)}
                                                </p>
                                            </div>
                                        )}

                                        <AnimatePresence>
                                            {newPassword.length === 64 && (
                                                <motion.div
                                                    initial={{ opacity: 0, height: 0 }}
                                                    animate={{ opacity: 1, height: "auto" }}
                                                    exit={{ opacity: 0, height: 0 }}
                                                 >
                                                    <p className="text-amber-500 dark:text-amber-400 text-[11px] font-medium mt-1.5 ml-1 flex items-center gap-1">
                                                        <SafeIcon icon={FiIcons.FiAlertCircle} className="w-3.5 h-3.5" />
                                                        Maximum password length reached (64/64).
                                                    </p>
                                                </motion.div>
                                            )}
                                        </AnimatePresence>
                                    </div>

                                    <div className="space-y-1.5 mt-2">
                                        <label htmlFor="confirmPwd" className="block text-[13px] font-bold text-slate-700 dark:text-slate-300 ml-1">Confirm Password</label>
                                        <div className="relative flex items-center">
                                            <SafeIcon icon={FiIcons.FiCheck} className="absolute left-4 w-4 h-4 text-slate-400" />
                                            <input 
                                                id="confirmPwd" 
                                                type="password" 
                                                required 
                                                disabled={loading} 
                                                autoComplete="new-password" 
                                                minLength={16}
                                                maxLength={64}
                                                className={`w-full pl-11 pr-4 py-3 bg-slate-50/50 dark:bg-slate-900/50 border rounded-xl focus:outline-none focus:ring-4 transition-all disabled:opacity-50 text-sm shadow-sm ${
                                                    !confirmHasStarted
                                                        ? "border-slate-200 dark:border-slate-800 focus:border-blue-500 focus:ring-blue-500/10 dark:text-white"
                                                        : isMatch
                                                            ? "border-emerald-500 border-2 focus:ring-emerald-500/20 text-emerald-900 dark:text-emerald-100"
                                                            : "border-red-500 border-2 focus:ring-red-500/20 text-red-900 dark:text-red-100"
                                                }`}
                                                placeholder="Repeat new password" 
                                                value={confirmPassword} 
                                                onChange={e => handlePasswordChange(e.target.value, setConfirmPassword)} 
                                            />
                                        </div>

                                        <AnimatePresence>
                                            {confirmHasStarted && !isMatch && (
                                                <motion.div
                                                    initial={{ opacity: 0, height: 0 }}
                                                    animate={{ opacity: 1, height: "auto" }}
                                                    exit={{ opacity: 0, height: 0 }}
                                                >
                                                     <p className="text-red-500 dark:text-red-400 text-[11px] font-medium mt-1.5 ml-1">
                                                        Passwords do not match.
                                                    </p>
                                                </motion.div>
                                            )}
                                        </AnimatePresence>
                                    </div>

                                    <button 
                                        type="submit" 
                                        disabled={loading || !isPasswordValid || !isMatch || passwordStrength < 3} 
                                        className="w-full mt-4 flex justify-center items-center bg-gradient-to-r from-blue-600 to-indigo-600 hover:from-blue-500 hover:to-indigo-500 text-white font-bold py-3.5 px-4 rounded-xl transition-all shadow-[0_8px_20px_rgba(79,70,229,0.25)] hover:shadow-[0_10px_25px_rgba(79,70,229,0.35)] disabled:opacity-70 disabled:hover:shadow-none text-[15px] disabled:cursor-not-allowed"
                                    >
                                        {loading ? (
                                            <>
                                                <SafeIcon icon={FiIcons.FiLoader} className="w-4 h-4 mr-2 animate-spin" />
                                                <span aria-live="polite">{statusText}</span>
                                            </>
                                        ) : "Secure Credentials & Continue"}
                                    </button>
                                </form>

                                {!loading && (
                                    <div className="mt-6 text-center">
                                        <button 
                                            onClick={onLogout}
                                            type="button" 
                                            className="group flex items-center justify-center w-full gap-2 text-[13px] font-bold text-slate-500 hover:text-slate-800 dark:text-slate-400 dark:hover:text-white transition-colors outline-none focus-visible:ring-2 focus-visible:ring-slate-400 rounded-md p-1"
                                         >
                                            <SafeIcon icon={FiIcons.FiLogOut} className="w-3.5 h-3.5 group-hover:-translate-x-1 transition-transform" />
                                            Securely terminate session
                                        </button>
                                    </div>
                                 )}
                            </div>
                        </div>
                    </motion.div>
                )}
             </AnimatePresence>
        </div>
    );
};

export default ForcePasswordUpdateModal;
