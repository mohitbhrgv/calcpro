import React, { useState, useEffect, useRef } from "react";
import Modal from "../common/Modal";
import SafeIcon from "@/common/SafeIcon";
import * as FiIcons from "react-icons/fi";
import { authenticatedFetch } from "@/admin/utils/apiService";
import toast from "react-hot-toast";
import { motion } from "framer-motion";

const {
  FiSave, FiFileText, FiType, FiMaximize, FiCalendar, 
  FiClipboard, FiUpload, FiRefreshCw, FiAlertTriangle, FiCheck, FiTrash2, FiX
} = FiIcons;

const InfoBlock = ({ label, value, icon }) => (
  <div className="flex flex-col p-2.5 rounded-lg bg-gray-50 dark:bg-gray-800/50 border border-gray-100 dark:border-gray-700/50">
    <span className="text-[10px] uppercase tracking-wider font-bold text-gray-400 dark:text-gray-500 mb-0.5 flex items-center gap-1.5">
      <SafeIcon icon={icon} className="w-3 h-3" /> {label}
    </span>
    <span className="text-xs font-medium text-gray-900 dark:text-gray-200 truncate" title={value}>
      {value}
    </span>
  </div>
);

const RadioCard = ({ title, desc, value, selected, onChange }) => (
  <div 
    onClick={() => onChange(value)}
    className={`relative p-3 rounded-lg border-2 cursor-pointer transition-all duration-200 group ${
      selected === value
        ? 'border-indigo-500 bg-indigo-50/50 dark:bg-indigo-900/10' 
        : 'border-gray-200 dark:border-gray-700 hover:border-gray-300 dark:hover:border-gray-600 hover:bg-gray-50 dark:hover:bg-gray-800'
    }`}
  >
    <div className="flex items-start gap-3">
      <div className={`mt-0.5 w-4 h-4 rounded-full border flex items-center justify-center flex-shrink-0 transition-colors ${
        selected === value 
          ? 'border-indigo-500 bg-indigo-500 text-white' 
          : 'border-gray-400 dark:border-gray-500'
      }`}>
        {selected === value && <SafeIcon icon={FiCheck} className="w-2.5 h-2.5" />}
      </div>
      <div>
        <h4 className={`text-sm font-bold ${selected === value ? 'text-indigo-700 dark:text-indigo-300' : 'text-gray-900 dark:text-gray-200'}`}>
            {title}
        </h4>
        <p className="text-xs text-gray-500 dark:text-gray-400 mt-0.5 leading-tight">
            {desc}
        </p>
      </div>
    </div>
  </div>
);

const MediaDetailsModal = ({ 
  mediaItem, 
  isOpen, 
  onClose, 
  onUpdate, 
  onDelete,
  disableAnimation = false,
  zIndexClass = "z-[60]" 
}) => {
  const [activeTab, setActiveTab] = useState("details");
  const [altText, setAltText] = useState("");
  const [isSubmitting, setIsSubmitting] = useState(false);
  const [newFile, setNewFile] = useState(null);
  const [previewUrl, setPreviewUrl] = useState(null);
  const [replacementType, setReplacementType] = useState("replace_only");
  const [dateOption, setDateOption] = useState("keep");
  const fileInputRef = useRef(null);

  useEffect(() => {
    if (newFile) {
      const objectUrl = URL.createObjectURL(newFile);
      setPreviewUrl(objectUrl);
      return () => {
        URL.revokeObjectURL(objectUrl);
      };
    } else {
      setPreviewUrl(null);
    }
  }, [newFile]);

  useEffect(() => {
    if (mediaItem) {
      setAltText(mediaItem.alt_text || "");
    }
    setActiveTab("details");
    setNewFile(null);
    setReplacementType("replace_only");
    setDateOption("keep");
  }, [mediaItem, isOpen]);

  if (!mediaItem) return null;

  const handleAltTextSave = async () => {
    if (isSubmitting) return; 
    
    setIsSubmitting(true);
    const toastId = "media-alt-save"; 
    toast.loading("Saving and syncing metadata...", { id: toastId });

    try {
      const apiCall = authenticatedFetch("/api/admin/update_media_details.php", {
          method: "POST",
          body: JSON.stringify({
            id: mediaItem.id,
            alt_text: altText
          }),
      });

      const minimumDelay = new Promise(resolve => setTimeout(resolve, 800));
      const [result] = await Promise.all([apiCall, minimumDelay]);

      if (result.success) {
        toast.success(result.message, { id: toastId });
        onUpdate({ ...mediaItem, alt_text: altText });
        onClose(); 
      } else {
        toast.error(result.message, { id: toastId });
      }
    } catch (error) {
       toast.error("Connection failed.", { id: toastId });
    } finally {
      setIsSubmitting(false);
    }
  };

  const handleFileChange = (event) => {
    const file = event.target.files[0];
    if (file) setNewFile(file);
    event.target.value = "";
  };

  const handleClearFile = (e) => {
    e.stopPropagation();
    setNewFile(null);
    if (fileInputRef.current) {
      fileInputRef.current.value = "";
    }
  };

  const handleReplace = async () => {
    if (isSubmitting) return; 
    
    const errorToastId = "media-replace-error";

    if (!newFile) return toast.error("Please select a file.", { id: errorToastId });

    // Double-check validations before sending to server
    if (replacementType === "replace_only" && newFile.name !== mediaItem.original_name) {
        return toast.error(`Filename mismatch! Must be '${mediaItem.original_name}'`, { id: errorToastId });
    }
    if (replacementType === "replace_and_update_links" && newFile.name === mediaItem.original_name) {
        return toast.error("New file must have a different name for Update Everything mode.", { id: errorToastId });
    }

    if (!window.confirm("This is a permanent action. Overwrite file?")) return;

    setIsSubmitting(true);
    const toastId = "media-replace-save";
    toast.loading("Processing replacement...", { id: toastId });
    
    const formData = new FormData();
    formData.append("media_id", mediaItem.id);
    formData.append("replacement_type", replacementType);
    formData.append("alt_text", altText);
    formData.append("date_option", dateOption);
    formData.append("newMediaFile", newFile);

    try {
      const envUrl = import.meta.env.VITE_API_BASE_URL || '';
      const cleanBase = envUrl.replace(/\/+$/, '');
      const token = localStorage.getItem("calcpro-admin-token");
      const apiCall = fetch(
        `${cleanBase}/api/admin/replace_media.php`,
        {
          method: "POST",
          headers: { Authorization: `Bearer ${token}` },
          body: formData,
        }
      ).then(res => res.json());

      const minimumDelay = new Promise(resolve => setTimeout(resolve, 800));
      const [result] = await Promise.all([apiCall, minimumDelay]);

      if (result.success) {
        toast.success(result.message, { id: toastId });
        onUpdate({ ...mediaItem, needs_refresh: true });
        onClose();
      } else {
        toast.error(result.message || "Upload failed.", { id: toastId });
      }
    } catch (error) {
      toast.error("Network error.", { id: toastId });
    } finally {
      setIsSubmitting(false);
    }
  };

  const formatSize = (bytes) => {
    if (!bytes) return "0 B";
    const k = 1024;
    const sizes = ["B", "KB", "MB"];
    const i = Math.floor(Math.log(bytes) / Math.log(k));
    return parseFloat((bytes / Math.pow(k, i)).toFixed(1)) + " " + sizes[i];
  };

  const formatDateTime = (dateString) => {
    if (!dateString) return "N/A";
    const isIsoFormat = dateString.includes('T') && dateString.endsWith('Z');
    const safeUtcString = isIsoFormat ? dateString : dateString.replace(' ', 'T') + 'Z';
    const dateObj = new Date(safeUtcString);
    if (isNaN(dateObj.getTime())) return "Invalid Date";
    
    const formattedDate = dateObj.toLocaleDateString('en-US', { month: 'long', day: 'numeric', year: 'numeric' });
    const formattedTime = dateObj.toLocaleTimeString('en-US', { hour: 'numeric', minute: '2-digit', hour12: true });
    
    return `${formattedDate} at ${formattedTime}`;
  };

  // Helper values for dynamic validation rendering
  const isStrictMismatch = newFile && replacementType === 'replace_only' && newFile.name !== mediaItem.original_name;
  const isGlobalDuplicate = newFile && replacementType === 'replace_and_update_links' && newFile.name === mediaItem.original_name;
  const isReplaceDisabled = !newFile || isSubmitting || isStrictMismatch || isGlobalDuplicate;

  return (
    <Modal 
      isOpen={isOpen} 
      onClose={onClose} 
      title="Media Details" 
      size="xl" 
      zIndexClass={zIndexClass} 
      disableAnimation={disableAnimation}
    >
      <div className="flex flex-col md:flex-row h-[70vh] md:h-[500px] overflow-hidden -mx-6 -mb-4 bg-white dark:bg-gray-900">
        
        {/* --- LEFT: PREVIEW AREA --- */}
        <div className="w-full md:w-5/12 bg-gray-100 dark:bg-black/40 border-r border-gray-200 dark:border-gray-700/50 flex flex-col relative group">
            <div className="absolute inset-0 opacity-[0.03] dark:opacity-[0.08]" 
                 style={{ 
                     backgroundImage: `linear-gradient(#000 1px, transparent 1px), linear-gradient(90deg, #000 1px, transparent 1px)`, 
                     backgroundSize: '20px 20px' 
                 }} 
            />

            <div className="flex-1 flex items-center justify-center p-4 relative z-10 overflow-hidden">
                {activeTab === 'replace' && previewUrl ? (
                    <motion.div
                        key="comparison-view"
                        initial={{ opacity: 0 }}
                        animate={{ opacity: 1 }}
                        transition={{ duration: 0.3 }}
                        className="flex flex-col items-stretch gap-0 w-full h-full"
                    >
                        {/* Current Image */}
                        <div className="flex-1 flex flex-col items-center gap-1.5 min-h-0">
                            <span className="inline-flex items-center gap-1 px-2.5 py-0.5 rounded-full bg-gray-700/80 text-white text-[9px] font-bold uppercase tracking-widest shrink-0 shadow">
                                Current
                            </span>
                            <div className="flex-1 flex items-center justify-center w-full overflow-hidden rounded-lg border-2 border-gray-300 dark:border-gray-600 bg-gray-100 dark:bg-gray-800/60 shadow-inner p-1.5 min-h-0">
                                <img src={mediaItem.full_url} alt={altText} className="max-w-full max-h-full object-contain rounded opacity-80" />
                            </div>
                        </div>

                        {/* Divider */}
                        <div className="flex flex-row items-center justify-center py-1.5 shrink-0 gap-1.5">
                            <div className="h-px flex-1 bg-gradient-to-r from-transparent via-gray-300 dark:via-gray-600 to-transparent" />
                            <div className={`flex items-center justify-center rounded-full p-2 shadow-lg ${isReplaceDisabled ? 'bg-red-500 shadow-red-500/30' : 'bg-indigo-600 shadow-indigo-500/30'} transition-colors`}>
                                <SafeIcon icon={isReplaceDisabled ? FiAlertTriangle : FiUpload} className="w-3.5 h-3.5 text-white" />
                            </div>
                            <div className="h-px flex-1 bg-gradient-to-r from-transparent via-gray-300 dark:via-gray-600 to-transparent" />
                        </div>

                        {/* New Image */}
                        <motion.div
                            key={previewUrl}
                            initial={{ opacity: 0, y: 12 }}
                            animate={{ opacity: 1, y: 0 }}
                            transition={{ duration: 0.3 }}
                            className="flex-1 flex flex-col items-center gap-1.5 min-h-0"
                        >
                            <span className={`inline-flex items-center gap-1 px-2.5 py-0.5 rounded-full text-white text-[9px] font-bold uppercase tracking-widest shrink-0 shadow ${isReplaceDisabled ? 'bg-red-500' : 'bg-indigo-600'} transition-colors`}>
                                Replacing With
                            </span>
                            <div className={`flex-1 flex items-center justify-center w-full overflow-hidden rounded-lg border-2 bg-indigo-50/40 dark:bg-indigo-900/20 shadow-md p-1.5 min-h-0 transition-colors ${isReplaceDisabled ? 'border-red-500 dark:border-red-400 shadow-red-200 dark:shadow-red-900/30' : 'border-indigo-500 dark:border-indigo-400 shadow-indigo-200 dark:shadow-indigo-900/30'}`}>
                                <img src={previewUrl} alt="New file preview" className="max-w-full max-h-full object-contain rounded" />
                            </div>
                        </motion.div>
                    </motion.div>
                ) : (
                    <motion.img
                        key={mediaItem.full_url}
                        initial={{ opacity: 0, scale: 0.95 }}
                        animate={{ opacity: 1, scale: 1 }}
                        src={mediaItem.full_url}
                        alt={altText}
                        className="max-w-full max-h-full object-contain shadow-lg rounded-lg"
                    />
                )}
            </div>
            
            <div className="p-4 relative z-10 border-t border-gray-200/50 dark:border-white/5" />
        </div>

        {/* --- RIGHT: CONTROL PANEL --- */}
        <div className="w-full md:w-7/12 flex flex-col bg-white dark:bg-gray-900">
            <div className="px-5 pt-5 pb-3 border-b border-gray-100 dark:border-gray-800">
                <div className="bg-gray-100 dark:bg-gray-800 p-1 rounded-lg flex">
                    <button 
                        onClick={() => setActiveTab('details')}
                        className={`flex-1 py-1.5 text-xs font-bold uppercase tracking-wide rounded-md transition-all ${
                            activeTab === 'details' ? 'bg-white dark:bg-gray-700 text-gray-900 dark:text-white shadow-sm' : 'text-gray-500 dark:text-gray-400 hover:text-gray-700 dark:hover:text-gray-200'
                        }`}
                    >
                        Metadata
                    </button>
                    <button 
                        onClick={() => setActiveTab('replace')}
                        className={`flex-1 py-1.5 text-xs font-bold uppercase tracking-wide rounded-md transition-all ${
                            activeTab === 'replace' ? 'bg-white dark:bg-gray-700 text-indigo-600 dark:text-indigo-400 shadow-sm' : 'text-gray-500 dark:text-gray-400 hover:text-gray-700 dark:hover:text-gray-200'
                        }`}
                    >
                        Replace File
                    </button>
                </div>
            </div>

            <div className="flex-1 overflow-y-auto px-5 py-4 space-y-5 custom-scrollbar">
                
                {/* === METADATA TAB === */}
                {activeTab === 'details' && (
                    <div className="space-y-5 animate-fade-in">
                         <div>
                            <div className="flex justify-between items-center mb-1.5">
                                <label className="text-xs font-bold text-gray-500 dark:text-gray-400 uppercase tracking-wider">Alt Text</label>
                            </div>
                            <textarea 
                                value={altText}
                                onChange={(e) => setAltText(e.target.value)}
                                rows={3}
                                className="w-full px-3 py-2 text-sm border border-gray-200 dark:border-gray-700 rounded-lg focus:ring-2 focus:ring-indigo-500 outline-none bg-gray-50 dark:bg-gray-800 text-gray-900 dark:text-white resize-none transition-shadow focus:bg-white dark:focus:bg-gray-900"
                                placeholder="Describe this image for SEO and accessibility..."
                            />
                            <p className="text-[11px] text-gray-500 dark:text-gray-400 mt-2">
                                Changes are automatically synced to all published posts and calculators.
                            </p>
                         </div>

                         <div>
                            <h4 className="text-xs font-bold text-gray-500 dark:text-gray-400 uppercase tracking-wider mb-2.5">Properties</h4>
                            <div className="grid grid-cols-2 gap-2">
                                <InfoBlock icon={FiFileText} label="Name" value={mediaItem.file_name} />
                                <InfoBlock icon={FiType} label="Type" value={mediaItem.file_type} />
                                <InfoBlock icon={FiMaximize} label="Size" value={formatSize(mediaItem.file_size)} />
                                <InfoBlock icon={FiCalendar} label="Updated At" value={formatDateTime(mediaItem.updated_at || mediaItem.created_at)} />
                            </div>
                         </div>

                         <div>
                            <label className="block text-xs font-bold text-gray-500 dark:text-gray-400 uppercase tracking-wider mb-1.5">Public URL</label>
                            <div className="flex items-center gap-2 p-1.5 pl-3 border border-gray-200 dark:border-gray-700 rounded-lg bg-gray-50 dark:bg-gray-800/50">
                                <span className="flex-1 text-[11px] font-mono text-gray-600 dark:text-gray-300 truncate">{mediaItem.full_url}</span>
                                <button 
                                    onClick={() => { navigator.clipboard.writeText(mediaItem.full_url); toast.success("Link copied!", { id: "copy-url" }); }}
                                    className="p-1.5 hover:bg-white dark:hover:bg-gray-700 rounded-md text-gray-500 hover:text-indigo-600 dark:text-gray-400 dark:hover:text-indigo-400 transition-colors shadow-sm"
                                    title="Copy Link"
                                >
                                    <SafeIcon icon={FiClipboard} className="w-3.5 h-3.5" />
                                </button>
                            </div>
                         </div>
                    </div>
                )}

                {/* === REPLACE TAB === */}
                {activeTab === 'replace' && (
                    <div className="space-y-4 animate-fade-in">
                         <div className="p-3 rounded-lg bg-amber-50 dark:bg-amber-900/20 border border-amber-200 dark:border-amber-800/50 flex gap-3 items-start">
                             <SafeIcon icon={FiAlertTriangle} className="w-4 h-4 text-amber-600 dark:text-amber-500 shrink-0 mt-0.5" />
                             <div>
                                 <h5 className="text-xs font-bold text-amber-900 dark:text-amber-100">Permanent Action</h5>
                                 <p className="text-[11px] text-amber-800 dark:text-amber-200/80 leading-snug">
                                     Replacing an image cannot be undone. Ensure dimensions match if layout critical.
                                 </p>
                             </div>
                         </div>

                         <div className="rounded-xl bg-amber-50 dark:bg-amber-900/10 border border-amber-200 dark:border-amber-700/40 p-3">
                             <label className="block text-xs font-bold text-amber-700 dark:text-amber-300 uppercase tracking-wider mb-2">
                                 New File
                             </label>

                             <input type="file" ref={fileInputRef} onChange={handleFileChange} className="hidden" accept="image/png, image/jpeg, image/gif, image/webp" />

                             {newFile ? (
                                 <div className="flex items-center gap-3 px-3 py-2.5 rounded-lg border-2 border-amber-400 dark:border-amber-500 bg-white dark:bg-amber-900/20 shadow-sm">
                                     <div className="shrink-0 p-1.5 bg-amber-100 dark:bg-amber-800/50 rounded-md">
                                         <SafeIcon icon={FiUpload} className="w-3.5 h-3.5 text-amber-600 dark:text-amber-400" />
                                     </div>
                                     <span className="flex-1 text-xs font-semibold text-amber-700 dark:text-amber-300 truncate min-w-0" title={newFile.name}>
                                         {newFile.name}
                                     </span>
                                     <button type="button" onClick={() => fileInputRef.current.click()} className="shrink-0 text-[10px] font-bold uppercase tracking-wide text-amber-600 dark:text-amber-400 hover:text-amber-800 dark:hover:text-amber-200 transition-colors px-1" title="Select a different file">
                                         Change
                                     </button>
                                     <button type="button" onClick={handleClearFile} className="shrink-0 p-1 rounded-md text-gray-400 hover:text-red-500 dark:hover:text-red-400 hover:bg-red-50 dark:hover:bg-red-900/20 transition-colors" title="Remove selected file">
                                         <SafeIcon icon={FiX} className="w-3.5 h-3.5" />
                                     </button>
                                 </div>
                             ) : (
                                 <div onClick={() => fileInputRef.current.click()} className="border-2 border-dashed border-amber-300 dark:border-amber-600/60 rounded-lg p-4 text-center hover:border-amber-500 dark:hover:border-amber-400 hover:bg-amber-100/60 dark:hover:bg-amber-800/20 cursor-pointer transition-all group">
                                     <div className="flex items-center justify-center gap-2 text-amber-600 dark:text-amber-400 group-hover:text-amber-800 dark:group-hover:text-amber-200">
                                         <SafeIcon icon={FiUpload} className="w-4 h-4" />
                                         <span className="text-xs font-bold">Select File</span>
                                     </div>
                                 </div>
                             )}
                         </div>

                         <div>
                            <label className="block text-xs font-bold text-gray-500 dark:text-gray-400 uppercase mb-2">Mode</label>
                            <div className="space-y-2">
                                <RadioCard title="Strict Replace" desc="Exact filename match only. Safest." value="replace_only" selected={replacementType} onChange={setReplacementType} />
                                <RadioCard title="Update Everything" desc="New filename allowed. Updates all posts." value="replace_and_update_links" selected={replacementType} onChange={setReplacementType} />
                            </div>
                         </div>

                         {/* === UNIVERSAL VERIFICATION BLOCK & WARNINGS === */}
                         {newFile && (
                            <motion.div
                                initial={{ opacity: 0, y: -10 }}
                                animate={{ opacity: 1, y: 0 }}
                                className={`p-3.5 rounded-lg border shadow-sm transition-colors ${isReplaceDisabled ? 'bg-red-50 dark:bg-red-900/20 border-red-200 dark:border-red-800/50' : 'bg-gray-50 dark:bg-gray-800/50 border-gray-200 dark:border-gray-700'}`}
                            >
                                <h5 className={`text-[10px] font-bold uppercase tracking-wider mb-3 ${isReplaceDisabled ? 'text-red-600 dark:text-red-400' : 'text-gray-500 dark:text-gray-400'}`}>
                                    {replacementType === 'replace_only' ? 'Strict Verification Check' : 'Global Update Verification'}
                                </h5>
                                
                                <div className="space-y-3">
                                    {/* 1. Filename Check */}
                                    <div className="space-y-1.5">
                                        <div className="flex items-center justify-between text-xs">
                                            <span className={`${isReplaceDisabled ? 'text-red-500 dark:text-red-400' : 'text-gray-500 dark:text-gray-400'} font-medium text-[11px] uppercase tracking-wide`}>Original Name:</span>
                                            <span className="font-mono text-[11px] text-gray-700 dark:text-gray-300 truncate max-w-[200px]" title={mediaItem.original_name}>
                                                {mediaItem.original_name}
                                            </span>
                                        </div>
                                        <div className={`flex items-center justify-between text-xs pb-3 border-b ${isReplaceDisabled ? 'border-red-200 dark:border-red-800/50' : 'border-gray-200 dark:border-gray-700/50'}`}>
                                            <span className={`${isReplaceDisabled ? 'text-red-500 dark:text-red-400' : 'text-gray-500 dark:text-gray-400'} font-medium text-[11px] uppercase tracking-wide`}>Replacement:</span>
                                            <div className="flex items-center gap-1.5 min-w-0">
                                                {/* Icon Logic: Green Check if rules are followed, Red Triangle if violated */}
                                                {!isReplaceDisabled ? (
                                                    <SafeIcon icon={FiCheck} className="w-3.5 h-3.5 text-green-500 shrink-0" />
                                                ) : (
                                                    <SafeIcon icon={FiAlertTriangle} className="w-3.5 h-3.5 text-red-500 shrink-0" />
                                                )}
                                                <span className={`font-mono text-[11px] truncate max-w-[180px] font-bold ${!isReplaceDisabled ? 'text-green-600 dark:text-green-400' : 'text-red-600 dark:text-red-400'}`} title={newFile.name}>
                                                    {newFile.name}
                                                </span>
                                            </div>
                                        </div>
                                    </div>

                                    {/* Warnings Messages mapped to strict rules */}
                                    {isStrictMismatch && (
                                        <div className="flex items-start gap-2 p-2 bg-white/50 dark:bg-black/20 rounded border border-red-100 dark:border-red-800/50">
                                            <SafeIcon icon={FiAlertTriangle} className="w-3.5 h-3.5 text-red-600 dark:text-red-400 shrink-0 mt-0.5" />
                                            <p className="text-[10px] text-red-800 dark:text-red-300 leading-tight">
                                                <strong>Filename Mismatch:</strong> Strict Replace requires the exact same name. Rename your file to <strong>{mediaItem.original_name}</strong> or use 'Update Everything' mode.
                                            </p>
                                        </div>
                                    )}
                                    {isGlobalDuplicate && (
                                        <div className="flex items-start gap-2 p-2 bg-white/50 dark:bg-black/20 rounded border border-red-100 dark:border-red-800/50">
                                            <SafeIcon icon={FiAlertTriangle} className="w-3.5 h-3.5 text-red-600 dark:text-red-400 shrink-0 mt-0.5" />
                                            <p className="text-[10px] text-red-800 dark:text-red-300 leading-tight">
                                                <strong>Duplicate Name:</strong> 'Update Everything' mode safely rewrites URLs. It requires the new file to have a <em>different</em> name. If you only want to replace content, switch to 'Strict Replace'.
                                            </p>
                                        </div>
                                    )}

                                    {/* 2. Properties Check - Size and Type Comparison */}
                                    <div className="grid grid-cols-2 gap-4 pt-1">
                                        <div>
                                            <span className={`block text-[9px] uppercase tracking-wider font-bold mb-1 ${isReplaceDisabled ? 'text-red-400 dark:text-red-500' : 'text-gray-400 dark:text-gray-500'}`}>Current</span>
                                            <div className={`text-[11px] font-medium border rounded py-1 px-2 inline-flex items-center ${isReplaceDisabled ? 'bg-white/50 dark:bg-black/20 border-red-200 dark:border-red-800 text-red-700 dark:text-red-300' : 'bg-white dark:bg-gray-800 border-gray-200 dark:border-gray-700 text-gray-700 dark:text-gray-300'}`}>
                                                {formatSize(mediaItem.file_size)} <span className="opacity-40 mx-1.5">|</span> {mediaItem.file_type.split('/')[1]?.toUpperCase() || 'IMG'}
                                            </div>
                                        </div>
                                        <div className="text-right flex flex-col items-end">
                                            <span className={`block text-[9px] uppercase tracking-wider font-bold mb-1 ${isReplaceDisabled ? 'text-red-500 dark:text-red-400' : 'text-indigo-500 dark:text-indigo-400'}`}>New</span>
                                            <div className={`text-[11px] font-medium border rounded py-1 px-2 inline-flex items-center ${isReplaceDisabled ? 'bg-red-100/50 dark:bg-red-900/30 border-red-300 dark:border-red-700/50 text-red-700 dark:text-red-300' : 'bg-indigo-50 dark:bg-indigo-900/30 border-indigo-200 dark:border-indigo-800/50 text-indigo-700 dark:text-indigo-300'}`}>
                                                {formatSize(newFile.size)} <span className="opacity-40 mx-1.5">|</span> {newFile.type.split('/')[1]?.toUpperCase() || 'IMG'}
                                            </div>
                                        </div>
                                    </div>
                                </div>
                            </motion.div>
                         )}
                    </div>
                )}
            </div>

            <div className="px-5 py-4 border-t border-gray-100 dark:border-gray-800 bg-gray-50 dark:bg-gray-900/50 flex justify-between items-center gap-3">
                {onDelete && (
                    <button onClick={() => onDelete(mediaItem.id)} disabled={isSubmitting} className="flex items-center gap-2 px-4 py-2 text-red-600 hover:bg-red-100 dark:text-red-400 dark:hover:bg-red-900/20 text-xs font-bold rounded-lg transition-colors disabled:opacity-50" title="Move to Trash">
                        <SafeIcon icon={FiTrash2} className="w-3.5 h-3.5" /> Trash
                    </button>
                )}

                <div className="flex gap-3 ml-auto">
                    <button onClick={onClose} className="px-4 py-2 text-gray-600 dark:text-gray-400 hover:bg-white dark:hover:bg-gray-700 text-xs font-bold rounded-lg transition-colors border border-transparent hover:border-gray-200 dark:hover:border-gray-600">
                        Cancel
                    </button>

                    {activeTab === 'details' ? (
                        <button onClick={handleAltTextSave} disabled={isSubmitting} className="flex items-center gap-2 px-5 py-2 bg-indigo-600 hover:bg-indigo-700 text-white text-xs font-bold rounded-lg shadow-sm hover:shadow-md transition-all transform active:scale-95 disabled:opacity-50 min-w-[90px] justify-center">
                            {isSubmitting ? <SafeIcon icon={FiRefreshCw} className="w-3.5 h-3.5 animate-spin" /> : <> <SafeIcon icon={FiSave} className="w-3.5 h-3.5" /> Update </>}
                        </button>
                    ) : (
                        <button onClick={handleReplace} disabled={isReplaceDisabled} className={`flex items-center gap-2 px-5 py-2 text-white text-xs font-bold rounded-lg shadow-sm transition-all transform active:scale-95 disabled:opacity-50 min-w-[100px] justify-center ${isReplaceDisabled ? 'bg-red-400 cursor-not-allowed' : 'bg-red-600 hover:bg-red-700 hover:shadow-md'}`}>
                            <SafeIcon icon={FiRefreshCw} className={`w-3.5 h-3.5 ${isSubmitting ? 'animate-spin' : ''}`} />
                            {isSubmitting ? "Processing..." : "Confirm"}
                        </button>
                    )}
                </div>
            </div>
        </div>
      </div>
    </Modal>
  );
};

export default MediaDetailsModal;
