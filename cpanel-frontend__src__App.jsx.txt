function App() {
  const [isOffline, setIsOffline] = React.useState(() => !!localStorage.getItem('calcpro_sys_offline_start'));
  const [incidentCode, setIncidentCode] = React.useState(() => localStorage.getItem('calcpro_sys_incident_code') || null);

  React.useEffect(() => {
    const handleOffline = (e) => {
      if (e.detail) {
          localStorage.setItem('calcpro_sys_incident_code', e.detail);
          setIncidentCode(e.detail);
      }
      setIsOffline(true);
    };
    const handleOnline = () => {
      // [ENTERPRISE UX FIX] Fresh Session Start to prevent immediate inactivity logout
      localStorage.setItem('calcpro_last_active', Date.now().toString());
      localStorage.removeItem('calcpro_sys_incident_code');
      setIncidentCode(null);
      setIsOffline(false);
    };

    window.addEventListener("SYSTEM_OFFLINE", handleOffline);
    window.addEventListener("SYSTEM_ONLINE", handleOnline);

    return () => {
      window.removeEventListener("SYSTEM_OFFLINE", handleOffline);
      window.removeEventListener("SYSTEM_ONLINE", handleOnline);
    };
  }, []);

  // [ENTERPRISE FIX] Absolute Shield
  // By returning here, we strictly forbid AuthProvider and ThemeProvider from mounting.
  // This guarantees zero background API calls (like get_me.php) are made while offline.
  if (isOffline) {
      return <SystemOffline incidentCode={incidentCode} />;
  }

  return (
    <ThemeProvider>
      <AuthProvider>
        <LayoutProvider>
          <AdminCacheProvider>
             <ErrorBoundary>
               <AppWrapper />
             </ErrorBoundary>
          </AdminCacheProvider>
        </LayoutProvider>
      </AuthProvider>
    </ThemeProvider>
  );
}

export default App;
