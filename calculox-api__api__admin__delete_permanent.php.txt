<?php
// Core utilities
require_once '../../core/config.php';
require_once '../../core/response.php';
require_once '../../core/auth.php';

// --- SECURITY CHECK ---
$admin_data = verifyAdminToken();

try {
    // 1. Get Incoming Data
    $json = file_get_contents('php://input');
    $data = json_decode($json);

    // 2. Validation
    if (!isset($data->id) || !is_numeric($data->id)) {
        sendResponse(false, null, 'A valid ID is required.', 400, 'INVALID_INPUT');
    }
    
    $id = (int)$data->id;

    // Optional: If your file handles multiple types (e.g., posts, pages) passed from the frontend, 
    // we capture it. Otherwise, we default to 'posts' since the error originated from the blog page.
    $type = isset($data->type) ? $data->type : 'posts';
    
    // Map the allowed tables to prevent SQL injection on table names
    $allowed_tables = [
        'posts' => 'posts',
        'pages' => 'pages',
        'calculators' => 'calculators'
    ];

    if (!array_key_exists($type, $allowed_tables)) {
        sendResponse(false, null, 'Invalid content type specified.', 400, 'INVALID_TYPE');
        exit;
    }

    $table = $allowed_tables[$type];

    // 3. The Permanent Delete Logic (PDO upgraded)
    // The "WHERE status = 'trash'" clause is a crucial safety check
    $sql = "DELETE FROM `{$table}` WHERE id = :id AND status = 'trash'";
    
    $stmt = $conn->prepare($sql);
    
    if ($stmt === false) {
        $errorInfo = $conn->errorInfo();
        throw new Exception('Failed to prepare the SQL statement: ' . $errorInfo[2]);
    }

    // Replace bind_param with PDO execute array
    if ($stmt->execute([':id' => $id])) {
        if ($stmt->rowCount() > 0) {
            sendResponse(true, null, 'Item permanently deleted.', 200);
        } else {
            sendResponse(false, null, 'Item not found in trash.', 404, 'NOT_FOUND');
        }
    } else {
        $errorInfo = $stmt->errorInfo();
        throw new Exception("Database query failed: " . $errorInfo[2]);
    }

} catch (Exception $e) {
    error_log("Error in delete_permanent.php: " . $e->getMessage());
    sendResponse(false, null, 'An internal server error occurred.', 500, 'SERVER_ERROR');
}

$conn = null;
?>
