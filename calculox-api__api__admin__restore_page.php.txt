<?php
// Core utilities
require_once '../../core/config.php';
require_once '../../core/response.php';
require_once '../../core/auth.php';

// --- SECURITY CHECK ---
$admin_data = verifyAdminToken();

try {
    // 1. Get Incoming Data
    $json = file_get_contents('php://input');
    $data = json_decode($json);

    // 2. Validation
    if (!isset($data->id) || !is_numeric($data->id)) {
        sendResponse(false, null, 'A valid Page ID is required.', 400, 'INVALID_INPUT');
    }
    $pageId = (int)$data->id;

    // 3. The Intelligent Restore Logic
    // Added: updated_at = updated_at to preserve the timestamp
    $sql = "UPDATE pages SET status = COALESCE(previous_status, 'draft'), previous_status = NULL, updated_at = updated_at WHERE id = :id AND status = 'trash'";
    $stmt = $conn->prepare($sql);

    if ($stmt === false) {
        $errorInfo = $conn->errorInfo();
        throw new Exception('Failed to prepare the SQL statement: ' . $errorInfo[2]);
    }

    if ($stmt->execute([':id' => $pageId])) {
        if ($stmt->rowCount() > 0) {
            sendResponse(true, null, 'Page restored successfully.', 200);
        } else {
            sendResponse(false, null, 'Page not found in trash or already restored.', 404, 'NOT_FOUND');
        }
    } else {
        $errorInfo = $stmt->errorInfo();
        throw new Exception("Database query failed: " . $errorInfo[2]);
    }

} catch (Exception $e) {
    error_log("Error in restore_page.php: " . $e->getMessage());
    sendResponse(false, null, 'An internal server error occurred.', 500, 'SERVER_ERROR');
}

$conn = null;
?>
