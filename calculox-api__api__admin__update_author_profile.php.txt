<?php
// Core utilities
require_once '../../core/config.php';
require_once '../../core/response.php';
require_once '../../core/auth.php';

// --- SECURITY CHECK ---
$admin_data = verifyAdminToken();

function getValidSlug($conn, $providedSlug, $name, $excludeId) {
    $baseSlug = !empty($providedSlug) ? $providedSlug : $name;
    $baseSlug = strtolower(trim(preg_replace('/[^A-Za-z0-9-]+/', '-', $baseSlug)));
    $baseSlug = preg_replace('/-+/', '-', $baseSlug);
    $baseSlug = trim($baseSlug, '-');
    if (empty($baseSlug)) $baseSlug = 'author';

    $slug = $baseSlug;
    $counter = 1;
    while(true) {
        $stmt = $conn->prepare("SELECT id FROM author_profiles WHERE slug = :slug AND id != :id");
        $stmt->execute([':slug' => $slug, ':id' => $excludeId]);
        if (!$stmt->fetch()) return $slug;
        $slug = $baseSlug . '-' . $counter++;
    }
}

try {
    $json = file_get_contents('php://input');
    $data = json_decode($json);

    $id = isset($data->id) ? (int)$data->id : null;
    $name = isset($data->name) ? trim($data->name) : '';
    $providedSlug = isset($data->slug) ? trim($data->slug) : '';
    $role = isset($data->role) && !empty(trim($data->role)) ? trim($data->role) : 'Author';
    $expertise = isset($data->expertise) && !empty(trim($data->expertise)) ? trim($data->expertise) : null;
    $short_description = isset($data->short_description) && !empty(trim($data->short_description)) ? trim($data->short_description) : null;
    $bio = isset($data->bio) ? trim($data->bio) : null;
    $profile_image_id = (isset($data->profile_image_id) && is_numeric($data->profile_image_id)) ? (int)$data->profile_image_id : null;

    if (!$id || empty($name)) {
        sendResponse(false, null, 'ID and Display Name are required.', 400, 'INVALID_INPUT');
    }

    // --- ENTERPRISE SECURITY FIX: Strict Length Boundaries (DoS & Buffer Overflow Mitigation) ---
    if (mb_strlen($name, 'UTF-8') > 100) sendResponse(false, null, 'Display name cannot exceed 100 characters.', 400, 'INVALID_INPUT');
    if (mb_strlen($providedSlug, 'UTF-8') > 100) sendResponse(false, null, 'Slug cannot exceed 100 characters.', 400, 'INVALID_INPUT');
    if (mb_strlen($role, 'UTF-8') > 100) sendResponse(false, null, 'Role cannot exceed 100 characters.', 400, 'INVALID_INPUT');
    if ($expertise && mb_strlen($expertise, 'UTF-8') > 100) sendResponse(false, null, 'Expertise cannot exceed 100 characters.', 400, 'INVALID_INPUT');
    if ($short_description && mb_strlen($short_description, 'UTF-8') > 500) sendResponse(false, null, 'Short description cannot exceed 500 characters.', 400, 'INVALID_INPUT');
    if ($bio && mb_strlen($bio, 'UTF-8') > 1000) sendResponse(false, null, 'Biography cannot exceed 1000 characters.', 400, 'INVALID_INPUT');
    // --------------------------------------------------------------------------------------------

    $slug = getValidSlug($conn, $providedSlug, $name, $id);

    // --- ENTERPRISE SECURITY FIX: STRICT URL SCHEME VALIDATION (OWASP A03) ---
    $valid_social_links = new stdClass();
    if (isset($data->social_links) && is_object($data->social_links)) {
        foreach ($data->social_links as $platform => $url) {
            $url = trim((string)$url);
            if (!empty($url)) {
                if (mb_strlen($url, 'UTF-8') > 500) continue; // Skip excessively long URLs
                
                $parsed = parse_url($url);
                if (isset($parsed['scheme']) && in_array(strtolower($parsed['scheme']), ['http', 'https'], true)) {
                    $valid_social_links->$platform = $url;
                }
            }
        }
    }
    $social_links = json_encode($valid_social_links);
    // ---------------------------------------------------------------------------------------------------

    $query = "UPDATE author_profiles 
              SET name = :name, slug = :slug, role = :role, expertise = :expertise, short_description = :short_description, bio = :bio, profile_image_id = :profile_image_id, social_links = :social_links 
              WHERE id = :id";
              
    $stmt = $conn->prepare($query);
    $result = $stmt->execute([
        ':name' => $name,
        ':slug' => $slug,
        ':role' => $role,
        ':expertise' => $expertise,
        ':short_description' => $short_description,
        ':bio' => $bio,
        ':profile_image_id' => $profile_image_id,
        ':social_links' => $social_links,
        ':id' => $id
    ]);

    if ($result) {
        sendResponse(true, ['slug' => $slug], 'Author profile updated successfully.', 200);
    } else {
        throw new Exception("Failed to update author record.");
    }

} catch (Exception $e) {
    error_log("DB Error in update_author_profile: " . $e->getMessage());
    sendResponse(false, null, 'An internal server error occurred.', 500, 'SERVER_ERROR');
}
$conn = null;
?>
