<?php
// Core utilities
require_once '../../core/config.php';
require_once '../../core/response.php';
require_once '../../core/auth.php'; // Ensure auth is present

// --- SECURITY CHECK ---
$admin_data = verifyAdminToken();

// Get the data sent from the frontend
$json = file_get_contents('php://input');
$data = json_decode($json);

// Basic Validation
if (!isset($data->id) || !is_numeric($data->id) || !isset($data->name) || !isset($data->slug)) {
    sendResponse(false, null, 'Category ID, name, and slug are required.', 400, 'INVALID_INPUT');
}

// Sanitize and Validate Inputs
$id = (int)$data->id;
$name = trim($data->name);
$slug = trim($data->slug);
$description = isset($data->description) ? trim($data->description) : null;

// --- SEO AND IMAGE FIELDS ---
$meta_title = isset($data->meta_title) ? trim($data->meta_title) : null;
$meta_description = isset($data->meta_description) ? trim($data->meta_description) : null;
$featured_image_id = (isset($data->featured_image_id) && $data->featured_image_id !== '') ? (int)$data->featured_image_id : null;

if (empty($name) || empty($slug)) {
    sendResponse(false, null, 'Category name and slug cannot be empty.', 400, 'INVALID_INPUT');
}

try {
    // --- 1. Check for Duplicates (excluding the current category) ---
    $sql_check = "SELECT id FROM categories WHERE (name = :name OR slug = :slug) AND id != :id";
    $stmt_check = $conn->prepare($sql_check);
    $stmt_check->execute([
        ':name' => $name,
        ':slug' => $slug,
        ':id' => $id
    ]);
    
    if ($stmt_check->fetch(PDO::FETCH_ASSOC)) {
        sendResponse(false, null, 'Another category with this name or slug already exists.', 409, 'DUPLICATE_ENTRY');
    }

    // --- 2. Update the Category in the database ---
    $sql_update = "UPDATE categories SET name = :name, slug = :slug, description = :description, meta_title = :meta_title, meta_description = :meta_description, featured_image_id = :featured_image_id WHERE id = :id";
    $stmt_update = $conn->prepare($sql_update);
    
    if ($stmt_update->execute([
        ':name' => $name,
        ':slug' => $slug,
        ':description' => $description,
        ':meta_title' => $meta_title,
        ':meta_description' => $meta_description,
        ':featured_image_id' => $featured_image_id,
        ':id' => $id
    ])) {
        // Fetch the updated category data to send back
        $sql_fetch = "
            SELECT c.id, c.name, c.slug, c.description, c.meta_title, c.meta_description, c.featured_image_id, c.created_at, m.upload_path AS featured_image_url
            FROM categories c
            LEFT JOIN media m ON c.featured_image_id = m.id
            WHERE c.id = :id
        ";
        $stmt_fetch = $conn->prepare($sql_fetch);
        $stmt_fetch->execute([':id' => $id]);
        $updated_category = $stmt_fetch->fetch(PDO::FETCH_ASSOC);

        // Get base URL for media paths (Strict Environment Extraction)
        $api_base_url = $_ENV['API_PUBLIC_URL'] ?? '';

        if ($updated_category) {
            $updated_category['post_count'] = 0;
            // Placeholder for frontend consistency
            $updated_category['id'] = (int)$updated_category['id'];
            $updated_category['featured_image_id'] = $updated_category['featured_image_id'] ? (int)$updated_category['featured_image_id'] : null;
            
            // Format URL
            if ($updated_category['featured_image_url']) {
                $updated_category['featured_image_url'] = rtrim($api_base_url, '/') . '/' . ltrim($updated_category['featured_image_url'], '/');
            }
        }

        sendResponse(true, $updated_category, 'Category updated successfully!', 200);
    } else {
        $errorInfo = $stmt_update->errorInfo();
        throw new Exception("Database update failed: " . $errorInfo[2]);
    }

} catch (Exception $e) {
    error_log("Error in update_category.php: " . $e->getMessage());
    sendResponse(false, null, 'An internal server error occurred.', 500, 'SERVER_ERROR');
}

$conn = null;
?>
