<?php

require_once dirname(__DIR__, 2) . '/core/config.php';

// --- ENTERPRISE FIX: START TIMING CLOCK IMMEDIATELY ---
$authStartTime = microtime(true);

/**
 * Universal Timing Gate:
 * Ensures all authentication responses take EXACTLY 2.0 seconds.
 * This completely eliminates User Enumeration via Timing Side-Channels.
 */
function secureAuthResponse($success, $data, $message, $statusCode = 200, $errorCode = null) {
    global $authStartTime;
    $elapsedMicroseconds = (microtime(true) - $authStartTime) * 1000000;
    $targetMicroseconds = 2000000; // 2.0 seconds (Covers maximum expected SMTP latency)
    
    if ($elapsedMicroseconds < $targetMicroseconds) {
        usleep((int)($targetMicroseconds - $elapsedMicroseconds));
    }
    
    sendResponse($success, $data, $message, $statusCode, $errorCode);
}

function generateAndSendOtp($conn, $admin, $ip_address) {
    $otp_code = random_int(100000, 999999);
    $otp_hash = password_hash($otp_code, PASSWORD_BCRYPT);
    $expires_at = gmdate('Y-m-d H:i:s', time() + 600); 

    $encrypted_ip = SecurityGuard::encryptIp($ip_address);
    $blind_index = SecurityGuard::generateBlindIndex($ip_address);
    
    $delete_stmt = $conn->prepare("DELETE FROM admin_otp_verifications WHERE admin_id = :admin_id AND ip_blind_index = :b_index");
    $delete_stmt->execute([':admin_id' => $admin['id'], ':b_index' => $blind_index]);
    
    $insert_stmt = $conn->prepare("INSERT INTO admin_otp_verifications (admin_id, otp_code_hash, ip_address, ip_blind_index, expires_at) VALUES (:admin_id, :otp_hash, :ip_enc, :b_index, :expires_at)");
    if (!$insert_stmt->execute([
        ':admin_id' => $admin['id'],
        ':otp_hash' => $otp_hash,
        ':ip_enc' => $encrypted_ip,
        ':b_index' => $blind_index,
        ':expires_at' => $expires_at
    ])) {
        throw new Exception('Failed to store OTP in the database.');
    }

    $safe_ip = htmlspecialchars($ip_address, ENT_QUOTES, 'UTF-8');
    $app_env = ucfirst($_ENV['APP_ENV'] ?? 'Production');
    
    $subject = '{{SITE_NAME}} Verification Code';
    $htmlBody = [
        'preheader'     => 'Your secure login code is ready',
        'event_type'    => 'AUTHENTICATION VERIFICATION',
        'greeting'      => "Hello {$admin['name']},",
        'alert_type'    => 'info',
        'alert_message' => 'A new login attempt requires verification.',
        'body'          => "<p>We received a request to access your administrator account. Please use the verification code below to complete your sign-in:</p>
                            <div class='otp-wrapper'>
                                <span class='otp-code' title='Double-click to select'>{$otp_code}</span>
                                <div class='otp-helper'>Double-click the code to select, then copy.</div>
                            </div>
                            <p>This code is valid for exactly <strong>10 minutes</strong>. If you didn't request this, your account is safe, but we recommend changing your password right away to ensure it stays that way.</p>",
        'meta_data'     => [
            'Triggered By' => $admin['email'],
            'Environment'  => $app_env,
            'IP Address'   => $safe_ip
        ]
    ];
    $altBody = "Your verification code is: {$otp_code}. It is valid for 10 minutes.";

    sendSecurityEmail($admin['email'], $admin['name'], $subject, $htmlBody, $altBody);
}

// --- Main Script Execution ---

$ip_address = getSecureRealIp();
$user_agent = $_SERVER['HTTP_USER_AGENT'] ?? 'unknown';
$ua_hash = hash('sha256', $user_agent);

$b_index = SecurityGuard::generateBlindIndex($ip_address);
$enc_ip = SecurityGuard::encryptIp($ip_address);

$json = file_get_contents('php://input');
$data = json_decode($json);
if (json_last_error() !== JSON_ERROR_NONE || !is_object($data)) {
    secureAuthResponse(false, null, 'Invalid request payload format.', 400, 'INVALID_PAYLOAD');
}

if (!isset($data->email) || !isset($data->password)) {
    secureAuthResponse(false, null, 'Email and password are required.', 400, 'INVALID_INPUT');
}

$email = trim($data->email);
$password = (string)$data->password;

$captcha_answer = trim($data->captcha_answer ?? '');
$captcha_token = trim($data->captcha_token ?? '');

if ($captcha_answer === '' || empty($captcha_token)) {
    secureAuthResponse(false, null, 'Security verification is required to proceed.', 400, 'CAPTCHA_REQUIRED');
}

try {
    $base64Standard = strtr($captcha_token, '-_', '+/');
    $padLength = strlen($base64Standard) % 4;
    
    if ($padLength > 0) {
        $base64Standard .= str_repeat('=', 4 - $padLength);
    }

    $decoded = base64_decode($base64Standard, true);
    if ($decoded === false) {
        secureAuthResponse(false, null, 'Security challenge validation failed. Please try again.', 400, 'CAPTCHA_INVALID');
    }

    $envelope = json_decode($decoded, true);
    if (!is_array($envelope) || !isset($envelope['p'], $envelope['s']) || !is_string($envelope['p']) || !is_string($envelope['s'])) {
        secureAuthResponse(false, null, 'Security challenge validation failed. Please try again.', 400, 'CAPTCHA_INVALID');
    }

    $payloadData       = $envelope['p']; 
    $providedSignature = $envelope['s']; 

    $secretKey = $_ENV['APP_ENCRYPTION_KEY'] ?? '';
    $decodedKey = base64_decode($secretKey, true);
    if (empty($secretKey) || $decodedKey === false || strlen($decodedKey) !== 32) {
        secureAuthResponse(false, null, 'System configuration error. Please contact administrator.', 500, 'SYS_ERR_01');
    }

    $expectedSignature = hash_hmac('sha256', $payloadData, $secretKey);
    if (!hash_equals($expectedSignature, $providedSignature)) {
        secureAuthResponse(false, null, 'Security challenge verification failed. Please try again.', 400, 'CAPTCHA_INVALID');
    }

    global $conn; 

    // --- Phase 1: ENTERPRISE FIX - TOCTOU Race Condition ---
    try {
        $stmt_consume = $conn->prepare("INSERT IGNORE INTO consumed_captchas (signature_hash, consumed_at) VALUES (:sig, UTC_TIMESTAMP())");
        $stmt_consume->execute([':sig' => $providedSignature]);
        
        if ($stmt_consume->rowCount() === 0) {
            secureAuthResponse(false, null, 'Security challenge has already been used. Please refresh.', 400, 'CAPTCHA_EXPIRED');
        }
    } catch (PDOException $e) {
        if ($e->getCode() == 23000 || $e->getCode() == 1062) {
            secureAuthResponse(false, null, 'Security challenge has already been used. Please refresh.', 400, 'CAPTCHA_EXPIRED');
        }
        throw $e;
    }

    $payload = json_decode($payloadData, true);
    
    if (!is_array($payload) || !isset($payload['a'], $payload['e'], $payload['ip'])) {
        secureAuthResponse(false, null, 'Invalid security session parameters.', 400, 'CAPTCHA_INVALID');
    }

    if (time() > (int)$payload['e']) {
        secureAuthResponse(false, null, 'Security challenge has expired. Please refresh.', 400, 'CAPTCHA_EXPIRED');
    }

    if ($payload['ip'] !== $ip_address) {
        secureAuthResponse(false, null, 'Network change detected. Please request a new challenge.', 400, 'CAPTCHA_INVALID_IP');
    }

    if ((int)$captcha_answer !== (int)$payload['a']) {
        secureAuthResponse(false, null, 'Security challenge validation failed. Please try again.', 400, 'CAPTCHA_FAILED');
    }

} catch (Exception $e) {
    secureAuthResponse(false, null, 'Security check encountered an error. Please reload.', 400, 'CAPTCHA_ERROR');
}

try {
    $guard = new SecurityGuard($conn, $ip_address);
    $securityStatus = $guard->checkIpStatus();
    
    if (!$securityStatus['allowed']) {
        secureAuthResponse(false, null, $securityStatus['message'], 403, $securityStatus['code']);
    }

    $sql = "SELECT id, name, email, password, role, password_last_updated, current_session_jti FROM admins WHERE email = :email";
    $stmt = $conn->prepare($sql);
    $stmt->execute([':email' => $email]);
    $admin = $stmt->fetch(PDO::FETCH_ASSOC);
    
    $dummy_hash = '$2y$10$abcdefghijklmnopqrstuv'; 
    $isValidLogin = false;
    
    if ($admin) {
        if (password_verify($password, $admin['password'])) {
            $isValidLogin = true;
        }
    } else {
        password_verify($password, $dummy_hash);
    }

    if ($isValidLogin) {
        $clear_login_attempts = $conn->prepare("DELETE FROM login_attempts WHERE ip_blind_index = :b_index");
        $clear_login_attempts->execute([':b_index' => $b_index]);

        $clear_admin_failures = $conn->prepare("DELETE FROM admin_login_failures WHERE ip_blind_index = :b_index");
        $clear_admin_failures->execute([':b_index' => $b_index]);
        
        $settings_sql = "SELECT setting_key, setting_value FROM settings WHERE setting_key IN ('security_2fa_enabled', 'security_2fa_frequency', 'password_expiry_days')";
        $settings_result = $conn->query($settings_sql);
        $settings = [];
        
        while ($row = $settings_result->fetch(PDO::FETCH_ASSOC)) {
            $settings[$row['setting_key']] = $row['setting_value'];
        }
        
        $is_2fa_enabled = ($settings['security_2fa_enabled'] ?? 'true') === 'true';
        $frequency = $settings['security_2fa_frequency'] ?? 'daily';
        
        $expirySetting = $settings['password_expiry_days'] ?? '0';
        $passwordLastUpdated = $admin['password_last_updated'] ?? date('Y-m-d H:i:s');
        $daysRemaining = null;
        $isExpired = false;
        $showWarning = false;

        if ($expirySetting !== '0') {
            $expiryDays = (int)$expirySetting;
            $lastUpdatedDate = new DateTime($passwordLastUpdated);
            $currentDate = new DateTime();
            $interval = $currentDate->diff($lastUpdatedDate);
            $daysPassed = $interval->days;

            $daysRemaining = $expiryDays - $daysPassed;
            
            if ($daysRemaining <= 0) {
                $isExpired = true;
                $daysRemaining = 0;
            } elseif ($daysRemaining <= 7) {
                $showWarning = true;
            }
        }

        $require_otp = false;
        
        if ($is_2fa_enabled) {
            if ($frequency === 'always') {
                $require_otp = true;
            } else {
                // ENTERPRISE FIX: Use absolute UTC timestamp for precise Rolling 24-Hour expiration
                // This eliminates timezone offset bugs (NOW() drift) between PHP and MySQL.
                $strict_24h_threshold = gmdate('Y-m-d H:i:s', time() - 86400); 
                
                $check_sql = "SELECT id FROM admin_trusted_devices WHERE admin_id = :admin_id AND ip_blind_index = :b_index AND user_agent_hash = :ua_hash AND last_verified_at > :trust_threshold";
                $trust_stmt = $conn->prepare($check_sql);
                $trust_stmt->execute([
                    ':admin_id' => $admin['id'], 
                    ':b_index' => $b_index, 
                    ':ua_hash' => $ua_hash,
                    ':trust_threshold' => $strict_24h_threshold
                ]);
                
                if (!$trust_stmt->fetch(PDO::FETCH_ASSOC)) {
                    $require_otp = true;
                }
            }
        }

        if ($require_otp) {
            generateAndSendOtp($conn, $admin, $ip_address);
            secureAuthResponse(true, ['otp_required' => true], 'Verification code sent to your email.');
        } else {
            // ENTERPRISE FIX: REMOVED "Sliding Window" trust refresh.
            // Previously, successful logins updated last_verified_at = NOW(), 
            // keeping the device trusted infinitely if the user logged in daily.
            // By omitting that update, we enforce an absolute 24-hour expiration from the exact moment of OTP verification.

            // --- THE NEW ADVANCED DUAL-TOKEN LOGIC (DYNAMIC T) ---
            $allowed_timeouts = [5, 10, 15];
            $timeout_input = isset($data->session_timeout) ? (int)$data->session_timeout : 15;
            $t_minutes = in_array($timeout_input, $allowed_timeouts) ? $timeout_input : 15;
            
            // Access Token Validity: Set to 50% of the selected T.
            $access_ttl = (int)(($t_minutes * 60) / 2);
            
            // Refresh Token Validity: Set to 100% of the selected T.
            $refresh_ttl = $t_minutes * 60;
            
            $jwtSecretKey = $_ENV['JWT_SECRET'];
            $session_jti = bin2hex(random_bytes(32));
            $access_jti = bin2hex(random_bytes(16));
            $refresh_jti = bin2hex(random_bytes(16));
            
            $access_payload = [
                'iat' => time(), 
                'iss' => $_SERVER['SERVER_NAME'], 
                'exp' => time() + $access_ttl, 
                'jti' => $access_jti,
                'session_jti' => $session_jti,
                'data' => ['adminId' => $admin['id'], 'adminRole' => $admin['role']]
            ];
            
            $refresh_payload = [
                'iat' => time(), 
                'iss' => $_SERVER['SERVER_NAME'], 
                'exp' => time() + $refresh_ttl,
                'jti' => $refresh_jti,
                'session_jti' => $session_jti,
                'data' => ['adminId' => $admin['id']]
            ];
            
            $accessToken = JWT::encode($access_payload, $jwtSecretKey, 'HS256');
            $refreshToken = JWT::encode($refresh_payload, $jwtSecretKey, 'HS256');
            
            $cookieDomain = !empty($_ENV['COOKIE_DOMAIN']) ? trim($_ENV['COOKIE_DOMAIN']) : '';
            
            setcookie('calcpro_access_token', $accessToken, [
                'expires' => time() + $access_ttl, 'path' => '/', 'domain' => $cookieDomain, 'secure' => true, 'httponly' => true, 'samesite' => 'None'
            ]);
            
            setcookie('calcpro_refresh_token', $refreshToken, [
                'expires' => time() + $refresh_ttl, 'path' => '/', 'domain' => $cookieDomain, 'secure' => true, 'httponly' => true, 'samesite' => 'None'
            ]);
            
            // Clear legacy token 
            setcookie('calcpro_admin_token', '', [
                'expires' => time() - 3600, 'path' => '/', 'domain' => $cookieDomain, 'secure' => true, 'httponly' => true, 'samesite' => 'None'
            ]);
            
            // --- Phase 2: ENTERPRISE FIX - BACKEND CSRF GENERATION (HMAC Double Submit) ---
            $rawCsrf = bin2hex(random_bytes(32));
            $hmacSecret = base64_decode($_ENV['APP_ENCRYPTION_KEY'] ?? '');
            if (empty($hmacSecret)) {
                $hmacSecret = $_ENV['JWT_SECRET'] ?? 'fallback_secret';
            }
            $hmacCsrf = hash_hmac('sha256', $rawCsrf, $hmacSecret);
            
            // 1. HttpOnly Validator Cookie (Secured from XSS)
            setcookie('calcpro_csrf_token', $hmacCsrf, [
                'expires' => time() + $refresh_ttl, 'path' => '/', 'domain' => $cookieDomain, 'secure' => true, 'httponly' => true, 'samesite' => 'None'
            ]);
            
            // 2. JS-Readable Payload Cookie
            setcookie('calcpro_csrf_payload', $rawCsrf, [
                'expires' => time() + $refresh_ttl, 'path' => '/', 'domain' => $cookieDomain, 'secure' => true, 'httponly' => false, 'samesite' => 'None'
            ]);
            
            // --- SECURITY ALERT: CONCURRENT OVERRIDE EMAIL ---
            if (!empty($admin['current_session_jti'])) {
                $safe_ip_alert = htmlspecialchars($ip_address, ENT_QUOTES, 'UTF-8');
                $app_env_alert = ucfirst($_ENV['APP_ENV'] ?? 'Production');
                
                $subjectAlert = 'Security Notice: New Login Detected';
                $htmlBodyAlert = [
                    'preheader'     => 'Your account was accessed from a new device.',
                    'event_type'    => 'CONCURRENT LOGIN ALERT',
                    'greeting'      => "Hello {$admin['name']},",
                    'alert_type'    => 'warning',
                    'alert_message' => 'A new login just ended your previous active session.',
                    'body'          => "<p>We detected a new successful login to your administrator account. Because your account enforces a strict Single Active Session policy, your previous session has been securely terminated.</p>
                                        <p><strong>If this was you:</strong> You can safely ignore this email.</p>
                                        <p><strong>If you did NOT log in recently:</strong> Someone else may have accessed your account. Please reset your password immediately using the 'Forgot Password' link on the login page to lock down your account.</p>",
                    'meta_data'     => [
                        'Environment' => $app_env_alert,
                        'IP Address'  => $safe_ip_alert,
                        'Action'      => 'Previous session completely revoked'
                    ],
                    'action_url'    => rtrim($_ENV['ADMIN_FRONTEND_URL'] ?? '', '/') . "/login", // ENTERPRISE FIX: Removed /#/admin fragment
                    'action_text'   => 'Secure Your Account'
                ];
                
                $altBodyAlert = "A new login just ended your previous active session. If this was not you, please reset your password immediately to secure your account.";
                
                sendSecurityEmail($admin['email'], $admin['name'], $subjectAlert, $htmlBodyAlert, $altBodyAlert);
            }

            // Bind new session JTI to DB
            $update_login = $conn->prepare("UPDATE admins SET last_login = NOW(), current_session_jti = :sjti WHERE id = :admin_id");
            $update_login->execute([':admin_id' => $admin['id'], ':sjti' => $session_jti]);

            $clear_specific_reset = $conn->prepare("DELETE FROM admin_password_resets WHERE admin_id = :admin_id AND ip_blind_index = :b_index");
            $clear_specific_reset->execute([':admin_id' => $admin['id'], ':b_index' => $b_index]);

            $clear_otp = $conn->prepare("DELETE FROM admin_otp_verifications WHERE admin_id = :admin_id AND ip_blind_index = :b_index");
            $clear_otp->execute([':admin_id' => $admin['id'], ':b_index' => $b_index]);

            $clear_unblock = $conn->prepare("DELETE FROM admin_unblock_tokens WHERE admin_id = :admin_id AND ip_blind_index = :b_index");
            $clear_unblock->execute([':admin_id' => $admin['id'], ':b_index' => $b_index]);

            $responseData = [
                'user' => [
                    'id' => $admin['id'], 
                    'name' => $admin['name'], 
                    'email' => $admin['email'], 
                    'role' => $admin['role'],
                    'passwordSecurity' => [
                        'daysRemaining' => $daysRemaining,
                        'isExpired'     => $isExpired,
                        'showWarning'   => $showWarning,
                        'expirySetting' => (int)$expirySetting
                    ]
                ]
            ];
            
            secureAuthResponse(true, $responseData, 'Login successful!', 200);
        }

    } else {
        if ($admin) {
            $guard->logFailure($email, $admin['id'], 'password');
        } else {
            $guard->logFailure($email, null, 'email');
        }

        $newStatus = $guard->checkIpStatus();
        
        if (!$newStatus['allowed']) {
            if ($admin) {
                $unblock_token = bin2hex(random_bytes(32));
                $unblock_token_hash = hash('sha256', $unblock_token);
                $unblock_expiry = gmdate('Y-m-d H:i:s', time() + 3600);
                $current_utc = gmdate('Y-m-d H:i:s');
                
                $block_level = ($newStatus['code'] === 'IP_PERMABANNED') ? 'permanent' : 'temporary';
                
                $insert_token_stmt = $conn->prepare("
                    INSERT INTO admin_unblock_tokens 
                    (admin_id, block_level, ip_address_blocked, ip_blind_index, token_hash, expires_at, created_at) 
                    VALUES (:admin_id, :block_level, :ip_enc, :b_index, :token_hash, :expires_at, :created_at)
                    ON DUPLICATE KEY UPDATE 
                    block_level = VALUES(block_level),
                    ip_address_blocked = VALUES(ip_address_blocked),
                    token_hash = VALUES(token_hash),
                    expires_at = VALUES(expires_at),
                    created_at = VALUES(created_at)
                ");
                
                if ($insert_token_stmt->execute([
                    ':admin_id' => $admin['id'],
                    ':block_level' => $block_level,
                    ':ip_enc' => $enc_ip,
                    ':b_index' => $b_index,
                    ':token_hash' => $unblock_token_hash,
                    ':expires_at' => $unblock_expiry,
                    ':created_at' => $current_utc
                ])) {
                    // ENTERPRISE FIX: Removed /#/admin/ fragment
                    $unlock_link = rtrim($_ENV['ADMIN_FRONTEND_URL'], '/') . "/unlock-account/" . $unblock_token;
                    
                    $safe_ip = htmlspecialchars($ip_address, ENT_QUOTES, 'UTF-8');
                    $app_env = ucfirst($_ENV['APP_ENV'] ?? 'Production');
                    
                    if ($block_level === 'permanent') {
                        $subject = 'Security Alert: Account Suspended';
                        $htmlBody = [
                            'preheader'     => 'Urgent: Unauthorized Access Blocked',
                            'event_type'    => 'CRITICAL SECURITY BREACH',
                            'greeting'      => "Attention {$admin['name']},",
                            'alert_type'    => 'danger',
                            'alert_message' => 'An IP address has been permanently blacklisted due to repeated severe violations against your account.',
                            'body'          => "<p>We detected multiple unauthorized login attempts targeting your administrator account. As a precaution, the offending IP address has been permanently banned from our network.</p>
                                                <p><strong>If this was you</strong> &mdash; for example, if you forgot your password &mdash; you can verify your identity and lift the ban using the button below.</p>
                                                <p><strong>If this wasn't you</strong>, your account is safe. However, we strongly recommend changing your password right away &mdash; from your Admin Profile if you're logged in, or via Forgot Password on the login screen if you're not.</p>",
                            'meta_data'     => [
                                'Triggered By' => 'System Security Protocol',
                                'Environment'  => $app_env,
                                'Offending IP' => $safe_ip
                            ],
                            'action_url'    => $unlock_link,
                            'action_text'   => 'Revoke Ban & Unblock IP'
                        ];
                    
                        $altBody = "Permanent Ban Unlock Link: {$unlock_link}\n\nIf this wasn't you, please reset your password immediately.";
                    } else {
                        $subject = 'Security Alert: Account Locked';
                        $htmlBody = [
                            'preheader'     => 'Security Alert: Temporary Lock Initiated',
                            'event_type'    => 'SECURITY LOCKOUT',
                            'greeting'      => "Hello {$admin['name']},",
                            'alert_type'    => 'warning',
                            'alert_message' => 'An IP address has been temporarily blocked due to suspicious activity.',
                            'body'          => "<p>We detected multiple failed login attempts targeting your administrator account. As a protective measure, we have temporarily locked out the offending IP address.</p>
                                                <p><strong>If this was you</strong> &mdash; for example, if you forgot your password &mdash; you can instantly unblock your access using the button below.</p>
                                                <p><strong>If this wasn't you</strong>, your account is currently safe. However, we strongly recommend changing your password right away &mdash; from your Admin Profile if you're logged in, or via Forgot Password on the login screen if you're not.</p>",
                            'meta_data'     => [
                                'Triggered By' => 'System Security Protocol',
                                'Environment'  => $app_env,
                                'Offending IP' => $safe_ip
                            ],
                            'action_url'    => $unlock_link,
                            'action_text'   => 'Unblock This IP Address'
                        ];
                     
                        $altBody = "Unlock Link: {$unlock_link}\n\nIf this wasn't you, please reset your password immediately.";
                    }
                    
                    sendSecurityEmail($admin['email'], $admin['name'], $subject, $htmlBody, $altBody);
                }
            }
            secureAuthResponse(false, null, $newStatus['message'], 429, $newStatus['code']);
        }
        secureAuthResponse(false, null, 'Invalid email or password.', 401, 'AUTH_FAILED');
    }

} catch (Exception $e) {
    error_log("Error in admin_login.php: " . $e->getMessage());
    secureAuthResponse(false, null, 'An internal server error occurred. Please check the system logs.', 500, 'SERVER_ERROR');
}
