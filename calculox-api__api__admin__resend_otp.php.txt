<?php
// File: calcpro-api/api/admin/resend_otp.php

use PHPMailer\PHPMailer\PHPMailer;
use PHPMailer\PHPMailer\Exception;

require_once dirname(__DIR__, 2) . '/core/config.php';
require_once dirname(__DIR__, 2) . '/core/response.php';
require_once dirname(__DIR__, 2) . '/core/SecurityGuard.php';

try {
    $ip_address = getSecureRealIp();
    $b_index = SecurityGuard::generateBlindIndex($ip_address);

    $json = file_get_contents('php://input');
    $data = json_decode($json);

    if (json_last_error() !== JSON_ERROR_NONE || !is_object($data)) {
        sendResponse(false, null, 'Invalid request payload format.', 400, 'INVALID_PAYLOAD');
    }

    if (!isset($data->email)) {
        sendResponse(false, null, 'Email address is required to resend OTP.', 400, 'INVALID_INPUT');
    }

    $email = trim($data->email);

    $guard = new SecurityGuard($conn, $ip_address);
    $securityStatus = $guard->checkIpStatus();
    if (!$securityStatus['allowed']) {
        sendResponse(false, null, $securityStatus['message'], 403, $securityStatus['code']);
    }

    $sql = "SELECT id, name, email FROM admins WHERE email = :email";
    $stmt = $conn->prepare($sql);
    $stmt->execute([':email' => $email]);
    $admin = $stmt->fetch(PDO::FETCH_ASSOC);

    if (!$admin) {
        sendResponse(true, null, 'If an account exists, a new verification code has been sent.');
    }

    $check_stmt = $conn->prepare("SELECT created_at FROM admin_otp_verifications WHERE admin_id = :admin_id AND ip_blind_index = :b_index ORDER BY created_at DESC LIMIT 1");
    $check_stmt->execute([':admin_id' => $admin['id'], ':b_index' => $b_index]);
    if ($last_otp = $check_stmt->fetch(PDO::FETCH_ASSOC)) {
        $time_since_last = time() - strtotime($last_otp['created_at'] . ' UTC');
        if ($time_since_last < 60) {
            sendResponse(false, null, 'Please wait 60 seconds before requesting a new code.', 429, 'RATE_LIMIT_EXCEEDED');
        }
    }

    $otp_code = random_int(100000, 999999);
    $otp_hash = password_hash($otp_code, PASSWORD_BCRYPT);
    $expires_at = gmdate('Y-m-d H:i:s', time() + 600); 

    $encrypted_ip = SecurityGuard::encryptIp($ip_address);

    $delete_stmt = $conn->prepare("DELETE FROM admin_otp_verifications WHERE admin_id = :admin_id AND ip_blind_index = :b_index");
    $delete_stmt->execute([':admin_id' => $admin['id'], ':b_index' => $b_index]);

    $insert_stmt = $conn->prepare("INSERT INTO admin_otp_verifications (admin_id, otp_code_hash, ip_address, ip_blind_index, expires_at) VALUES (:admin_id, :otp_hash, :ip_enc, :b_index, :expires_at)");
    $insert_stmt->execute([
        ':admin_id' => $admin['id'],
        ':otp_hash' => $otp_hash,
        ':ip_enc' => $encrypted_ip,
        ':b_index' => $b_index,
        ':expires_at' => $expires_at
    ]);

    $safe_ip = htmlspecialchars($ip_address, ENT_QUOTES, 'UTF-8');
    $app_env = ucfirst($_ENV['APP_ENV'] ?? 'Production');
    
    $subject = 'Your New {{SITE_NAME}} Verification Code';
    
    $htmlBody = [
        'preheader'     => 'Your new secure login code is ready',
        'event_type'    => 'AUTHENTICATION VERIFICATION',
        'greeting'      => "Hello {$admin['name']},",
        'alert_type'    => 'info',
        'alert_message' => 'A new login attempt requires verification.',
        'body'          => "<p>We received a request to issue a new verification code for your administrator account. Please use the code below to securely authenticate your session:</p>
                            <div class='otp-wrapper'>
                                <span class='otp-code' title='Double-click to select'>{$otp_code}</span>
                                <div class='otp-helper'>Double-click the code to select, then copy.</div>
                            </div>
                            <p>This code is valid for exactly <strong>10 minutes</strong>. If you didn't request this, your account is safe, but we recommend changing your password right away to ensure it stays that way.</p>",
        'meta_data'     => [
            'Triggered By' => $admin['email'],
            'Environment'  => $app_env,
            'IP Address'   => $safe_ip
        ]
    ];
    $altBody = "Your new verification code is: {$otp_code}. It is valid for 10 minutes.";

    sendSecurityEmail($admin['email'], $admin['name'], $subject, $htmlBody, $altBody);

    sendResponse(true, null, 'A new verification code has been sent to your email.');

} catch (Exception $e) {
    error_log("Error in resend_otp.php: " . $e->getMessage());
    sendResponse(false, null, 'An internal server error occurred.', 500, 'SERVER_ERROR');
}

$conn = null;
?>
