<?php
// Core utilities
require_once '../../core/config.php';
require_once '../../core/response.php';
require_once '../../core/auth.php';

// --- SECURITY CHECK ---
$admin_data = verifyAdminToken();

try {
    // --- 1. Get Incoming Data ---
    $json = file_get_contents('php://input');
    $data = json_decode($json);

    // --- 2. Validation ---
    if (!isset($data->ids) || !is_array($data->ids) || empty($data->ids)) {
        sendResponse(false, null, 'No items selected.', 400, 'INVALID_INPUT');
    }
    if (!isset($data->action)) {
        sendResponse(false, null, 'No action specified.', 400, 'INVALID_INPUT');
    }

    $action = $data->action;
    
    // Sanitize IDs to ensure they are all integers
    $ids = array_filter(array_map('intval', $data->ids));
    
    if (empty($ids)) {
        sendResponse(false, null, 'Invalid IDs provided.', 400, 'INVALID_INPUT');
    }

    // Generate dynamic secure placeholders (e.g., "?, ?, ?")
    $placeholders = implode(',', array_fill(0, count($ids), '?'));
    $paramsList = array_values($ids);

    // --- 3. Execute Action ---
    $sql = "";
    $successMsg = "";

    switch ($action) {
        case 'trash':
            $sql = "UPDATE redirects SET status = 'trash' WHERE id IN ($placeholders) AND status != 'trash'";
            $successMsg = "Selected redirects moved to Trash.";
            break;

        case 'delete_permanent':
            $sql = "DELETE FROM redirects WHERE id IN ($placeholders) AND status = 'trash'";
            $successMsg = "Selected redirects permanently deleted.";
            break;

        case 'restore':
            $sql = "UPDATE redirects SET status = 'active' WHERE id IN ($placeholders) AND status = 'trash'";
            $successMsg = "Selected redirects restored.";
            break;

        case 'activate':
            $sql = "UPDATE redirects SET status = 'active' WHERE id IN ($placeholders) AND status != 'trash'";
            $successMsg = "Selected redirects activated.";
            break;

        case 'deactivate':
            $sql = "UPDATE redirects SET status = 'inactive' WHERE id IN ($placeholders) AND status != 'trash'";
            $successMsg = "Selected redirects deactivated.";
            break;

        default:
            sendResponse(false, null, 'Invalid action specified.', 400, 'INVALID_ACTION');
            break;
    }

    // --- 4. Run Query ---
    $stmt = $conn->prepare($sql);

    if ($stmt === false) {
        $errorInfo = $conn->errorInfo();
        throw new Exception("Database prepare statement failed: " . $errorInfo[2]);
    }

    if ($stmt->execute($paramsList)) {
         $affected = $stmt->rowCount();
         if ($affected > 0) {
             sendResponse(true, ['affected' => $affected], $successMsg, 200);
         } else {
             sendResponse(false, null, 'No records were updated. They may not meet the criteria for this action.', 404, 'NO_CHANGES');
         }
    } else {
         $errorInfo = $stmt->errorInfo();
         throw new Exception("Database query failed: " . $errorInfo[2]);
    }

} catch (Exception $e) {
    error_log("Error in manage_redirects_bulk.php: " . $e->getMessage());
    sendResponse(false, null, 'An internal server error occurred.', 500, 'SERVER_ERROR');
}

$conn = null;
?>
