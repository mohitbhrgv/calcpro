// --- Premium Viewport-Aware Headless Portal Dropdown ---
const CustomSelect = ({ value, onChange, options, className }) => {
  const [isOpen, setIsOpen] = useState(false);
  const dropdownRef = useRef(null);
  const portalRef = useRef(null);
  const [dropdownStyles, setDropdownStyles] = useState({});

  useEffect(() => {
    const handleClickOutside = (event) => {
      const clickedOutsideButton = dropdownRef.current && !dropdownRef.current.contains(event.target);
      const clickedOutsidePortal = portalRef.current && !portalRef.current.contains(event.target);
      
      if (clickedOutsideButton && clickedOutsidePortal) {
          setIsOpen(false);
      }
    };
    document.addEventListener('mousedown', handleClickOutside);
    
    const updatePosition = () => {
        if (isOpen && dropdownRef.current) {
            const rect = dropdownRef.current.getBoundingClientRect();
            const spaceBelow = window.innerHeight - rect.bottom;
            const placeAbove = spaceBelow < 250 && rect.top > spaceBelow;

            setDropdownStyles({
                top: placeAbove ? 'auto' : rect.bottom + window.scrollY + 4,
                bottom: placeAbove ? (window.innerHeight - rect.top - window.scrollY + 4) : 'auto',
                left: rect.left + window.scrollX,
                width: rect.width
            });
        }
    };
    
    window.addEventListener('scroll', updatePosition, true);
    window.addEventListener('resize', updatePosition);
    
    return () => {
        document.removeEventListener('mousedown', handleClickOutside);
        window.removeEventListener('scroll', updatePosition, true);
        window.removeEventListener('resize', updatePosition);
    };
  }, [isOpen]);

  const handleToggle = () => {
     if (!isOpen && dropdownRef.current) {
         const rect = dropdownRef.current.getBoundingClientRect();
         const spaceBelow = window.innerHeight - rect.bottom;
         const placeAbove = spaceBelow < 250 && rect.top > spaceBelow;

         setDropdownStyles({
            top: placeAbove ? 'auto' : rect.bottom + window.scrollY + 4,
            bottom: placeAbove ? (window.innerHeight - rect.top - window.scrollY + 4) : 'auto',
            left: rect.left + window.scrollX,
            width: rect.width
         });
     }
     setIsOpen(!isOpen);
  };

  const selectedLabel = options.find(opt => String(opt.value) === String(value))?.label || '';

  const dropdownContent = (
      <AnimatePresence>
        {isOpen && (
          <motion.div
            ref={portalRef}
            initial={{ opacity: 0, y: dropdownStyles.bottom !== 'auto' ? 5 : -5 }} 
            animate={{ opacity: 1, y: 0 }} 
            exit={{ opacity: 0, y: dropdownStyles.bottom !== 'auto' ? 5 : -5 }} 
            transition={{ duration: 0.15 }}
            style={dropdownStyles}
            className="absolute z-[99999] bg-white dark:bg-gray-800 border border-gray-200 dark:border-gray-700 rounded-xl shadow-2xl overflow-hidden py-1"
          >
            <div className="max-h-60 overflow-y-auto custom-scrollbar">
                {options.map((opt) => (
                  <button
                    key={opt.value}
                    type="button"
                    onClick={(e) => { e.stopPropagation(); onChange(opt.value); setIsOpen(false); }}
                    className={`w-full text-left px-4 py-2.5 text-sm transition-colors ${
                      String(value) === String(opt.value) 
                        ? 'bg-indigo-50 dark:bg-indigo-900/30 text-indigo-700 dark:text-indigo-400 font-bold' 
                        : 'text-gray-900 dark:text-gray-300 hover:bg-gray-50 dark:hover:bg-gray-700 font-medium'
                    }`}
                  >
                    {opt.label}
                  </button>
                ))}
            </div>
          </motion.div>
        )}
      </AnimatePresence>
  );

  return (
    <div className={`relative ${className || ''}`} ref={dropdownRef}>
      <button
        type="button"
        onClick={handleToggle}
        className="flex items-center justify-between w-full bg-white dark:bg-gray-800 border border-gray-300 dark:border-gray-600 text-gray-900 dark:text-white rounded-xl px-4 py-2.5 text-sm focus:ring-2 focus:ring-indigo-500 outline-none transition-shadow font-medium cursor-pointer shadow-sm"
      >
        <span className="truncate pr-2">{selectedLabel}</span>
        <SafeIcon 
            icon={FiChevronDown} 
            className={`w-4 h-4 text-gray-500 shrink-0 transition-transform duration-200 ${isOpen ? 'rotate-180' : ''}`} 
        />
      </button>
      {createPortal(dropdownContent, document.body)}
    </div>
  );
};

const limitOptions = [
    { value: 10, label: "10 Rows" },
    { value: 25, label: "25 Rows" },
    { value: 50, label: "50 Rows" },
    { value: 100, label: "100 Rows" }
];

const Posts = () => {
  const navigate = useNavigate();
  const { getCache, setCacheData, invalidateCache, isInitialLoadComplete } = useAdminCache();

  // --- Core State ---
  const [filterStatus, setFilterStatus] = useState("all");
  const [searchTerm, setSearchTerm] = useState("");
  const [debouncedSearch, setDebouncedSearch] = useState("");
  const [selectedCategory, setSelectedCategory] = useState("");
  const [selectedAuthor, setSelectedAuthor] = useState("");
  const [selectedIds, setSelectedIds] = useState([]);

  // Filter Options Data
  const [filterOptions, setFilterOptions] = useState({ categories: [], authors: [] });

  const cacheKey = `posts_${filterStatus}_${selectedCategory}_${selectedAuthor}`;
  const cachedData = getCache(cacheKey);

  const [allPosts, setAllPosts] = useState(cachedData?.data || []);
  const [counts, setCounts] = useState(cachedData?.counts || { all: 0, published: 0, draft: 0, trash: 0 });

  // UX State Architecture (FCP Setup)
  const [initialLoading, setInitialLoading] = useState(!cachedData);
  const [isUpdating, setIsUpdating] = useState(false);
  const [isManualRefresh, setIsManualRefresh] = useState(false);

  // Pagination & Display Settings
  const [isSettingsOpen, setIsSettingsOpen] = useState(false);
  const [isSubmitting, setIsSubmitting] = useState(false);
  const [displayLimit, setDisplayLimit] = useState(() => Number(localStorage.getItem('posts_display_limit')) || 25);
  const [tempDisplayLimit, setTempDisplayLimit] = useState(displayLimit);
  const [currentPage, setCurrentPage] = useState(1);

  // Modal State
  const [seoModal, setSeoModal] = useState({ isOpen: false, post: null, action: null });
  const [confirmDeleteInput, setConfirmDeleteInput] = useState("");

  // --- Fetch Filter Options (Categories/Authors) ---
  const fetchFilters = useCallback(async () => {
    try {
      const result = await authenticatedFetch("/api/admin/get_post_filters.php");
      if (result.success) {
        setFilterOptions(result.data);
      }
    } catch (e) {
      console.error("Failed to load filters");
    }
  }, []);

  useEffect(() => {
    fetchFilters();
  }, [fetchFilters]);

  // --- Fetch Posts Logic ---
  const fetchPosts = useCallback(
    async (forceRefresh = false) => {
      const currentKey = `posts_${filterStatus}_${selectedCategory}_${selectedAuthor}`;

      if (!forceRefresh) {
        const cached = getCache(currentKey);
        if (cached) {
          if (JSON.stringify(cached.data) !== JSON.stringify(allPosts)) {
            setAllPosts(cached.data);
            setCounts(cached.counts);
          }
          setInitialLoading(false);
          return;
        }
      }

      setIsUpdating(true);

      const params = new URLSearchParams({ status: filterStatus });
      if (selectedCategory) params.append("category", selectedCategory);
      if (selectedAuthor) params.append("author", selectedAuthor);

      try {
        const result = await authenticatedFetch(
          `/api/admin/get_posts.php?${params.toString()}`
        );
        if (result.success) {
          setAllPosts(result.data.posts);
          setCounts(result.data.counts);
          setCacheData(currentKey, {
            data: result.data.posts,
            counts: result.data.counts,
          });
        } else {
          toast.error(result.message || "Failed to fetch posts.");
        }
      } catch (error) {
        if (error.message !== "Unauthorized") {
          toast.error("Could not connect to the server.");
        }
      } finally {
        setIsUpdating(false);
        setInitialLoading(false);
      }
    },
    [filterStatus, selectedCategory, selectedAuthor, getCache, setCacheData, allPosts]
  );

  useEffect(() => {
    fetchPosts();
    setSelectedIds([]); 
  }, [filterStatus, selectedCategory, selectedAuthor, fetchPosts]);

  // Search Debounce Engine
  useEffect(() => {
    const handler = setTimeout(() => {
        setDebouncedSearch(searchTerm);
    }, 500);
    return () => clearTimeout(handler);
  }, [searchTerm]);

  // Reset pagination when dataset filters change
  useEffect(() => {
      setCurrentPage(1);
  }, [filterStatus, selectedCategory, selectedAuthor, debouncedSearch]);

  const filteredPosts = useMemo(() => {
    if (!debouncedSearch) return allPosts;
    return allPosts.filter(
      (post) =>
        post.title.toLowerCase().includes(debouncedSearch.toLowerCase()) ||
        (post.author_name && post.author_name.toLowerCase().includes(debouncedSearch.toLowerCase()))
    );
  }, [allPosts, debouncedSearch]);

  // Custom Pagination Math
  const totalRecords = filteredPosts.length;
  const totalPages = Math.max(1, Math.ceil(totalRecords / displayLimit));
  
  const paginatedPosts = useMemo(() => {
      const start = (currentPage - 1) * displayLimit;
      return filteredPosts.slice(start, start + displayLimit);
  }, [filteredPosts, currentPage, displayLimit]);

  // --- UX Enhancement: Manual Refresh with Labor Gap ---
  const handleManualRefresh = async () => {
    if (isUpdating || isManualRefresh) return;
    setIsManualRefresh(true);
    invalidateCache(`posts_${filterStatus}_${selectedCategory}_${selectedAuthor}`);
    await fetchPosts(true);
    setTimeout(() => setIsManualRefresh(false), 800); 
  };

  // --- UX Enhancement: Settings Manager & Dirty State ---
  const handleOpenSettings = () => {
      setTempDisplayLimit(displayLimit);
      setIsSettingsOpen(true);
  };

  const handleSaveSettings = async () => {
      setIsSubmitting(true);
      // Professional Labor Gap
      await new Promise(resolve => setTimeout(resolve, 800));
      
      setDisplayLimit(tempDisplayLimit);
      localStorage.setItem('posts_display_limit', tempDisplayLimit);
      setCurrentPage(1);
      
      setIsSubmitting(false);
      setIsSettingsOpen(false);
      toast.success("Display settings strictly updated.");
  };

  const isSettingsDirty = tempDisplayLimit !== displayLimit;

  // --- Handlers ---
  const handleSelectRow = (id) => {
    setSelectedIds((prev) => prev.includes(id) ? prev.filter((i) => i !== id) : [...prev, id]);
  };

  const handleSelectAll = (e) => {
    if (e.target.checked) {
      setSelectedIds(paginatedPosts.map((p) => p.id));
    } else {
      setSelectedIds([]);
    }
  };

  const handleEdit = (post) => navigate(`/content/posts/edit/${post.id}`);
  const handleView = (post) => window.open(`http://localhost:5174/#/blog/${post.slug}`, "_blank");

  // Generic API Action Handler
  const handleApiAction = useCallback(
    async (url, body, successMessage, errorMessage) => {
      try {
        setIsUpdating(true);
        const result = await authenticatedFetch(url, { method: "POST", body: JSON.stringify(body) });
        if (result.success) {
          toast.success(successMessage);
          invalidateCache("posts_");
          setSelectedIds([]);
          await fetchPosts(true);
        } else {
          toast.error(result.message || errorMessage);
        }
      } catch (error) {
        toast.error("Action failed.");
      } finally {
        setIsUpdating(false);
      }
    },
    [fetchPosts, invalidateCache]
  );

  // --- Action Triggers ---
  const initiateTrash = (post) => {
    setSeoModal({ isOpen: true, post, action: post.status === "published" ? "trash" : "trash_draft" });
  };
  const initiateDeletePermanent = (post) => {
    setConfirmDeleteInput("");
    setSeoModal({ isOpen: true, post, action: "delete_permanent" });
  };
  const initiateBulkAction = (actionType) => {
    if (selectedIds.length === 0) return;
    if (actionType === "delete_permanent") setConfirmDeleteInput("");
    setSeoModal({ isOpen: true, post: null, action: `bulk_${actionType}` });
  };

  const handleConfirmAction = () => {
    const { post, action } = seoModal;
    setSeoModal({ ...seoModal, isOpen: false });

    if (action === "trash" || action === "trash_draft") {
      handleApiAction(
        "/api/admin/delete_post.php",
        { id: post.id },
        "Post moved to trash.",
        "Failed to move post."
      );
    } else if (action === "delete_permanent") {
      handleApiAction(
        "/api/admin/delete_permanent.php",
        { id: post.id },
        "Post permanently deleted.",
        "Failed to delete post."
      );
    } else if (action.startsWith("bulk_")) {
      const bulkAction = action.replace("bulk_", "");
      handleApiAction(
        "/api/admin/manage_posts_bulk.php",
        { ids: selectedIds, action: bulkAction },
        "Bulk action completed successfully.",
        "Failed to perform bulk action."
      );
    }
  };

  const handleRestore = (post) => {
    handleApiAction(
      "/api/admin/restore_post.php",
      { id: post.id },
      "Post restored.",
      "Failed to restore."
    );
  };

  const handleCopyPath = () => {
    if (seoModal.post) {
      navigator.clipboard.writeText(`blog/${seoModal.post.slug}`);
      toast.success("Path copied!");
    }
  };

  // --- Modal Content Logic ---
  const getModalContent = () => {
    const { post, action } = seoModal;
    const count = selectedIds.length;

    if (action === "bulk_trash")
      return {
        title: `Trash ${count} Posts?`,
        type: "warning",
        icon: FiTrash2,
        colorClass: "text-amber-600",
        bgClass: "bg-amber-50 dark:bg-amber-900/20 border-amber-200 dark:border-amber-800",
        heading: "Bulk Trash Action",
        message: `Are you sure you want to move ${count} posts to the trash? They will stop being visible publically.`,
        buttonText: "Move to Trash",
        btnClass: "bg-amber-600 hover:bg-amber-700",
      };
    if (action === "bulk_restore")
      return {
        title: `Restore ${count} Posts?`,
        type: "default",
        icon: FiRefreshCw,
        colorClass: "text-blue-600",
        bgClass: "bg-blue-50 dark:bg-blue-900/20 border-blue-200 dark:border-blue-800",
        heading: "Bulk Restore",
        message: `Restore ${count} posts to their previous status?`,
        buttonText: "Restore All",
        btnClass: "bg-blue-600 hover:bg-blue-700",
      };
    if (action === "bulk_delete_permanent")
      return {
        title: `Delete ${count} Posts Forever?`,
        type: "danger",
        icon: FiAlertTriangle,
        colorClass: "text-red-600",
        bgClass: "bg-red-50 dark:bg-red-900/20 border-red-200 dark:border-red-800",
        heading: "Permanent Data Loss",
        message: `You are about to PERMANENTLY delete ${count} posts. This cannot be undone.`,
        buttonText: "Delete Forever",
        btnClass: "bg-red-600 hover:bg-red-700",
        requireInput: true,
      };

    if (!post) return {};

    if (action === "delete_permanent")
      return {
        title: "Permanent Deletion",
        type: "danger",
        icon: FiAlertTriangle,
        colorClass: "text-red-600",
        bgClass: "bg-red-50 dark:bg-red-900/20 border-red-200 dark:border-red-800",
        heading: "Permanent URL Loss",
        message: (
          <>You are removing <strong>"{post.title}"</strong> forever. This action cannot be undone.</>
        ),
        buttonText: "Delete Forever",
        btnClass: "bg-red-600 hover:bg-red-700",
        showRedirect: true,
        requireInput: true,
      };
    if (action === "trash")
      return {
        title: "Unpublish Post?",
        type: "warning",
        icon: FiAlertTriangle,
        colorClass: "text-amber-600",
        bgClass: "bg-amber-50 dark:bg-amber-900/20 border-amber-200 dark:border-amber-800",
        heading: "Potential 404 Error",
        message: (
          <>You are moving a <strong>Published</strong> post to Trash. The URL will become inaccessible.</>
        ),
        buttonText: "Move to Trash",
        btnClass: "bg-amber-600 hover:bg-amber-700",
        showRedirect: true,
      };

    return {
      title: "Move to Trash?",
      type: "default",
      icon: FiInfo,
      colorClass: "text-blue-600",
      bgClass: "bg-blue-50 dark:bg-blue-900/20 border-blue-200 dark:border-blue-800",
      heading: "Confirm Action",
      message: (
        <>Move <strong>"{post.title}"</strong> to trash? You can restore it later.</>
      ),
      buttonText: "Move to Trash",
      btnClass: "bg-blue-600 hover:bg-blue-700",
    };
  };
  const modalContent = getModalContent();

  if (!isInitialLoadComplete) return null;

  return (
    <div className="space-y-6">
      
      {/* --- DISPLAY SETTINGS MODAL --- */}
      <Modal isOpen={isSettingsOpen} onClose={() => !isSubmitting && setIsSettingsOpen(false)} title="Posts Configuration" size="md" disableAnimation={true}>
        <div className="space-y-5 pt-2 pb-4 max-h-[60vh] overflow-y-auto custom-scrollbar pr-2">
            
            <div className="bg-blue-50/50 dark:bg-blue-900/10 p-5 rounded-2xl border border-blue-100 dark:border-blue-800/30 shadow-sm transition-colors">
                <h3 className="text-sm font-extrabold text-blue-900 dark:text-blue-300 uppercase tracking-wider flex items-center gap-2 mb-2">
                    <SafeIcon icon={FiList} className="w-4 h-4 text-blue-500" />
                    Display Settings
                </h3>
                <p className="text-[11px] text-blue-800/80 dark:text-blue-200/70 mb-5 leading-relaxed font-medium">
                    Configure your local display preferences for the Posts module. This controls how many records are rendered per page.
                </p>
                <div>
                    <label className="block text-[11px] font-bold text-gray-500 dark:text-gray-400 uppercase tracking-wide mb-2">
                        Pagination Limit
                    </label>
                    <CustomSelect 
                        value={tempDisplayLimit} 
                        onChange={(val) => setTempDisplayLimit(parseInt(val))} 
                        options={limitOptions} 
                    />
                </div>
            </div>

        </div>
        <div className="flex justify-end gap-3 pt-4 border-t border-gray-100 dark:border-gray-700 bg-white dark:bg-gray-800">
            <button 
                onClick={() => setIsSettingsOpen(false)} 
                disabled={isSubmitting} 
                className="px-4 py-2.5 text-sm font-bold text-gray-600 dark:text-gray-300 bg-white dark:bg-gray-800 border border-gray-200 dark:border-gray-600 hover:bg-gray-50 dark:hover:bg-gray-700 rounded-lg transition-colors shadow-sm focus:outline-none disabled:opacity-50"
            >
                Cancel
            </button>
            <button 
                onClick={handleSaveSettings} 
                disabled={isSubmitting || !isSettingsDirty} 
                className="flex items-center gap-2 px-5 py-2.5 text-sm font-bold text-white bg-indigo-600 hover:bg-indigo-700 rounded-lg transition-all shadow-md active:scale-95 disabled:opacity-50 disabled:cursor-not-allowed focus:outline-none"
            >
                {isSubmitting ? (
                    <><SafeIcon icon={FiRefreshCw} className="animate-spin w-4 h-4" /> Saving...</>
                ) : (
                    <><SafeIcon icon={FiSave} className="w-4 h-4" /> Save Policies</>
                )}
            </button>
        </div>
      </Modal>

      {/* --- CONFIRMATION MODAL --- */}
      <Modal
        isOpen={seoModal.isOpen}
        onClose={() => setSeoModal({ ...seoModal, isOpen: false })}
        title={modalContent.title}
        type={modalContent.type}
        size="md"
        disableAnimation={true}
      >
        <div className="space-y-5">
          <div className={`${modalContent.bgClass} p-4 rounded-lg border flex gap-3 items-start`}>
            <SafeIcon icon={modalContent.icon} className={`w-5 h-5 ${modalContent.colorClass} shrink-0 mt-0.5`} />
            <div>
              <h4 className={`text-sm font-bold ${modalContent.colorClass === "text-amber-600" ? "text-amber-900 dark:text-amber-100" : modalContent.colorClass === "text-red-600" ? "text-red-900 dark:text-red-100" : "text-blue-900 dark:text-blue-100"}`}>
                {modalContent.heading}
              </h4>
              <p className={`text-xs mt-1 leading-relaxed ${modalContent.colorClass === "text-amber-600" ? "text-amber-800 dark:text-amber-200/80" : modalContent.colorClass === "text-red-600" ? "text-red-800 dark:text-red-200/80" : "text-blue-800 dark:text-blue-200/80"}`}>
                {modalContent.message}
              </p>
            </div>
          </div>

          {modalContent.showRedirect && (
            <div className="space-y-2">
              <label className="text-xs font-bold text-gray-500 uppercase tracking-wider">Source Path</label>
              <div className="flex rounded-md shadow-sm">
                <input
                  type="text"
                  value={`blog/${seoModal.post?.slug}`}
                  readOnly
                  className="flex-1 px-3 py-2 rounded-l-md border border-gray-300 bg-gray-100 text-gray-600 text-sm font-mono focus:outline-none"
                />
                <button onClick={handleCopyPath} className="px-4 border border-l-0 border-gray-300 rounded-r-md bg-white hover:bg-gray-50 text-gray-500 transition-colors focus:outline-none">
                  <SafeIcon icon={FiCopy} className="w-4 h-4" />
                </button>
              </div>
            </div>
          )}

          {modalContent.requireInput && (
            <div className="mt-3">
              <label className="block text-xs font-bold text-gray-500 mb-1 uppercase">
                Type <strong>DELETE</strong> to confirm:
              </label>
              <input
                type="text"
                value={confirmDeleteInput}
                onChange={(e) => setConfirmDeleteInput(e.target.value)}
                className="w-full px-3 py-2 border border-gray-300 dark:border-gray-600 rounded-lg outline-none text-sm focus:ring-2 focus:ring-red-500 bg-white dark:bg-gray-800 text-gray-900 dark:text-white"
                placeholder="DELETE"
              />
            </div>
          )}

          <div className="flex flex-col sm:flex-row gap-3 pt-2">
            {modalContent.showRedirect && (
              <button
                onClick={() => window.open("/#/seo/redirects", "_blank")}
                className="flex-1 flex items-center justify-center gap-2 px-4 py-2.5 border border-indigo-600 text-indigo-600 dark:text-indigo-400 rounded-lg hover:bg-indigo-50 dark:hover:bg-indigo-900/20 font-medium transition-colors text-sm focus:outline-none"
              >
                <SafeIcon icon={FiCornerUpRight} className="w-4 h-4" /> Set Redirect
              </button>
            )}
            <button
              onClick={handleConfirmAction}
              disabled={modalContent.requireInput && confirmDeleteInput !== "DELETE"}
              className={`flex-1 px-4 py-2.5 text-white rounded-lg font-bold shadow-sm transition-colors disabled:opacity-50 disabled:cursor-not-allowed text-sm focus:outline-none ${modalContent.btnClass}`}
            >
              {modalContent.buttonText}
            </button>
          </div>
        </div>
      </Modal>

      {/* --- HEADER CARD --- */}
      <motion.div 
        initial={{ opacity: 0, y: -10 }} animate={{ opacity: 1, y: 0 }} 
        className="flex items-start gap-4 p-5 rounded-xl bg-indigo-50 dark:bg-indigo-900/10 border border-indigo-100 dark:border-indigo-800/30 shadow-sm"
      >
        <div className="p-3 bg-indigo-100 dark:bg-indigo-800/50 rounded-xl text-indigo-600 dark:text-indigo-400 shrink-0 shadow-sm border border-indigo-200 dark:border-indigo-700/50">
          <SafeIcon icon={FiFileText} className="w-6 h-6" />
        </div>
        <div className="flex-1">
          <div className="flex flex-col md:flex-row md:items-center justify-between gap-4">
            <div>
              <h1 className="text-xl font-bold text-gray-900 dark:text-white tracking-tight">
                Blog Content
              </h1>
              <p className="text-sm text-gray-600 dark:text-gray-300 mt-1 leading-relaxed">
                Manage articles, news, and guides for the blog section of your website.
              </p>
            </div>
            <motion.button
              whileHover={{ scale: 1.02 }}
              whileTap={{ scale: 0.98 }}
              onClick={() => navigate("/content/posts/new")}
              className="flex items-center justify-center space-x-2 bg-indigo-600 text-white px-5 py-2.5 rounded-lg shadow-sm hover:bg-indigo-700 transition-colors font-bold text-sm whitespace-nowrap focus:outline-none"
            >
              <SafeIcon icon={FiPlus} className="w-4 h-4" />
              <span>New Post</span>
            </motion.button>
          </div>
        </div>
      </motion.div>

      {/* TIER 3: HTML DATA TABLE & TOOLBAR (FCP Optimized) */}
      <motion.div 
        initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} transition={{ delay: 0.2 }}
        className="bg-white dark:bg-gray-800 rounded-xl shadow-sm border border-gray-200 dark:border-gray-700 overflow-hidden relative flex flex-col min-h-[400px]"
      >
        {/* The dimming overlay logic */}
        <AnimatePresence>
          {(initialLoading || isUpdating) && (
            <motion.div
              initial={{ opacity: 0 }}
              animate={{ opacity: 1 }}
              exit={{ opacity: 0 }}
              className="absolute inset-0 bg-white/50 dark:bg-gray-800/50 backdrop-blur-[1px] z-10 flex items-center justify-center"
            >
              <LoadingSpinner />
            </motion.div>
          )}
        </AnimatePresence>

        <div className="px-6 py-4 border-b border-gray-200 dark:border-gray-700 bg-gray-50 dark:bg-gray-700/50 flex flex-col xl:flex-row justify-between items-center gap-4 relative z-20">
          {/* Left: TABS + FILTERS */}
          <div className="flex flex-wrap items-center gap-3 w-full xl:w-auto">
            {/* Status Tabs */}
            <div className="flex items-center gap-1 bg-white dark:bg-gray-800 p-1 rounded-lg border border-gray-200 dark:border-gray-600">
              {["all", "published", "draft", "trash"].map((status) => (
                <button
                  key={status}
                  onClick={() => {
                    setFilterStatus(status);
                    setSearchTerm("");
                    setSelectedIds([]);
                  }}
                  className={`px-3 py-1.5 text-xs font-bold rounded-md transition-all flex items-center gap-1.5 focus:outline-none ${
                    filterStatus === status
                      ? "bg-indigo-50 text-indigo-700 dark:bg-indigo-900/40 dark:text-indigo-300 shadow-sm ring-1 ring-black/5"
                      : "text-gray-500 hover:text-gray-700 dark:text-gray-400 dark:hover:text-gray-200 hover:bg-gray-50 dark:hover:bg-gray-700"
                  }`}
                >
                  {status.charAt(0).toUpperCase() + status.slice(1)}
                  <span
                    className={`px-1.5 py-0.5 rounded-full text-[10px] ${filterStatus === status ? "bg-indigo-200 text-indigo-800 dark:bg-indigo-800 dark:text-indigo-200" : "bg-gray-200 text-gray-600 dark:bg-gray-700 dark:text-gray-300"}`}
                  >
                    {counts[status]}
                  </span>
                </button>
              ))}
            </div>

            {/* Category Filter */}
            <div className="relative">
              <SafeIcon
                icon={FiFilter}
                className="absolute left-3 top-1/2 -translate-y-1/2 text-gray-400 w-3.5 h-3.5"
              />
              <select
                value={selectedCategory}
                onChange={(e) => setSelectedCategory(e.target.value)}
                className="pl-9 pr-8 py-2 border border-gray-300 dark:border-gray-600 rounded-lg text-sm font-medium bg-white dark:bg-gray-800 text-gray-900 dark:text-white focus:ring-2 focus:ring-indigo-500 outline-none appearance-none cursor-pointer shadow-sm hover:bg-gray-50 dark:hover:bg-gray-700"
              >
                <option value="">All Categories</option>
                {filterOptions.categories.map((cat) => (
                  <option key={cat.id} value={cat.id}>
                    {cat.name}
                  </option>
                ))}
              </select>
            </div>
          </div>

          {/* Right: BULK ACTIONS, SEARCH & SETTINGS */}
          <div className="flex items-center gap-3 w-full xl:w-auto justify-end">
            <AnimatePresence>
              {selectedIds.length > 0 && (
                <motion.div
                  initial={{ opacity: 0, x: 10 }}
                  animate={{ opacity: 1, x: 0 }}
                  exit={{ opacity: 0, x: 10 }}
                  className="flex items-center gap-2 pr-3 border-r border-gray-300 dark:border-gray-600"
                >
                  <span className="text-xs font-bold text-gray-500 hidden md:block">
                    {selectedIds.length} Selected
                  </span>
                  {filterStatus !== "trash" ? (
                    <button
                      onClick={() => initiateBulkAction("trash")}
                      className="flex items-center gap-1 px-3 py-1.5 bg-red-50 text-red-600 hover:bg-red-100 dark:bg-red-900/20 dark:hover:bg-red-900/40 rounded-lg text-xs font-bold transition-colors focus:outline-none shadow-sm"
                    >
                      <SafeIcon icon={FiTrash2} className="w-3.5 h-3.5" /> Trash
                    </button>
                  ) : (
                    <>
                      <button
                        onClick={() => initiateBulkAction("restore")}
                        className="flex items-center gap-1 px-3 py-1.5 bg-blue-50 text-blue-600 hover:bg-blue-100 dark:bg-blue-900/20 dark:hover:bg-blue-900/40 rounded-lg text-xs font-bold transition-colors focus:outline-none shadow-sm"
                      >
                        <SafeIcon icon={FiRefreshCw} className="w-3.5 h-3.5" />{" "}
                        Restore
                      </button>
                      <button
                        onClick={() => initiateBulkAction("delete_permanent")}
                        className="flex items-center gap-1 px-3 py-1.5 bg-red-50 text-red-600 hover:bg-red-100 dark:bg-red-900/20 dark:hover:bg-red-900/40 rounded-lg text-xs font-bold transition-colors focus:outline-none shadow-sm border border-red-100 dark:border-red-800/50"
                      >
                        <SafeIcon icon={FiAlertTriangle} className="w-3.5 h-3.5" />{" "}
                        Delete
                      </button>
                    </>
                  )}
                </motion.div>
              )}
            </AnimatePresence>

            <div className="relative w-full sm:w-64 flex items-center">
              <SafeIcon
                icon={FiSearch}
                className="absolute left-3 top-1/2 transform -translate-y-1/2 text-gray-400 w-4 h-4"
              />
              <input
                type="text"
                placeholder="Search posts..."
                value={searchTerm}
                onChange={(e) => setSearchTerm(e.target.value)}
                className="w-full pl-9 pr-4 py-2 border border-gray-300 dark:border-gray-600 rounded-lg text-sm bg-white text-gray-900 dark:bg-gray-800 dark:text-white focus:ring-2 focus:ring-indigo-500 outline-none transition-shadow shadow-sm"
              />
            </div>
            
            {/* Standardized Toolbar Actions (Settings & Refresh) */}
            <div className="flex items-center gap-1 border-l border-gray-200 dark:border-gray-700 pl-3 ml-auto shrink-0">
                <button
                    onClick={handleOpenSettings}
                    className="p-2 text-gray-400 hover:text-gray-700 dark:hover:text-gray-300 hover:bg-gray-200 dark:hover:bg-gray-600 rounded-lg transition-colors focus:outline-none"
                    title="Display Settings"
                >
                    <SafeIcon icon={FiSettings} className="w-4 h-4" />
                </button>
                <button
                    onClick={handleManualRefresh}
                    disabled={isUpdating || isManualRefresh}
                    className={`p-2 text-indigo-600 bg-indigo-50 hover:bg-indigo-100 dark:bg-indigo-900/20 dark:text-indigo-300 dark:hover:bg-indigo-900/40 rounded-lg transition-colors border border-indigo-100 dark:border-indigo-800 focus:outline-none ${
                        isManualRefresh || isUpdating ? "opacity-50 cursor-not-allowed" : ""
                    }`}
                    title="Refresh Data"
                >
                    <SafeIcon icon={FiRefreshCw} className={`w-4 h-4 ${isManualRefresh ? "animate-spin" : ""}`} />
                </button>
            </div>
          </div>
        </div>

        <div className={`w-full overflow-x-auto flex flex-col flex-1 transition-opacity duration-200 ${(initialLoading || isUpdating) ? "opacity-50 pointer-events-none" : ""}`}>
          {paginatedPosts.length === 0 && !initialLoading ? (
            <div className="flex flex-col items-center justify-center p-16 text-center flex-1">
              <SafeIcon icon={FiFileText} className="w-12 h-12 text-gray-400 dark:text-gray-500 mb-4 opacity-50" />
              <h3 className="text-lg font-bold text-gray-900 dark:text-white mb-2">No Posts Found</h3>
              <p className="text-sm text-gray-500 dark:text-gray-400 max-w-sm">No posts match the current filter or search criteria.</p>
            </div>
          ) : (
            <table className="w-full text-left border-collapse min-w-[1000px]">
              <thead>
                <tr className="bg-gray-50 dark:bg-gray-700/50 text-gray-600 dark:text-gray-300 text-xs font-bold uppercase tracking-wider border-b border-gray-200 dark:border-gray-700">
                  <th className="px-6 py-4 w-12 text-center">
                      <input 
                          type="checkbox" 
                          className="rounded border-gray-300 text-indigo-600 focus:ring-indigo-500 w-4 h-4 bg-white dark:bg-gray-900 dark:border-gray-600 transition-colors cursor-pointer" 
                          checked={paginatedPosts.length > 0 && selectedIds.length === paginatedPosts.length} 
                          onChange={handleSelectAll} 
                      />
                  </th>
                  <th className="px-6 py-4">Title</th>
                  <th className="px-6 py-4">Author</th>
                  <th className="px-6 py-4">Category</th>
                  <th className="px-6 py-4">Redirection</th>
                  <th className="px-6 py-4">Modified</th>
                  <th className="px-6 py-4">Status</th>
                  <th className="px-6 py-4 text-right">Actions</th>
                </tr>
              </thead>
              <tbody className="divide-y divide-gray-200 dark:divide-gray-700 text-sm text-gray-900 dark:text-gray-100">
                {paginatedPosts.map((post) => {
                  const isSelected = selectedIds.includes(post.id);
                  return (
                  <tr key={post.id} className={`transition-colors ${isSelected ? 'bg-indigo-50/50 dark:bg-indigo-900/20' : 'hover:bg-gray-50 dark:hover:bg-gray-700/30'}`}>
                    <td className="px-6 py-4 text-center">
                        <input 
                            type="checkbox" 
                            className="rounded border-gray-300 text-indigo-600 focus:ring-indigo-500 w-4 h-4 bg-white dark:bg-gray-900 dark:border-gray-600 transition-colors cursor-pointer" 
                            checked={isSelected} 
                            onChange={() => handleSelectRow(post.id)} 
                        />
                    </td>
                    <td className="px-6 py-4 font-semibold text-gray-900 dark:text-white whitespace-normal">
                      {post.title}
                    </td>
                    <td className="px-6 py-4">
                      <div className="flex items-center gap-3">
                        <div className="w-7 h-7 rounded-full bg-gradient-to-br from-slate-100 to-gray-200 dark:from-slate-800 dark:to-gray-700 flex items-center justify-center text-xs font-bold text-slate-700 dark:text-slate-300 shadow-sm border border-slate-200/50 dark:border-slate-600/50">
                          {post.author_name ? post.author_name.charAt(0).toUpperCase() : "?"}
                        </div>
                        <span className="text-[13px] font-semibold text-gray-900 dark:text-gray-100">
                          {post.author_name || "System"}
                        </span>
                      </div>
                    </td>
                    <td className="px-6 py-4">
                      {post.category_name ? (
                        <div className="flex items-center gap-2">
                           <div className="w-1.5 h-1.5 rounded-full bg-gray-300 dark:bg-gray-600"></div>
                           <span className="text-[13px] font-medium text-gray-800 dark:text-gray-200">
                             {post.category_name}
                           </span>
                        </div>
                      ) : (
                        <span className="text-gray-400 dark:text-gray-500 text-[13px]">-</span>
                      )}
                    </td>
                    <td className="px-6 py-4">
                      {post.active_redirect ? (
                        <div className="flex flex-col items-start max-w-[150px]">
                          <span className="inline-flex items-center gap-1 px-2 py-0.5 rounded text-[10px] font-bold bg-indigo-50 text-indigo-700 dark:bg-indigo-900/30 dark:text-indigo-300 border border-indigo-100 dark:border-indigo-800 uppercase tracking-wide shadow-sm">
                            <SafeIcon icon={FiCornerUpRight} className="w-3 h-3" /> Active
                          </span>
                        </div>
                      ) : (
                        <span className="inline-flex px-2 py-0.5 rounded text-[10px] font-medium bg-gray-100 text-gray-500 dark:bg-gray-800 dark:text-gray-400 border border-gray-200 dark:border-gray-700 uppercase tracking-wide">
                          Not Set
                        </span>
                      )}
                    </td>
                    <td className="px-6 py-4 text-xs text-gray-500 dark:text-gray-400 font-medium">
                      {new Date(post.updated_at).toLocaleDateString()}
                    </td>
                    <td className="px-6 py-4">
                       <span className={`inline-flex items-center px-2.5 py-1 rounded-full text-[11px] font-bold capitalize shadow-sm border ${
                            post.status === "published"
                              ? "bg-green-50 text-green-700 border-green-200 dark:bg-green-900/20 dark:text-green-400 dark:border-green-800"
                              : post.status === "trash"
                                ? "bg-red-50 text-red-700 border-red-200 dark:bg-red-900/20 dark:text-red-400 dark:border-red-800"
                                : "bg-amber-50 text-amber-700 border-amber-200 dark:bg-amber-900/20 dark:text-amber-400 dark:border-amber-800"
                          }`}
                        >
                          {post.status === "published" && <FiCheckCircle className="w-3 h-3 mr-1" />}
                          {post.status === "draft" && <FiFileText className="w-3 h-3 mr-1" />}
                          {post.status === "trash" && <FiTrash2 className="w-3 h-3 mr-1" />}
                          {post.status}
                        </span>
                    </td>
                    <td className="px-6 py-4 text-right">
                      <div className="flex items-center justify-end gap-1">
                        {filterStatus !== "trash" && (
                           <button 
                              onClick={() => handleEdit(post)} 
                              className="p-2 text-indigo-600 hover:bg-indigo-50 dark:hover:bg-indigo-900/30 dark:text-indigo-400 rounded-lg transition-colors focus:outline-none" 
                              title="Edit Post"
                           >
                              <SafeIcon icon={FiEdit} className="w-4 h-4" />
                           </button>
                        )}
                        {filterStatus === "published" && (
                           <button 
                              onClick={() => handleView(post)} 
                              className="p-2 text-gray-400 hover:text-indigo-600 hover:bg-indigo-50 dark:hover:bg-indigo-900/30 dark:hover:text-indigo-400 rounded-lg transition-colors focus:outline-none" 
                              title="View Post"
                           >
                              <SafeIcon icon={FiEye} className="w-4 h-4" />
                           </button>
                        )}
                        {filterStatus === "trash" && (
                           <button 
                              onClick={() => handleRestore(post)} 
                              className="p-2 text-blue-600 hover:bg-blue-50 dark:hover:bg-blue-900/30 dark:text-blue-400 rounded-lg transition-colors focus:outline-none" 
                              title="Restore Post"
                           >
                              <SafeIcon icon={FiRefreshCw} className="w-4 h-4" />
                           </button>
                        )}
                        <button 
                            onClick={() => filterStatus === "trash" ? initiateDeletePermanent(post) : initiateTrash(post)} 
                            className="p-2 text-gray-400 hover:text-red-600 hover:bg-red-50 dark:hover:bg-red-900/30 rounded-lg transition-colors focus:outline-none" 
                            title={filterStatus === "trash" ? "Delete Forever" : "Move to Trash"}
                        >
                            <SafeIcon icon={filterStatus === "trash" ? FiTrash2 : HiOutlineTrash} className="w-4 h-4" />
                        </button>
                      </div>
                    </td>
                  </tr>
                )})}
              </tbody>
            </table>
          )}

          {/* CUSTOM PAGINATION ENGINE */}
          {((totalRecords > 0) || (totalPages > 1)) && (
              <div className="px-6 py-4 border-t border-gray-200 dark:border-gray-700 bg-gray-50/50 dark:bg-gray-800/50 flex flex-col sm:flex-row items-center justify-between gap-4 transition-all mt-auto rounded-b-xl">
                  
                  <div className="flex items-center text-sm text-gray-500 dark:text-gray-400">
                      Showing
                      <span className="mx-1.5 px-2 py-0.5 rounded-md bg-white dark:bg-gray-700 border border-gray-200 dark:border-gray-600 font-semibold text-gray-900 dark:text-white shadow-sm">
                          {totalRecords 
                              ? Math.min(((currentPage - 1) * displayLimit) + 1, totalRecords) 
                              : currentPage}
                      </span>
                      {totalRecords ? (
                          <>
                              to
                              <span className="mx-1.5 px-2 py-0.5 rounded-md bg-white dark:bg-gray-700 border border-gray-200 dark:border-gray-600 font-semibold text-gray-900 dark:text-white shadow-sm">
                                  {Math.min((currentPage * displayLimit), totalRecords)}
                              </span>
                              of
                              <span className="mx-1.5 px-2 py-0.5 rounded-md bg-white dark:bg-gray-700 border border-gray-200 dark:border-gray-600 font-semibold text-gray-900 dark:text-white shadow-sm">
                                  {totalRecords}
                              </span>
                              entries
                          </>
                      ) : <span className="text-gray-900 dark:text-gray-100 font-medium ml-1">Page View</span>}
                  </div>

                  {(totalPages > 1) && (
                      <div className="flex items-center p-1 bg-gray-100/80 dark:bg-gray-900/50 rounded-lg border border-gray-200 dark:border-gray-700/50 shadow-inner">
                          <button
                              onClick={() => setCurrentPage(Math.max(1, currentPage - 1))}
                              disabled={currentPage === 1}
                              className="flex items-center gap-1 px-3 py-1.5 text-sm font-medium rounded-md text-gray-600 dark:text-gray-400 hover:bg-white dark:hover:bg-gray-800 hover:text-indigo-600 dark:hover:text-indigo-400 hover:shadow-sm disabled:opacity-40 disabled:hover:bg-transparent disabled:hover:shadow-none disabled:hover:text-gray-600 transition-all focus:outline-none"
                          >
                              <SafeIcon icon={FiChevronLeft} className="w-4 h-4" /> Prev
                          </button>

                          <div className="flex items-center px-4 border-x border-gray-200 dark:border-gray-700/50">
                              <span className="text-sm font-bold text-gray-900 dark:text-white">
                                  {currentPage}
                              </span>
                              <span className="text-sm text-gray-400 dark:text-gray-500 mx-1.5">/</span>
                              <span className="text-sm font-medium text-gray-500 dark:text-gray-400">
                                  {totalPages}
                              </span>
                          </div>

                          <button
                              onClick={() => setCurrentPage(Math.min(totalPages, currentPage + 1))}
                              disabled={currentPage === totalPages}
                              className="flex items-center gap-1 px-3 py-1.5 text-sm font-medium rounded-md text-gray-600 dark:text-gray-400 hover:bg-white dark:hover:bg-gray-800 hover:text-indigo-600 dark:hover:text-indigo-400 hover:shadow-sm disabled:opacity-40 disabled:hover:bg-transparent disabled:hover:shadow-none disabled:hover:text-gray-600 transition-all focus:outline-none"
                          >
                              Next <SafeIcon icon={FiChevronRight} className="w-4 h-4" />
                          </button>
                      </div>
                  )}
              </div>
          )}
        </div>
      </motion.div>
    </div>
  );
};

export default Posts;
