// --- UX FIX: Custom Select for Dark Mode Consistency ---
const CustomSelect = ({ value, onChange, options, className, buttonClassName, dropdownClassName, disabled, icon }) => {
  const [isOpen, setIsOpen] = useState(false);
  const dropdownRef = useRef(null);

  useEffect(() => {
    const handleClickOutside = (event) => {
      if (dropdownRef.current && !dropdownRef.current.contains(event.target)) {
          setIsOpen(false);
      }
    };
    
    document.addEventListener('mousedown', handleClickOutside);
    return () => document.removeEventListener('mousedown', handleClickOutside);
  }, []);

  const selectedLabel = options.find(opt => String(opt.value) === String(value))?.label || '';

  return (
    <div className={`relative ${className || ''}`} ref={dropdownRef}>
      <button
        type="button"
        disabled={disabled}
        onClick={() => !disabled && setIsOpen(!isOpen)}
        className={buttonClassName || `w-full flex items-center justify-between px-3 py-2 bg-white dark:bg-gray-800 border border-gray-300 dark:border-gray-600 rounded-lg text-sm focus:ring-2 focus:ring-indigo-500 outline-none text-gray-900 dark:text-white transition-shadow cursor-pointer shadow-sm font-medium ${disabled ? 'opacity-50 cursor-not-allowed' : ''}`}
      >
        <div className="flex items-center gap-2 truncate">
            {icon && <SafeIcon icon={icon} className="w-4 h-4 text-gray-400 shrink-0" />}
            <span className="truncate">{selectedLabel}</span>
        </div>
        <SafeIcon icon={FiChevronDown} className={`w-4 h-4 text-gray-400 ml-2 shrink-0 transition-transform duration-200 ${isOpen ? 'rotate-180' : ''}`} />
      </button>

      <AnimatePresence>
        {isOpen && !disabled && (
          <motion.div
            initial={{ opacity: 0, y: -5 }} 
            animate={{ opacity: 1, y: 0 }} 
            exit={{ opacity: 0, y: -5 }} 
            transition={{ duration: 0.15 }}
            className={`absolute left-0 top-full mt-1 w-full min-w-[160px] z-[100] bg-white dark:bg-gray-800 border border-gray-200 dark:border-gray-700 rounded-lg shadow-xl overflow-hidden py-1 ${dropdownClassName || ''}`}
          >
            <div className="max-h-60 overflow-y-auto custom-scrollbar">
                {options.map((opt) => (
                  <button
                    key={opt.value}
                    type="button"
                    onClick={(e) => { 
                      e.stopPropagation(); 
                      onChange(opt.value); 
                      setIsOpen(false); 
                    }}
                    className={`w-full text-left px-4 py-2.5 text-sm transition-colors ${
                      String(value) === String(opt.value) 
                        ? 'bg-indigo-50 dark:bg-indigo-900/30 text-indigo-700 dark:text-indigo-400 font-bold' 
                        : 'text-gray-900 dark:text-gray-300 hover:bg-gray-100 dark:hover:bg-gray-700 font-medium'
                    }`}
                  >
                    {opt.label}
                  </button>
                ))}
            </div>
          </motion.div>
        )}
      </AnimatePresence>
    </div>
  );
};

const SeoRedirects = () => {
  const location = useLocation();
  const { getCache, setCacheData } = useAdminCache();

  // Hook to access state passed from 404 Monitor
  const [redirects, setRedirects] = useState([]);
  const [counts, setCounts] = useState({
    all: 0,
    active: 0,
    inactive: 0,
    trash: 0,
  });

  const [initialLoading, setInitialLoading] = useState(true);
  const [isUpdating, setIsUpdating] = useState(false);
  const [isSubmitting, setIsSubmitting] = useState(false);

  // Filter and Pagination State
  const [filterStatus, setFilterStatus] = useState("all");
  const [actionFilter, setActionFilter] = useState("all");
  const [selectedIds, setSelectedIds] = useState([]);
  const [currentPage, setCurrentPage] = useState(1);
  const [itemsPerPage, setItemsPerPage] = useState(
    Number(localStorage.getItem("redirects_limit")) || 20
  );

  // Form State
  const [sourceInput, setSourceInput] = useState("");
  // UX IMPROVEMENT: Set default to Internal Path (empty string)
  const [targetProtocol, setTargetProtocol] = useState("");
  const [targetUrl, setTargetUrl] = useState("");
  const [statusCode, setStatusCode] = useState(""); // Default to empty
  
  // Modals State
  const [isEditModalOpen, setIsEditModalOpen] = useState(false);
  const [editingRedirect, setEditingRedirect] = useState(null);

  const [isConfirmOpen, setIsConfirmOpen] = useState(false);
  const [confirmAction, setConfirmAction] = useState(null);

  // --- PHASE 2 ENTERPRISE SAFEGUARD STATE ---
  const [isLiveWarningOpen, setIsLiveWarningOpen] = useState(false);
  const [pendingSubmission, setPendingSubmission] = useState(null);
  
  const [isHelpOpen, setIsHelpOpen] = useState(false);
  const [isSettingsOpen, setIsSettingsOpen] = useState(false);
  
  const [baseDomain, setBaseDomain] = useState("https://calcpro.com/");

  // --- BRIDGED TERMINOLOGY FOR SEO PROS ---
  const STATUS_OPTIONS = [
    { value: "301", label: "301 / 308 Permanent Move (Modern)" },
    { value: "302", label: "302 / 307 Temporary Move (Secure)" },
    { value: "410", label: "410 Content Deleted (SEO Purge)" },
    { value: "451", label: "451 Unavailable (Legal/DMCA)" },
  ];

  const TARGET_PROTOCOL_OPTIONS = [
    { value: "https://", label: "https://" },
    { value: "http://", label: "http://" },
    { value: "", label: "Internal Path (/)" },
  ];

  // --- Fetch Base Domain for UI Prefix ---
  useEffect(() => {
    const loadDomain = async () => {
      const cachedSettings = getCache("settings_general");
      if (cachedSettings && cachedSettings.site_url) {
        const url = cachedSettings.site_url;
        setBaseDomain(url.endsWith("/") ? url : `${url}/`);
        return;
      }
      try {
        const result = await authenticatedFetch(
          "/api/admin/get_settings.php"
        );
        if (result.success && result.data) {
          setCacheData("settings_general", result.data);
          const url = result.data.site_url || "https://calcpro.com";
          setBaseDomain(url.endsWith("/") ? url : `${url}/`);
        }
      } catch (error) {
        console.error("Failed to fetch base domain for redirect manager.");
      }
    };
    loadDomain();
  }, [getCache, setCacheData]);

  // --- 404 Monitor Integration: Auto-fill from state ---
  useEffect(() => {
    if (location.state && location.state.sourcePath) {
      const cleanPath = location.state.sourcePath.replace(/^\/+/, "");
      setSourceInput(cleanPath);
      window.history.replaceState({}, document.title);
      toast.success("Imported URL from 404 Monitor", { id: 'import-toast' });
    }
  }, [location]);

  // Reset pagination when filters change
  useEffect(() => {
    setCurrentPage(1);
  }, [filterStatus, actionFilter]);

  const fetchRedirects = useCallback(async () => {
    try {
      const result = await authenticatedFetch(
        `/api/admin/get_redirects.php?status=${filterStatus}&page=${currentPage}&limit=${itemsPerPage}&action_type=${actionFilter}`
      );
      if (result.success) {
        setRedirects(result.data.redirects);
        setCounts(result.data.counts);
      }
    } catch (error) {
      toast.error(error.message || "Failed to load redirects.");
    }
  }, [filterStatus, currentPage, itemsPerPage, actionFilter]);

  useEffect(() => {
    const loadData = async () => {
      if (initialLoading) {
        await fetchRedirects();
        setInitialLoading(false);
      } else {
        setIsUpdating(true);
        await Promise.all([fetchRedirects(), uxDelay(300)]); // Small delay for filter switching
        setIsUpdating(false);
      }
    };
    loadData();
    setSelectedIds([]); // Reset selection on filter change
  }, [filterStatus, fetchRedirects, currentPage, itemsPerPage, actionFilter]);

  const handleManualRefresh = async () => {
    setIsUpdating(true);
    await Promise.all([fetchRedirects(), uxDelay(600)]);
    setIsUpdating(false);
  };

  const handleSaveSettings = (newLimit) => {
    setItemsPerPage(newLimit);
    localStorage.setItem("redirects_limit", newLimit);
    setCurrentPage(1);
    setIsSettingsOpen(false);
  };

  const handleAdd = async (e) => {
    e.preventDefault();
    if (isSubmitting) return;

    const toastId = "redirect-add-error";

    if (!sourceInput.trim()) {
      toast.error("Please enter the original source path.", { id: toastId });
      return;
    }

    if (!statusCode) {
      toast.error("Please explicitly select an Action Type.", { id: toastId });
      return;
    }

    const finalSourcePath = sourceInput.trim().replace(/^\/+/, "");

    const isDuplicate = redirects.some(
      (rule) => rule.source_path.toLowerCase() === finalSourcePath.toLowerCase()
    );

    if (isDuplicate) {
      toast.error("This URL is already redirected. Please modify the existing rule below.", { id: toastId });
      return;
    }

    const isTargetRequired = !statusCode.startsWith("4");
    let finalTargetUrl = "";

    if (isTargetRequired) {
      if (!targetUrl.trim()) {
        toast.error("A destination URL is required for this action type.", { id: toastId });
        return;
      }
      
      let cleanedInput = targetUrl.trim().replace(/^https?:\/\//i, '');

      if (targetProtocol === "" && !cleanedInput.startsWith("/")) {
        cleanedInput = "/" + cleanedInput;
      }
      
      finalTargetUrl = targetProtocol + cleanedInput;
    }

    setIsSubmitting(true);
    toast.dismiss(toastId);

    try {
      const payload = {
        source_path: finalSourcePath,
        target_url: finalTargetUrl,
        status_code: statusCode,
      };

      const [result] = await Promise.all([
        authenticatedFetch(
          "/api/admin/add_redirect.php",
          {
            method: "POST",
            body: JSON.stringify(payload),
          }
        ),
        uxDelay(600)
      ]);

      if (result.success) {
        toast.success("Redirect rule created.");
        setSourceInput("");
        setTargetUrl("");
        setTargetProtocol("");
        setStatusCode(""); 
        fetchRedirects();
      } else {
        toast.error(result.message);
      }
    } catch (error) {
      // --- PHASE 2 ENTERPRISE SAFEGUARD: Catch Live URL Block ---
      if (error.errorCode === 'URL_CURRENTLY_LIVE') {
        setPendingSubmission({ type: 'add', payload: {
          source_path: finalSourcePath,
          target_url: finalTargetUrl,
          status_code: statusCode,
        } });
        setIsLiveWarningOpen(true);
      } else {
        toast.error(error.message || "Connection error.");
      }
    } finally {
      setIsSubmitting(false);
    }
  };

  const openEditModal = (rule) => {
    let p; // FIX: Declared without useless assignment
    let u = rule.target_url || "";

    if (u.startsWith("https://")) {
      p = "https://";
      u = u.replace("https://", "");
    } else if (u.startsWith("http://")) {
      p = "http://";
      u = u.replace("http://", "");
    } else {
      p = "";
    }

    setEditingRedirect({
      ...rule,
      source_path: rule.source_path,
      target_protocol: p,
      target_url_clean: u,
      status_code: rule.status_code.toString(),
    });
    setIsEditModalOpen(true);
  };

  const handleUpdate = async () => {
    if (!editingRedirect || isSubmitting) return;

    const isTargetRequired = !editingRedirect.status_code.startsWith("4");
    let finalTarget = "";

    if (isTargetRequired) {
      if (!editingRedirect.target_url_clean.trim()) {
         toast.error("Destination URL is required.");
         return;
      }
      let cleaned = editingRedirect.target_url_clean.trim().replace(/^https?:\/\//i, '');

      if (editingRedirect.target_protocol === "" && !cleaned.startsWith("/")) {
        cleaned = "/" + cleaned;
      }
      finalTarget = editingRedirect.target_protocol + cleaned;
    }

    setIsSubmitting(true);

    try {
      const payload = {
        ...editingRedirect,
        target_url: finalTarget
      };

      const [result] = await Promise.all([
        authenticatedFetch(
          "/api/admin/update_redirect.php",
          {
            method: "POST",
            body: JSON.stringify(payload),
          }
        ),
        uxDelay(600)
      ]);

      if (result.success) {
        toast.success("Rule updated successfully.");
        setIsEditModalOpen(false);
        await fetchRedirects();
      } else {
        toast.error(result.message);
      }
    } catch (error) {
      // --- PHASE 2 ENTERPRISE SAFEGUARD: Catch Live URL Block ---
      if (error.errorCode === 'URL_CURRENTLY_LIVE') {
        setPendingSubmission({ type: 'update', payload: {
          ...editingRedirect,
          target_url: finalTarget
        } });
        setIsLiveWarningOpen(true);
      } else {
        toast.error(error.message || "Connection error.");
      }
    } finally {
      setIsSubmitting(false);
    }
  };

  // --- PHASE 2 ENTERPRISE SAFEGUARD: Override Handler ---
  const handleForceOverride = async () => {
    if (!pendingSubmission) return;
    setIsLiveWarningOpen(false);
    setIsSubmitting(true);
    const { type, payload } = pendingSubmission;
    const endpoint = type === 'add' ? '/api/admin/add_redirect.php' : '/api/admin/update_redirect.php';
    
    try {
      const [result] = await Promise.all([
        authenticatedFetch(endpoint, {
          method: "POST",
          body: JSON.stringify({ ...payload, force_override: true })
        }),
        uxDelay(600)
      ]);

      if (result.success) {
        toast.success(type === 'add' ? "Redirect rule created." : "Rule updated successfully.");
        if (type === 'add') {
          setSourceInput("");
          setTargetUrl("");
          setTargetProtocol("");
          setStatusCode("");
        } else {
          setIsEditModalOpen(false);
        }
        await fetchRedirects();
      } else {
        toast.error(result.message);
      }
    } catch (error) {
      toast.error(error.message || "Connection error.");
    } finally {
      setIsSubmitting(false);
      setPendingSubmission(null);
    }
  };

  const handleToggleStatus = async (rule) => {
    setIsUpdating(true);
    const newStatus = rule.status === "active" ? "inactive" : "active";
    
    // If we are viewing a specific tab (Active/Inactive), toggling the status
    // will cause the item to disappear from the current view.
    const willEmptyPage = filterStatus !== "all" && redirects.length === 1 && currentPage > 1;

    try {
      await Promise.all([
        authenticatedFetch(
          "/api/admin/update_redirect.php",
          {
            method: "POST",
            body: JSON.stringify({ id: rule.id, status: newStatus }),
          }
        ),
        uxDelay(600)
      ]);
      
      if (willEmptyPage) {
        setCurrentPage((prev) => prev - 1);
      } else {
        await fetchRedirects();
      }
    } catch (error) {
      toast.error(error.message || "Failed to update status.");
    } finally {
      setIsUpdating(false);
    }
  };

  const openConfirmModal = (action) => {
    setConfirmAction(action);
    setIsConfirmOpen(true);
  };

  const getModalContent = () => {
    if (!confirmAction)
      return {
        title: "Confirm",
        message: "Are you sure?",
        button: "Confirm",
        color: "bg-blue-600 hover:bg-blue-700",
      };

    const type = confirmAction.type;
    const count = selectedIds.length;

    switch (type) {
      case "delete":
        return filterStatus === "trash"
          ? {
              title: "Delete Permanently?",
              message:
                "This rule will be deleted forever. This cannot be undone.",
              button: "Delete Forever",
              color: "bg-red-600 hover:bg-red-700",
            }
          : {
              title: "Move to Trash?",
              message:
                "The redirect will stop working. You can restore it later from the Trash tab.",
              button: "Move to Trash",
              color: "bg-red-600 hover:bg-red-700",
            };
      case "restore":
        return {
          title: "Restore Rule?",
          message: "This rule will be restored to Active status.",
          button: "Restore",
          color: "bg-blue-600 hover:bg-blue-700",
        };
      case "bulk_trash":
        return {
          title: `Trash ${count} Rules?`,
          message: `Move ${count} selected rules to the Trash?`,
          button: "Trash All",
          color: "bg-red-600 hover:bg-red-700",
        };
      case "bulk_delete_permanent":
        return {
          title: `Delete ${count} Rules?`,
          message: `Permanently delete ${count} rules?\nThis cannot be undone.`,
          button: "Delete Forever",
          color: "bg-red-600 hover:bg-red-700",
        };
      case "bulk_restore":
        return {
          title: `Restore ${count} Rules?`,
          message: `Restore ${count} rules to Active status?`,
          button: "Restore All",
          color: "bg-blue-600 hover:bg-blue-700",
        };
      case "bulk_activate":
        return {
          title: "Activate Selected?",
          message: `Set ${count} rules to Active status?`,
          button: "Activate",
          color: "bg-green-600 hover:bg-green-700",
        };
      case "bulk_deactivate":
        return {
          title: "Deactivate Selected?",
          message: `Set ${count} rules to Inactive status?`,
          button: "Deactivate",
          color: "bg-amber-600 hover:bg-amber-700",
        };
      default:
        return {
          title: "Confirm Action",
          message: "Are you sure?",
          button: "Confirm",
          color: "bg-blue-600 hover:bg-blue-700",
        };
    }
  };

  const modalContent = getModalContent();

  const handleConfirm = () => {
    if (!confirmAction) return;
    const { type, rule } = confirmAction;

    if (type.startsWith("bulk_")) {
      handleBulkAction(type.replace("bulk_", ""));
    } else if (type === "restore") {
      handleRestore(rule);
    } else if (type === "delete") {
      handleDelete(rule);
    }
    setIsConfirmOpen(false);
    setConfirmAction(null);
  };

  const handleRestore = async (rule) => {
    // Restoring an item moves it out of the Trash view
    const willEmptyPage = redirects.length === 1 && currentPage > 1;
    setIsUpdating(true); // Manually trigger full-page overlay

    try {
      const [result] = await Promise.all([
        authenticatedFetch(
          "/api/admin/restore_redirect.php",
          {
            method: "POST",
            body: JSON.stringify({ id: rule.id }),
          }
        ),
        uxDelay(600)
      ]);

      if (result.success) {
        toast.success("Rule restored.");
        if (willEmptyPage) {
           setCurrentPage((prev) => prev - 1);
        } else {
           fetchRedirects();
        }
      } else {
        toast.error(result.message);
      }
    } catch (error) {
      toast.error(error.message || "Restore failed.");
    } finally {
      setIsUpdating(false);
    }
  };

  const handleDeleteClick = (rule) => {
    openConfirmModal({ type: "delete", rule });
  };

  const handleDelete = async (rule) => {
    const isPermanent = filterStatus === "trash";
    const url = isPermanent ? "delete_permanent_redirect.php" : "delete_redirect.php";
    const successMsg = isPermanent ? "Rule permanently deleted." : "Rule moved to Trash.";

    // Moving to trash OR permanent deletion removes the item from the current view
    const willEmptyPage = redirects.length === 1 && currentPage > 1;
    setIsUpdating(true); // Manually trigger full-page overlay

    try {
      const [result] = await Promise.all([
        authenticatedFetch(
          `/api/admin/${url}`,
          {
            method: "POST",
            body: JSON.stringify({ id: rule.id }),
          }
        ),
        uxDelay(600)
      ]);

      if (result.success) {
        toast.success(successMsg);
        if (willEmptyPage) {
           setCurrentPage((prev) => prev - 1);
        } else {
           fetchRedirects();
        }
      } else {
        toast.error(result.message);
      }
    } catch (error) {
      toast.error(error.message || "Failed to delete rule.");
    } finally {
      setIsUpdating(false);
    }
  };

  const handleBulkAction = async (action) => {
    if (selectedIds.length === 0) return;
    setIsUpdating(true);

    // Calculate if this specific bulk action removes the selected items from the CURRENT view
    const isRemoving = 
        ["bulk_trash", "bulk_delete_permanent", "bulk_restore"].includes(action) ||
        (action === "bulk_activate" && filterStatus === "inactive") ||
        (action === "bulk_deactivate" && filterStatus === "active");

    const willEmptyPage = isRemoving && selectedIds.length === redirects.length && currentPage > 1;

    try {
      const [result] = await Promise.all([
        authenticatedFetch(
          "/api/admin/manage_redirects_bulk.php",
          {
            method: "POST",
            body: JSON.stringify({ ids: selectedIds, action: action }),
          }
        ),
        uxDelay(600)
      ]);

      if (result.success) {
        toast.success(result.message);
        setSelectedIds([]);
        if (willEmptyPage) {
           setCurrentPage((prev) => prev - 1);
        } else {
           await fetchRedirects();
        }
      } else {
        toast.error(result.message);
      }
    } catch (error) {
      toast.error(error.message || "Bulk action failed.");
    } finally {
      setIsUpdating(false);
    }
  };

  const handleSelectAll = (e) => {
    if (e.target.checked) {
      setSelectedIds(redirects.map((r) => r.id));
    } else {
      setSelectedIds([]);
    }
  };

  const handleSelectRow = (id) => {
    setSelectedIds((prev) =>
      prev.includes(id) ? prev.filter((i) => i !== id) : [...prev, id]
    );
  };

  const StatusBadge = ({ code }) => {
    const codeStr = String(code);
    const configs = {
      301: {
        bg: "bg-green-100 dark:bg-green-900/40",
        text: "text-green-700 dark:text-green-400",
        border: "border-green-200 dark:border-green-800",
        label: "301 / 308 Permanent",
      },
      302: {
        bg: "bg-amber-100 dark:bg-amber-900/40",
        text: "text-amber-700 dark:text-amber-400",
        border: "border-amber-200 dark:border-amber-800",
        label: "302 / 307 Temporary",
      },
      307: {
        bg: "bg-blue-100 dark:bg-blue-900/40",
        text: "text-blue-700 dark:text-blue-400",
        border: "border-blue-200 dark:border-blue-800",
        label: "307 Temporary",
      },
      410: {
        bg: "bg-red-100 dark:bg-red-900/40",
        text: "text-red-700 dark:text-red-400",
        border: "border-red-200 dark:border-red-800",
        label: "410 Gone",
      },
      451: {
        bg: "bg-gray-100 dark:bg-gray-700",
        text: "text-gray-700 dark:text-gray-300",
        border: "border-gray-200 dark:border-gray-600",
        label: "451 Legal",
      },
    };

    const c = configs[codeStr] || configs["301"];

    return (
      <span
        className={`inline-flex items-center px-2.5 py-0.5 rounded-md text-xs font-bold border ${c.bg} ${c.text} ${c.border}`}
      >
        {c.label}
      </span>
    );

                    </th>
                    <th className="px-2 py-4 w-8"></th>
                    <th className="px-6 py-4 text-xs font-bold text-gray-500 dark:text-gray-400 uppercase tracking-wider">
                      Destination
                    </th>
                    {filterStatus !== "trash" && (
                      <th className="px-6 py-4 text-xs font-bold text-gray-500 dark:text-gray-400 uppercase tracking-wider text-center w-24">
                        State
                      </th>
                    )}
                    <th className="px-6 py-4 text-xs font-bold text-gray-500 dark:text-gray-400 uppercase tracking-wider text-right w-32">
                      Actions
                    </th>
                  </tr>
                </thead>
                <tbody className="divide-y divide-gray-100 dark:divide-gray-700/50">
                  {redirects.length === 0 ? (
                    <tr>
                      <td colSpan="7" className="px-6 py-16 text-center">
                        <div className="flex flex-col items-center opacity-50">
                          <SafeIcon
                            icon={FiActivity}
                            className="w-12 h-12 text-gray-300 mb-3"
                          />
                          <p className="text-gray-500 font-medium">
                            No rules in this view.
                          </p>
                        </div>
                      </td>
                    </tr>
                  ) : (
                    redirects.map((rule) => (
                      <tr
                        key={rule.id}
                        className={`group transition-colors ${selectedIds.includes(rule.id) ? "bg-indigo-50 dark:bg-indigo-900/10" : "hover:bg-gray-50 dark:hover:bg-gray-700/20"}`}
                      >
                        <td className="px-4 py-3 align-middle">
                          <input
                            type="checkbox"
                            checked={selectedIds.includes(rule.id)}
                            onChange={() => handleSelectRow(rule.id)}
                            className="rounded border-gray-300 text-indigo-600 focus:ring-indigo-500 cursor-pointer"
                          />
                        </td>
                         <td className="px-6 py-3 whitespace-nowrap align-middle">
                          <StatusBadge code={rule.status_code} />
                        </td>
                        <td className="px-6 py-3 align-middle max-w-[260px]">
                          <div
                            className="font-mono text-sm text-gray-700 dark:text-gray-300 truncate"
                            title={rule.source_path}
                           >
                            <span className="text-gray-400 select-none">/</span>
                            {rule.source_path}
                            </div>
                        </td>
                        <td className="px-2 py-3 text-center align-middle">
                          <SafeIcon
                            icon={FiArrowRight}
                            className="w-4 h-4 text-gray-300 dark:text-gray-600"
                          />
                        </td>
                        <td className="px-6 py-3 align-middle max-w-[300px]">
                          {String(rule.status_code).startsWith("4") ? (
                            <span className="text-sm text-red-500/80 font-medium italic flex items-center gap-1.5">
                              <SafeIcon
                                icon={FiAlertCircle}
                                className="w-3.5 h-3.5"
                              />{" "}
                              No Destination
                              </span>
                          ) : (
                            <a
                              href={rule.target_url.startsWith("/") ? `${baseDomain.replace(/\/$/, "")}${rule.target_url}` : rule.target_url}
                              target="_blank"
                              rel="noopener noreferrer"
                              className="text-sm text-blue-600 dark:text-blue-400 hover:underline font-medium flex items-center gap-1.5 group/link truncate"
                              title={rule.target_url.startsWith("/") ? `${baseDomain.replace(/\/$/, "")}${rule.target_url}` : rule.target_url}
                            >
                              <SafeIcon
                                icon={FiGlobe}
                                className="w-3.5 h-3.5 opacity-70 group-hover/link:opacity-100"
                              />
                              <span className="truncate">
                                {rule.target_url}
                              </span>
                            </a>
                          )}
                        </td>

                        {filterStatus !== "trash" && (
                            <td className="px-6 py-3 whitespace-nowrap text-center align-middle">
                            <ToggleSwitch
                              checked={rule.status === "active"}
                              onChange={() => handleToggleStatus(rule)}
                            />
                          </td>
                        )}

                        <td className="px-6 py-3 whitespace-nowrap text-right align-middle">
                          <div className="flex items-center justify-end gap-2">
                            {filterStatus === "trash" ? (
                              <button
                                onClick={() =>
                                  openConfirmModal({ type: "restore", rule })
                                }
                                className="p-2 text-gray-400 hover:text-blue-600 rounded-lg"
                                title="Restore Rule"
                              >
                                <SafeIcon
                                  icon={FiRefreshCw}
                                  className="w-4 h-4"
                                />
                              </button>
                              ) : (
                              <button
                                onClick={() => openEditModal(rule)}
                                className="p-2 text-gray-400 hover:text-blue-600 rounded-lg"
                                title="Edit Rule"
                              >
                                  <SafeIcon icon={FiEdit2} className="w-4 h-4" />
                              </button>
                            )}
                            <button
                              onClick={() => handleDeleteClick(rule)}
                              className="p-2 text-gray-400 hover:text-red-600 rounded-lg"
                              title={
                                filterStatus === "trash"
                                  ? "Delete Permanently"
                                  : "Move to Trash"
                              }
                            >
                               <SafeIcon icon={FiTrash2} className="w-4 h-4" />
                            </button>
                          </div>
                        </td>
                      </tr>
                    ))
                  )}
                </tbody>
              </table>
             </div>

            {/* --- PREMIUM PAGINATION FOOTER --- */}
            {currentFilterCount > 0 && (
              <div className="px-6 py-4 border-t border-gray-200 dark:border-gray-700 bg-gray-50/50 dark:bg-gray-800/50 flex flex-col sm:flex-row items-center justify-between gap-4 transition-all mt-auto rounded-b-xl">
                
                {/* Metric Badging */}
                <div className="flex items-center text-sm text-gray-500 dark:text-gray-400">
                  Showing
                  <span className="mx-1.5 px-2 py-0.5 rounded-md bg-white dark:bg-gray-700 border border-gray-200 dark:border-gray-600 font-semibold text-gray-900 dark:text-white shadow-sm">
                   {Math.min((currentPage - 1) * itemsPerPage + 1, currentFilterCount)}
                  </span>
                  to
                  <span className="mx-1.5 px-2 py-0.5 rounded-md bg-white dark:bg-gray-700 border border-gray-200 dark:border-gray-600 font-semibold text-gray-900 dark:text-white shadow-sm">
                    {Math.min(currentPage * itemsPerPage, currentFilterCount)}
                  </span>
                  of
                  <span className="mx-1.5 px-2 py-0.5 rounded-md bg-white dark:bg-gray-700 border border-gray-200 dark:border-gray-600 font-semibold text-gray-900 dark:text-white shadow-sm">
                    {currentFilterCount}
                  </span>
                  entries
                </div>

                {/* Segmented Control Pagination */}
                {totalPages > 1 && (
                  <div className="flex items-center p-1 bg-gray-100/80 dark:bg-gray-900/50 rounded-lg border border-gray-200 dark:border-gray-700/50 shadow-inner">
                    <button
                      onClick={() => setCurrentPage(p => Math.max(1, p - 1))}
                      disabled={currentPage === 1}
                      className="flex items-center gap-1 px-3 py-1.5 text-sm font-medium rounded-md text-gray-600 dark:text-gray-400 hover:bg-white dark:hover:bg-gray-800 hover:text-indigo-600 dark:hover:text-indigo-400 hover:shadow-sm disabled:opacity-40 disabled:hover:bg-transparent disabled:hover:shadow-none disabled:hover:text-gray-600 transition-all"
                    >
                      <SafeIcon icon={FiChevronLeft} className="w-4 h-4" />
                      Prev
                    </button>

                    <div className="flex items-center px-4 border-x border-gray-200 dark:border-gray-700/50">
                      <span className="text-sm font-bold text-gray-900 dark:text-white">
                        {currentPage}
                       </span>
                      <span className="text-sm text-gray-400 dark:text-gray-500 mx-1.5">/</span>
                      <span className="text-sm font-medium text-gray-500 dark:text-gray-400">
                        {totalPages}
                      </span>
                     </div>

                    <button
                      onClick={() => setCurrentPage(p => Math.min(totalPages, p + 1))}
                      disabled={currentPage === totalPages}
                       className="flex items-center gap-1 px-3 py-1.5 text-sm font-medium rounded-md text-gray-600 dark:text-gray-400 hover:bg-white dark:hover:bg-gray-800 hover:text-indigo-600 dark:hover:text-indigo-400 hover:shadow-sm disabled:opacity-40 disabled:hover:bg-transparent disabled:hover:shadow-none disabled:hover:text-gray-600 transition-all"
                    >
                      Next
                      <SafeIcon icon={FiChevronRight} className="w-4 h-4" />
                    </button>
                  </div>
                )}
              </div>
            )}
            
           </div>
        </div>
    </div>
  );
};

export default SeoRedirects;
