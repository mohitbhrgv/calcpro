<?php
// Core utilities
require_once '../../core/config.php';
require_once '../../core/response.php';
require_once '../../core/auth.php';

// --- SECURITY CHECK ---
$admin_data = verifyAdminToken();

try {
    // --- 1. Get Incoming Data ---
    $json = file_get_contents('php://input');
    $data = json_decode($json);

    // --- 2. Validation ---
    if (!isset($data->ids) || !is_array($data->ids) || empty($data->ids)) {
        sendResponse(false, null, 'No items selected.', 400, 'INVALID_INPUT');
    }

    // Sanitize IDs to strictly integers
    $ids = array_filter(array_map('intval', $data->ids));
    
    if (empty($ids)) {
        sendResponse(false, null, 'Invalid IDs provided.', 400, 'INVALID_INPUT');
    }

    // Generate dynamic placeholders (e.g., "?, ?, ?")
    $placeholders = implode(',', array_fill(0, count($ids), '?'));
    $paramsList = array_values($ids);

    // --- 3. Execute Deletion ---
    // We delegate relationship constraints natively to the database engine.
    $sql = "DELETE FROM categories WHERE id IN ($placeholders)";
    $stmt = $conn->prepare($sql);
    
    if ($stmt->execute($paramsList)) {
         $affected = $stmt->rowCount();
         if ($affected > 0) {
             sendResponse(true, ['affected' => $affected], "Selected categories deleted successfully.", 200);
         } else {
             sendResponse(false, null, 'No records were deleted. They may have already been removed.', 404, 'NO_CHANGES');
         }
    } else {
         $errorInfo = $stmt->errorInfo();
         throw new Exception("Database query failed: " . $errorInfo[2]);
    }

} catch (Exception $e) {
    error_log("Error in delete_categories_bulk.php: " . $e->getMessage());
    
    // Fallback detection for MySQL 1451 Constraint Violation Error
    // This catches strict non-cascade DB locks and returns a clean 409 response
    if (strpos($e->getMessage(), '1451') !== false || (is_a($e, 'PDOException') && isset($e->errorInfo[1]) && $e->errorInfo[1] == 1451)) {
        sendResponse(false, null, 'Cannot delete one or more categories because they are currently in use.', 409, 'CONFLICT_IN_USE');
    } else {
        sendResponse(false, null, 'An internal server error occurred.', 500, 'SERVER_ERROR');
    }
}

$conn = null;
?>
