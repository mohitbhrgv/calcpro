// --- SECURE MICRO-EDITOR FOR FAQS ---
const MicroFaqEditor = ({ value, onChange }) => {
  const editor = useEditor({
    extensions: [
      StarterKit.configure({
        heading: false,
        bulletList: false,
        orderedList: false,
        codeBlock: false,
        blockquote: false,
        horizontalRule: false,
        strike: false,
        code: false, 
      })
    ],
    content: value || "",
    editorProps: {
      attributes: {
        // FIX: Explicitly enforce bg-white and text-gray-900 for light mode
        class: 'w-full min-h-[100px] px-3 py-2 border border-gray-300 dark:border-gray-600 rounded-lg focus:outline-none focus:ring-2 focus:ring-indigo-500 focus:border-transparent bg-white text-gray-900 dark:bg-gray-700 dark:text-white transition-colors text-sm',
      },
    },
    onUpdate: ({ editor }) => {
      onChange(editor.getHTML());
    },
  });

  return (
    <div className="relative group">
      <style>{`
        .group .ProseMirror p { margin-bottom: 0.5em; line-height: 1.5; }
        .group .ProseMirror p:last-child { margin-bottom: 0; }
        .group .ProseMirror b, .group .ProseMirror strong { font-weight: bold; color: inherit; }
        .group .ProseMirror i, .group .ProseMirror em { font-style: italic; }
      `}</style>
      <EditorContent editor={editor} />
    </div>
  );
};

// --- Local FAQ Editor ---
const SimpleFaqEditor = ({ faqs, onFaqChange }) => {
  const handleFaqItemChange = (index, field, value) => {
    // FIX: Map over the array to create a fresh object instead of mutating the existing one
    const newFaqs = faqs.map((faq, i) => 
      i === index ? { ...faq, [field]: value } : faq
    );
    onFaqChange(newFaqs);
  };
  
  const addFaqItem = () => {
    onFaqChange([...faqs, { question: "", answer: "" }]);
  };
  
  const removeFaqItem = (index) => {
    if (window.confirm("Delete this FAQ item?")) {
      onFaqChange(faqs.filter((_, i) => i !== index));
    }
  };

  // FIX: Explicit bg-white text-gray-900 placeholder-gray-400 for light mode
  const inputClass = "w-full px-3 py-2 border border-gray-300 dark:border-gray-600 rounded-lg focus:ring-2 focus:ring-indigo-500 focus:border-transparent bg-white text-gray-900 dark:bg-gray-700 dark:text-white placeholder-gray-400 transition-colors outline-none";

  return (
    <div className="space-y-4">
      {faqs.map((faq, index) => (
        <div
          key={index}
          className="p-5 border rounded-lg bg-gray-50 dark:bg-gray-700/30 border-gray-200 dark:border-gray-700"
        >
          <div className="flex justify-between items-center mb-3">
            <label className="text-xs font-bold text-gray-500 dark:text-gray-400 uppercase tracking-wider">
               Question {index + 1}
            </label>
            <button
              type="button"
              onClick={() => removeFaqItem(index)}
              className="text-red-500 hover:text-red-600 p-1.5 hover:bg-red-50 dark:hover:bg-red-900/20 rounded-md transition-colors focus:outline-none"
            >
               <SafeIcon icon={FiTrash2} className="w-4 h-4" />
            </button>
          </div>
          <div className="space-y-3">
            <div>
               <input
                 type="text"
                 value={faq.question}
                 onChange={(e) => handleFaqItemChange(index, "question", e.target.value)}
                 placeholder="Enter the question"
                 className={inputClass}
               />
            </div>
            <div>
               <div className="flex justify-between items-end mb-1 px-1">
                  <span className="text-[10px] font-bold text-gray-400 uppercase tracking-wider">Answer</span>
                  <span className="text-[10px] font-semibold text-indigo-500 dark:text-indigo-400 select-none">
                     Tip: Select text + Ctrl+B (Bold) / Ctrl+I (Italic)
                  </span>
               </div>
               <MicroFaqEditor
                 value={faq.answer}
                 onChange={(val) => handleFaqItemChange(index, "answer", val)}
               />
            </div>
          </div>
        </div>
      ))}
      <button
        type="button"
        onClick={addFaqItem}
        className="w-full py-3 border-2 border-dashed border-gray-300 dark:border-gray-600 rounded-xl text-gray-500 dark:text-gray-400 font-medium hover:border-indigo-500 hover:text-indigo-500 dark:hover:border-indigo-400 dark:hover:text-indigo-400 transition-colors flex items-center justify-center gap-2 focus:outline-none focus:ring-2 focus:ring-indigo-500"
      >
        <SafeIcon icon={FiPlus} className="w-5 h-5" />
        <span>Add FAQ Item</span>
      </button>
    </div>
  );
};

const CalculatorEditor = () => {
  const { id } = useParams();
  const navigate = useNavigate();
  const { invalidateCache } = useAdminCache();
  
  const titleTextareaRef = useRef(null);

  const isEditing = Boolean(id);
  
  // --- Global State ---
  const [initialData, setInitialData] = useState(null);
  const [globalSettings, setGlobalSettings] = useState(null);
  const [loading, setLoading] = useState(true);
  const [isSubmitting, setIsSubmitting] = useState(false);
  const [initialLoad, setInitialLoad] = useState(true);

  // --- Form Data ---
  const [isRobotsDefault, setIsRobotsDefault] = useState(true);

  const [formData, setFormData] = useState({
    name: "",
    slug: "",
    content: "",
    faq_json: [],
    status: "draft",
    categories: [],
    primary_category_id: null, 
    tags: [],
    is_featured: false,
    featuredImageId: null,

    // SEO - General
    metaTitle: "",
    metaDescription: "",
    focusKeywords: "",
    canonical_url: "",

    // SEO - Social (Facebook)
    ogTitle: "",
    ogDescription: "",
    og_image: "",
    
    // SEO - Social (Twitter)
    twitterTitle: "",
    twitterDescription: "",
    twitterImage: "",
    twitterSync: true,

    // Robots Meta
    no_index: false,
    no_follow: false,
    no_archive: false,
    no_image_index: false,
    no_snippet: false,
    maxSnippet: null,
    maxVideoPreview: null,
    maxImagePreview: null,
    
    schema_enabled: true,
  });

  const [isMinorEdit, setIsMinorEdit] = useState(true);
  
  const [activeRedirect, setActiveRedirect] = useState(null);
  const [originalSlug, setOriginalSlug] = useState(null);

  // --- Aux Data ---
  const [allCategories, setAllCategories] = useState([]);
  const [allTags, setAllTags] = useState([]);
  const [allMedia, setAllMedia] = useState([]);
  const [featuredImage, setFeaturedImage] = useState(null);

  // --- UI Toggles ---
  const [expandedSections, setExpandedSections] = useState({
    seo: true,
    categories: true,
    tags: true,
    featuredImage: true,
  });

  // --- Modals State ---
  const [isSnippetModalOpen, setIsSnippetModalOpen] = useState(false);
  const [isSeoHazardModalOpen, setIsSeoHazardModalOpen] = useState(false);
  const [isPermalinkModalOpen, setIsPermalinkModalOpen] = useState(false);
  const [showSaveConfirmation, setShowSaveConfirmation] = useState(false);
  const [confirmationInput, setConfirmationInput] = useState("");
  const [publishConfirmInput, setPublishConfirmInput] = useState("");

  const [isFeaturedMediaLibraryOpen, setIsFeaturedMediaLibraryOpen] = useState(false);
  const [isEditorMediaLibraryOpen, setIsEditorMediaLibraryOpen] = useState(false);
  const [editorImageContext, setEditorImageContext] = useState(null);

  // --- FLAT CATEGORY LOGIC ---
  const flatCategories = useMemo(() => allCategories, [allCategories]);

  // --- PRIMARY CATEGORY LOGIC ---
  useEffect(() => {
    const finalSelectedIds = Array.from(new Set(formData.categories));
    let newPrimaryId = formData.primary_category_id;

    if (finalSelectedIds.length > 0) {
        if (!finalSelectedIds.includes(newPrimaryId)) {
            newPrimaryId = finalSelectedIds[0];
        }
    } else {
        newPrimaryId = null;
    }
    
    const prevCats = [...formData.categories].sort();
    const newCats = [...finalSelectedIds].sort();

    if (
      JSON.stringify(prevCats) !== JSON.stringify(newCats) ||
      formData.primary_category_id !== newPrimaryId
    ) {
      setFormData(prev => ({
        ...prev,
        categories: finalSelectedIds,
        primary_category_id: newPrimaryId
      }));
    }
  }, [formData.categories, formData.primary_category_id]);

  const hasChanges = useMemo(() => {
    return JSON.stringify(formData) !== JSON.stringify(initialData);
  }, [formData, initialData]);

  // --- Data Fetching ---
  const fetchData = useCallback(async () => {
    setLoading(true);
    try {
      const [categoriesResult, tagsResult, mediaResult, settingsResult] = await Promise.all([
        authenticatedFetch("/api/admin/get_calculator_categories.php"),
        authenticatedFetch("/api/admin/get_tags.php?context=calculator"),
        authenticatedFetch("/api/admin/get_media.php"),
        authenticatedFetch("/api/admin/get_settings.php"),
      ]);

      if (categoriesResult.success) setAllCategories(categoriesResult.data);
      if (tagsResult.success) setAllTags(tagsResult.data);
      if (mediaResult.success) setAllMedia(mediaResult.data.media);

      let globalSet = {};
      if (settingsResult.success) {
          globalSet = settingsResult.data;
          setGlobalSettings(globalSet);
      }

      if (isEditing) {
        const result = await authenticatedFetch(
          `/api/admin/get_calculator.php?id=${id}`
        );

        if (result.success) {
          const calcData = result.data;

          setActiveRedirect(calcData.active_redirect || null);

          // Robots Logic
          const gIndex = globalSet.seo_robots_global_noindex === 'true';
          const isRobotsNull = 
              calcData.no_index === null &&
              calcData.no_follow === null &&
              calcData.no_archive === null &&
              calcData.no_image_index === null &&
              calcData.no_snippet === null;

          const isAdvancedNull = 
              calcData.max_snippet === null && 
              calcData.max_video_preview === null && 
              calcData.max_image_preview === null;

          setIsRobotsDefault(isRobotsNull && isAdvancedNull);
          
          delete calcData.active_redirect;

          const normalizedData = {
              ...calcData,
              // Core
              name: calcData.name || "",
              slug: calcData.slug || "",
              content: calcData.content || "",
              status: calcData.status || "draft",
              is_featured: Boolean(calcData.is_featured),
              featuredImageId: calcData.featuredImageId || null, 
              
              // Relations
              faq_json: Array.isArray(calcData.faq_json) ? calcData.faq_json : [],
              categories: calcData.categories || [],
              primary_category_id: calcData.primary_category_id || null, 
              tags: calcData.tags || [],
              
              // SEO - General
              metaTitle: calcData.meta_title || "",
              metaDescription: calcData.meta_description || "",
              focusKeywords: calcData.focusKeywords || "",
              canonical_url: calcData.canonical_url || "",

              // Social SEO (Facebook)
              ogTitle: calcData.ogTitle || "",
              ogDescription: calcData.ogDescription || "",
              og_image: calcData.og_image || "",

              // Social SEO (Twitter)
              twitterTitle: calcData.twitterTitle || calcData.twitter_title || "",
              twitterDescription: calcData.twitterDescription || calcData.twitter_description || "",
              twitterImage: calcData.twitterImage || calcData.twitter_image || "",
              twitterSync: calcData.twitterSync !== undefined ? calcData.twitterSync : true,
              
              // Robots
              no_index: calcData.no_index === null ? gIndex : Boolean(calcData.no_index),
              no_follow: calcData.no_follow === null ? (globalSet.seo_robots_global_nofollow === 'true') : Boolean(calcData.no_follow),
              no_archive: calcData.no_archive === null ? (globalSet.seo_robots_global_noarchive === 'true') : Boolean(calcData.no_archive),
              no_image_index: calcData.no_image_index === null ? (globalSet.seo_robots_global_noimageindex === 'true') : Boolean(calcData.no_image_index),
              no_snippet: calcData.no_snippet === null ? (globalSet.seo_robots_global_nosnippet === 'true') : Boolean(calcData.no_snippet),
              
              maxSnippet: calcData.max_snippet,
              maxVideoPreview: calcData.max_video_preview,
              maxImagePreview: calcData.max_image_preview,
              schema_enabled: calcData.schema_enabled !== undefined ? Boolean(calcData.schema_enabled) : true,
          };

          setFormData(normalizedData);
          // FIX: Deep clone initial data so Update button works securely
          setInitialData(JSON.parse(JSON.stringify(normalizedData)));
          setOriginalSlug(calcData.slug);

          if (normalizedData.featuredImageId) {
            const image = mediaResult.data.media.find(item => item.id === normalizedData.featuredImageId);
            if (image) setFeaturedImage(image);
          }
        } else {
          toast.error(result.message || "Failed to load calculator data.");
          navigate("/content/calculators");
        }
      } else {
        // Create New: Defaults
        let defaultData = { ...formData };
        if(globalSet) {
             defaultData = {
                 ...defaultData,
                 no_index: globalSet.seo_robots_global_noindex === 'true',
                 no_follow: globalSet.seo_robots_global_nofollow === 'true',
                 no_archive: globalSet.seo_robots_global_noarchive === 'true',
                 no_image_index: globalSet.seo_robots_global_noimageindex === 'true',
                 no_snippet: globalSet.seo_robots_global_nosnippet === 'true',
                 maxSnippet: null,
                 maxVideoPreview: null,
                 maxImagePreview: null
             };
             setFormData(defaultData);
         }
         setIsRobotsDefault(true);
         setInitialData(JSON.parse(JSON.stringify(defaultData)));
      }
    } catch (error) {
       toast.error("Could not connect to the server.");
    } finally {
      setLoading(false);
      setInitialLoad(false);
    }
  }, [id, isEditing, navigate]);

  useEffect(() => {
    fetchData();
  }, [id]);

  // --- Handlers ---
  const toggleRobotsDefault = () => {
      if (!globalSettings) return;

      if (!isRobotsDefault) {
          // Revert
          setFormData(prev => ({
              ...prev,
              no_index: globalSettings.seo_robots_global_noindex === 'true',
              no_follow: globalSettings.seo_robots_global_nofollow === 'true',
              no_archive: globalSettings.seo_robots_global_noarchive === 'true',
              no_image_index: globalSettings.seo_robots_global_noimageindex === 'true',
              no_snippet: globalSettings.seo_robots_global_nosnippet === 'true',
              maxSnippet: null,
              maxVideoPreview: null,
              maxImagePreview: null
          }));
          setIsRobotsDefault(true);
          toast.success("Restored to global defaults.");
      } else {
          // Unlock
          setFormData(prev => ({
            ...prev,
            no_index: globalSettings.seo_robots_global_noindex === 'true',
            no_follow: globalSettings.seo_robots_global_nofollow === 'true',
            no_archive: globalSettings.seo_robots_global_noarchive === 'true',
            no_image_index: globalSettings.seo_robots_global_noimageindex === 'true',
            no_snippet: globalSettings.seo_robots_global_nosnippet === 'true',
            maxSnippet: parseInt(globalSettings.seo_robots_global_max_snippet),
            maxVideoPreview: parseInt(globalSettings.seo_robots_global_max_video_preview),
            maxImagePreview: globalSettings.seo_robots_global_max_image_preview
          }));
          setIsRobotsDefault(false);
      }
  };

  useEffect(() => {
    if (titleTextareaRef.current) {
      titleTextareaRef.current.style.height = "auto";
      titleTextareaRef.current.style.height =
        titleTextareaRef.current.scrollHeight + "px";
    }
  }, [formData.name]);

  const handleChange = (e) => {
    const { name, value, type, checked } = e.target;
    setFormData((prev) => {
      const newValues = {
        ...prev,
        [name]: type === "checkbox" ? checked : value,
      };
      if (name === "name" && !isEditing) {
        newValues.slug = value.toLowerCase().replace(/\s+/g, "-").replace(/[^a-z0-9-]/g, "");
        newValues.metaTitle = value;
      }
      return newValues;
    });
  };
  
  const handleSlugChange = (e) => {
    const cleanSlug = e.target.value.toLowerCase().replace(/[^a-z0-9-]/g, "-");
    setFormData((prev) => ({ ...prev, slug: cleanSlug }));
  };

  const handleSave = async () => {
    if (isEditing && formData.slug !== originalSlug) {
      setIsPermalinkModalOpen(true);
      return;
    }
    if (isEditing && initialData.status === "published" && formData.status !== "published") {
      setIsSeoHazardModalOpen(true);
      return;
    }
    setShowSaveConfirmation(true);
  };
  
  const executeSave = async () => {
    setIsSubmitting(true);
    toast.loading("Saving calculator...", { id: "calc-save-toast" });

    let dataToSend = { 
        ...formData,
        max_snippet: formData.maxSnippet,
        max_video_preview: formData.maxVideoPreview,
        max_image_preview: formData.maxImagePreview
    };

    if (isRobotsDefault) {
        dataToSend.no_index = null;
        dataToSend.no_follow = null;
        dataToSend.no_archive = null;
        dataToSend.no_image_index = null;
        dataToSend.no_snippet = null;
        dataToSend.max_snippet = null;
        dataToSend.max_video_preview = null;
        dataToSend.max_image_preview = null;
    }

    if (isEditing) dataToSend.id = id;
    if (isEditing && isMinorEdit) dataToSend.is_minor_edit = true;

    const url = isEditing
      ? `/api/admin/update_calculator.php`
      : `/api/admin/add_calculator.php`;

    // FIX: Standardized delay to 600ms matching Phase 1 & 2
    const minDelay = new Promise((resolve) => setTimeout(resolve, 600));

    try {
      const apiCall = authenticatedFetch(url, {
        method: "POST",
        body: JSON.stringify(dataToSend),
      });

      const [result] = await Promise.all([apiCall, minDelay]);

      if (result.success) {
        toast.success(result.message, { id: "calc-save-toast" });
        setIsMinorEdit(true); 
        setPublishConfirmInput("");

        if (!isEditing && result.data && result.data.id) {
          invalidateCache("calculators_");
          navigate(`/content/calculators/edit/${result.data.id}`, { replace: true });
        } else if (isEditing) {
          await fetchData();
        }
      } else {
        toast.error(result.message || "An unknown error occurred.", { id: "calc-save-toast" });
      }
    } catch (error) {
      toast.error("Could not connect to the server.", { id: "calc-save-toast" });
    } finally {
      setIsSubmitting(false);
    }
  };

  const handleCopyPath = () => {
    navigator.clipboard.writeText(`calculator/${originalSlug}`);
    toast.success("Path copied to clipboard!");
  };

  const getButtonLabel = () => {
    if (isSubmitting) return "Saving...";
    if (formData.status === "draft") return "Save Draft";
    return isEditing ? "Update" : "Save & Publish";
  };

  const toggleSection = (section) => {
    setExpandedSections((prev) => ({ ...prev, [section]: !prev[section] }));
  };

  const handleSelectFeaturedMedia = (media) => {
    setFeaturedImage(media);
    setFormData((prev) => ({ ...prev, featuredImageId: media.id }));
    setIsFeaturedMediaLibraryOpen(false);
  };

  const handleRemoveFeaturedImage = () => {
    setFeaturedImage(null);
    setFormData((prev) => ({ ...prev, featuredImageId: null }));
  };

  // --- Image Handling specific to Tiptap Editor Content ---
  const handleEditorMediaRequest = (context) => {
    setEditorImageContext(context);
    setIsEditorMediaLibraryOpen(true);
  };

  const handleSelectEditorMedia = (media) => {
    if (editorImageContext && editorImageContext.insertImage) {
      editorImageContext.insertImage(media.full_url, media.alt_text || '');
    }
    setIsEditorMediaLibraryOpen(false);
    setEditorImageContext(null);
  };

  if (loading && initialLoad) return <LoadingSpinner size="xl" className="h-64" />;

  const siteUrl = (globalSettings?.site_url || "https://yourdomain.com").replace(/\/$/, "");

  return (
    <>
      <MediaLibrary
        isOpen={isFeaturedMediaLibraryOpen}
        onClose={() => setIsFeaturedMediaLibraryOpen(false)}
        onSelectMedia={handleSelectFeaturedMedia}
        disableAnimation={true}
        initialSelectedUrl={featuredImage ? featuredImage.full_url : ""} 
      />

      <MediaLibrary
        isOpen={isEditorMediaLibraryOpen}
        onClose={() => { setIsEditorMediaLibraryOpen(false); setEditorImageContext(null); }}
        onSelectMedia={handleSelectEditorMedia}
        disableAnimation={true}
        initialSelectedUrl={editorImageContext?.src || ""} 
      />
      
      <PreviewSnippetEditor
        isOpen={isSnippetModalOpen}
        onClose={() => setIsSnippetModalOpen(false)}
        formData={formData}
        onFormDataChange={handleChange}
        basePath="calculator/"
        featuredImage={featuredImage}
        globalSettings={globalSettings}
      />

      <Modal
        isOpen={isSeoHazardModalOpen}
        onClose={() => setIsSeoHazardModalOpen(false)}
        title="SEO Warning: Unpublishing Content"
        type="warning"
      >
        <div className="space-y-4">
          <div className="bg-amber-50 dark:bg-amber-900/20 p-4 rounded-lg border border-amber-200 dark:border-amber-800 flex gap-3">
             <SafeIcon icon={FiAlertTriangle} className="w-6 h-6 text-amber-600 shrink-0" />
              <p className="text-xs text-amber-800 dark:text-amber-200">You are unpublishing a live calculator. This will break existing links.</p>
          </div>
          <div className="flex justify-end gap-2">
              <button onClick={() => setIsSeoHazardModalOpen(false)} className="px-4 py-2 text-gray-600 hover:bg-gray-100 rounded focus:outline-none">Cancel</button>
              <button onClick={() => { setIsSeoHazardModalOpen(false); executeSave(); }} className="px-4 py-2 bg-amber-600 text-white rounded focus:outline-none">Proceed</button>
          </div>
        </div>
      </Modal>

      <Modal
        isOpen={showSaveConfirmation}
        onClose={() => setShowSaveConfirmation(false)}
        title={formData.status === 'draft' ? "Save Draft" : (isEditing ? "Update Calculator" : "Publish Calculator")}
        type="default"
      >
         <div className="space-y-4">
             <div className="flex items-start space-x-3">
                  <div className={`p-2 rounded-full ${formData.status === "published" ? "bg-green-100 text-green-600" : "bg-blue-100 text-blue-600"}`}>
                       <SafeIcon icon={formData.status === "published" ? FiCheckCircle : FiFileText} className="w-5 h-5" />
                  </div>
                  <div>
                       <p className="text-gray-800 dark:text-gray-200 font-medium">
                           {formData.status === 'draft' ? "Your calculator will be saved as a draft." : "This content will be visible to the public."}
                       </p>
                  </div>
             </div>

             {formData.status !== 'draft' && (
                 <div className="mt-3">
                     <label className="block text-xs font-bold text-gray-500 mb-1 uppercase">Type PUBLISH to confirm:</label>
                     {/* FIX: Explicit background and text color added for light mode */}
                     <input 
                        type="text" 
                        placeholder="PUBLISH" 
                        value={publishConfirmInput} 
                        onChange={(e) => setPublishConfirmInput(e.target.value)} 
                        className="w-full px-3 py-2 border border-gray-300 dark:border-gray-600 rounded-lg bg-white text-gray-900 placeholder-gray-400 dark:bg-gray-700 dark:text-white focus:ring-2 focus:ring-blue-500 outline-none" 
                     />
                 </div>
             )}
             
             {isEditing && formData.status === "published" && (
                 <div className="p-3 bg-blue-50 dark:bg-blue-900/20 rounded-lg border border-blue-100 dark:border-blue-800">
                   <label className="flex items-start gap-3 cursor-pointer select-none">
                        <div className="flex items-center h-5">
                            <input type="checkbox" checked={isMinorEdit} onChange={(e) => setIsMinorEdit(e.target.checked)} className="w-4 h-4 text-blue-600 rounded focus:ring-blue-500 focus:outline-none" />
                         </div>
                        <div>
                            <span className="block text-sm font-semibold text-blue-800 dark:text-blue-300">Minor Update</span>
                            <span className="block text-xs text-blue-600 dark:text-blue-400">Preserve original publish date.</span>
                        </div>
                    </label>
                </div>
            )}

             <div className="flex justify-end gap-2 pt-3">
                  <button onClick={() => setShowSaveConfirmation(false)} className="px-4 py-2 text-gray-600 hover:bg-gray-100 rounded transition-colors focus:outline-none">Cancel</button>
                 <button 
                    onClick={() => { setShowSaveConfirmation(false); executeSave(); }}
                    disabled={formData.status !== 'draft' && publishConfirmInput !== 'PUBLISH'}
                    className={`px-5 py-2 text-white rounded-lg disabled:opacity-50 transition-colors focus:outline-none ${formData.status === 'published' ? 'bg-green-600 hover:bg-green-700' : 'bg-blue-600 hover:bg-blue-700'}`}
                >
                    Confirm
                </button>
             </div>
         </div>
      </Modal>

      <Modal
        isOpen={isPermalinkModalOpen}
        onClose={() => { setIsPermalinkModalOpen(false); setConfirmationInput(""); }}
        title="Danger Zone: URL Change"
        type="danger"
      >
        <div className="space-y-4">
             <p className="text-red-600 dark:text-red-400">Changing the slug will break existing links. Type CONFIRM to proceed.</p>
             {/* FIX: Explicit background and text color added for light mode */}
             <input type="text" value={confirmationInput} onChange={(e) => setConfirmationInput(e.target.value)} className="w-full border border-gray-300 dark:border-gray-600 p-2 rounded bg-white text-gray-900 dark:bg-gray-700 dark:text-white focus:ring-2 focus:ring-red-500 outline-none" />
             <div className="flex justify-end gap-2">
                 <button onClick={() => setIsPermalinkModalOpen(false)} className="px-4 py-2 text-gray-600 hover:bg-gray-100 rounded transition-colors focus:outline-none">Cancel</button>
                 <button onClick={() => { if(confirmationInput==='CONFIRM') { setIsPermalinkModalOpen(false); executeSave(); } }} disabled={confirmationInput !== 'CONFIRM'} className="px-4 py-2 bg-red-600 text-white rounded disabled:opacity-50 transition-colors focus:outline-none">Change</button>
             </div>
        </div>
      </Modal>

      <div className="space-y-6">
        <div className="flex items-center justify-between">
          <div className="flex items-center space-x-4">
            <button
              onClick={() => navigate("/content/calculators")}
              className="p-2 hover:bg-gray-100 dark:hover:bg-gray-700 rounded-lg transition-colors focus:outline-none"
            >
              <SafeIcon icon={FiArrowLeft} className="w-5 h-5 dark:text-gray-300" />
            </button>
            <div>
              <h1 className="text-2xl font-bold text-gray-900 dark:text-white">
                {isEditing ? "Edit Calculator" : "New Calculator"}
              </h1>
            </div>
          </div>
          <motion.button
            whileHover={!isSubmitting && hasChanges ? { scale: 1.05 } : {}}
            whileTap={!isSubmitting && hasChanges ? { scale: 0.95 } : {}}
            onClick={handleSave}
            disabled={isSubmitting || !hasChanges}
            className={`flex items-center space-x-2 px-5 py-2.5 rounded-xl shadow-sm font-semibold transition-all focus:outline-none ${!hasChanges ? "bg-gray-200 text-gray-500 dark:bg-gray-700 cursor-not-allowed" : formData.status === "published" ? "bg-green-600 text-white hover:bg-green-700" : "bg-white text-gray-700 border border-gray-200 hover:bg-gray-50 dark:bg-gray-800 dark:text-white dark:border-gray-700 dark:hover:bg-gray-700"}`}
          >
            <SafeIcon icon={FiSave} className="w-4 h-4" />
            <span>{getButtonLabel()}</span>
          </motion.button>
        </div>

        <div className="grid grid-cols-1 lg:grid-cols-12 gap-6">
          <div className="lg:col-span-9 space-y-6">
            {activeRedirect && (
              <div className="bg-indigo-50 dark:bg-indigo-900/20 border border-indigo-200 dark:border-indigo-800 rounded-xl p-4 flex items-start gap-4">
                <div className="p-2 bg-indigo-100 dark:bg-indigo-800/50 rounded-lg text-indigo-600 dark:text-indigo-400 shrink-0"><SafeIcon icon={FiCornerUpRight} className="w-6 h-6" /></div>
                <div className="flex-1 min-w-0">
                  <h4 className="text-sm font-bold text-indigo-900 dark:text-indigo-100 flex items-center gap-2">
                    Active Redirection Detected
                  </h4>
                  <p className="text-sm text-indigo-800 dark:text-indigo-300 mt-1">
                    Traffic to this page is currently being redirected to: {activeRedirect.target_url}
                  </p>
                </div>
                <button onClick={() => window.open("/seo/redirects", "_blank")} className="px-4 py-2 bg-white border border-indigo-200 rounded-lg text-sm text-indigo-600 hover:bg-indigo-50 transition-colors focus:outline-none">Manage</button>
              </div>
            )}

            <div className="bg-white dark:bg-gray-800 rounded-xl p-6 border border-gray-200 dark:border-gray-700 space-y-4">
              {/* FIX: Explicit text-gray-900 placeholder-gray-400 added for light mode */}
              <textarea
                ref={titleTextareaRef}
                name="name"
                value={formData.name}
                onChange={handleChange}
                rows={1}
                placeholder="Enter calculator name..."
                className="w-full text-3xl font-bold border-none focus:ring-0 bg-transparent text-gray-900 dark:text-white p-0 placeholder-gray-400 dark:placeholder-gray-500 resize-none overflow-hidden leading-snug"
                style={{ minHeight: "42px" }}
              />
              {formData.slug && (
                 <div className="text-sm text-gray-500 dark:text-gray-400 flex items-center gap-1 mb-4 mt-1">
                  <span className="font-medium text-gray-400 dark:text-gray-500">Permalink:</span>
                  <a href={`${siteUrl}/calculator/${formData.slug}`} target="_blank" rel="noopener noreferrer" className="text-blue-500 hover:underline break-all hover:text-blue-600 flex items-center gap-1">
                    {`${siteUrl}/calculator/${formData.slug}`} <SafeIcon icon={FiExternalLink} className="w-3.5 h-3.5 opacity-75" />
                  </a>
                  <button onClick={handleCopyPath} className="ml-2 text-gray-400 hover:text-gray-600 transition-colors focus:outline-none" title="Copy Slug"><SafeIcon icon={FiCopy} className="w-3.5 h-3.5" /></button>
                </div>
              )}
              
              <div className="relative min-h-[500px] border-t border-gray-100 dark:border-gray-700 pt-4">
                <TiptapEditor
                  value={formData.content}
                  onChange={(val) => setFormData(prev => ({ ...prev, content: val }))}
                  onRequestMedia={handleEditorMediaRequest}
                />
              </div>
            </div>

            <div className="bg-white dark:bg-gray-800 rounded-xl p-6 border border-gray-200 dark:border-gray-700 shadow-sm">
              <h3 className="text-lg font-semibold text-gray-900 dark:text-white mb-4 flex items-center gap-2">
                 <SafeIcon icon={FiFileText} className="w-5 h-5 text-indigo-500" /> FAQ Section
              </h3>
              <SimpleFaqEditor
                faqs={formData.faq_json}
                onFaqChange={(newFaqs) => setFormData(prev => ({ ...prev, faq_json: newFaqs }))}
              />
            </div>
          </div>

          <div className="lg:col-span-3">
             <CalculatorSidebar 
                formData={formData}
                initialData={initialData}
                categoryTree={flatCategories}
                allTags={allTags}
                globalSettings={globalSettings}
                featuredImage={featuredImage}
                
                expandedSections={expandedSections}
                toggleSection={(section) => setExpandedSections(prev => ({...prev, [section]: !prev[section]}))}
                isRobotsDefault={isRobotsDefault}
                
                handleChange={handleChange}
                handleSlugChange={handleSlugChange}
                handleStatusChange={(val) => setFormData(prev => ({ ...prev, status: val }))}
                
                handleCategoryChange={(id) => {
                     setFormData(prev => ({
                        ...prev,
                        categories: prev.categories.includes(id)
                          ? prev.categories.filter((c) => c !== id)
                          : [...prev.categories, id],
                     }));
                }}

                handleTagsChange={(newTags) => setFormData(prev => ({...prev, tags: newTags}))}
                
                getKeywordsArray={() => formData.focusKeywords ? formData.focusKeywords.split(',').filter(Boolean) : []}
                handleKeywordsChange={(newKeywords) => setFormData(prev => ({ ...prev, focusKeywords: newKeywords.join(',') }))}
                
                toggleRobotsDefault={toggleRobotsDefault}
                setIsSnippetModalOpen={setIsSnippetModalOpen}
                
                setIsFeaturedMediaLibraryOpen={setIsFeaturedMediaLibraryOpen}
                handleRemoveFeaturedImage={handleRemoveFeaturedImage}
                
                onMaxSnippetChange={(val) => setFormData(prev => ({...prev, maxSnippet: val}))}
                onMaxVideoChange={(val) => setFormData(prev => ({...prev, maxVideoPreview: val}))}
                onMaxImageChange={(val) => setFormData(prev => ({...prev, maxImagePreview: val}))}
             />
          </div>
        </div>
      </div>
    </>
  );
};

export default CalculatorEditor;
