<?php
// Core utilities
require_once '../../core/config.php';
require_once '../../core/response.php';
require_once '../../core/auth.php';

use Firebase\JWT\JWT;
use Firebase\JWT\Key;

// --- SECURITY CHECK ---
// Verifies the JWT and gets the logged-in admin's ID.
$admin_data = verifyAdminToken();
$admin_id = $admin_data->adminId;

try {
    // --- 1. Get Incoming Data ---
    $json = file_get_contents('php://input');
    $data = json_decode($json);

    // --- 2. Input Presence Validation ---
    if (!isset($data->currentPassword) || empty($data->currentPassword)) {
        sendResponse(false, null, 'Current password is required.', 400, 'INVALID_INPUT');
    }
    if (!isset($data->newPassword) || empty($data->newPassword)) {
        sendResponse(false, null, 'New password is required.', 400, 'INVALID_INPUT');
    }

    // --- 3. Sanitize Data ---
    // Explicitly cast to string to prevent type-juggling attacks, do not trim.
    $currentPassword = (string)$data->currentPassword;
    $newPassword = (string)$data->newPassword;

    // --- 4. Strict Cryptographic Policy Enforcement ---
    // Prevent CPU Exhaustion (Denial of Service)
    if (strlen($newPassword) > 64) {
        sendResponse(false, null, 'Password is too long. Maximum allowed length is 64 characters.', 400, 'PASSWORD_TOO_LONG');
    }

    // [UPDATED] Strict 16-character minimum (NIST Guidelines - Length over Complexity)
    if (strlen($newPassword) < 16) {
        sendResponse(false, null, 'Password must be at least 16 characters long.', 400, 'WEAK_PASSWORD');
    }

    // --- 5. Verify Current Password ---
    $sql_fetch = "SELECT password FROM admins WHERE id = :id";
    $stmt_fetch = $conn->prepare($sql_fetch);
    $stmt_fetch->execute([':id' => $admin_id]);
    $admin = $stmt_fetch->fetch(PDO::FETCH_ASSOC);

    if (!$admin) {
        sendResponse(false, null, 'Admin account not found.', 404, 'NOT_FOUND');
    }
    
    $currentHashedPassword = $admin['password'];
    
    if (!password_verify($currentPassword, $currentHashedPassword)) {
        // TYPO FORGIVENESS: Changed from 401 to 400.
        // This prevents the frontend apiService.js from logging the user out instantly if they make a typo.
        sendResponse(false, null, 'The current password you entered is incorrect.', 400, 'INVALID_CREDENTIALS');
    }

    // --- 6. Hash and Update the New Password (TRANSACTIONAL) ---
    $newHashedPassword = password_hash($newPassword, PASSWORD_BCRYPT);
    
    $conn->beginTransaction();
    try {
        // A. Update password and force concurrency session kill
        $sql_update = "UPDATE admins SET password = :password, password_last_updated = CURRENT_TIMESTAMP, current_session_jti = NULL WHERE id = :id";
        $stmt_update = $conn->prepare($sql_update);
        $stmt_update->execute([
            ':password' => $newHashedPassword,
            ':id' => $admin_id
        ]);

        // B. ORPHANED TRUST TOKEN PURGE: Destroy all trusted devices to enforce 2FA globally
        $sql_purge_trust = "DELETE FROM admin_trusted_devices WHERE admin_id = :id";
        $stmt_purge_trust = $conn->prepare($sql_purge_trust);
        $stmt_purge_trust->execute([':id' => $admin_id]);

        $conn->commit();
    } catch (Exception $dbEx) {
        $conn->rollBack();
        throw new Exception("Database password update failed: " . $dbEx->getMessage());
    }

    // --- 7. SESSION REVOCATION (DUAL-TOKEN ARCHITECTURE) ---
    $jwtSecretKey = $_ENV['JWT_SECRET'] ?? '';
    $accessToken = $_COOKIE['calcpro_access_token'] ?? null;
    $refreshToken = $_COOKIE['calcpro_refresh_token'] ?? null;
    $sessionRevoked = false;

    if (!empty($jwtSecretKey)) {
        try {
            // Idempotent insertion using ON DUPLICATE KEY UPDATE to prevent MySQL crash
            $stmt_kill = $conn->prepare("INSERT INTO jwt_blacklist (jti, expires_at) VALUES (:jti, :exp) ON DUPLICATE KEY UPDATE expires_at = :exp2");
            
            if ($accessToken) {
                $decAccess = JWT::decode($accessToken, new Key($jwtSecretKey, 'HS256'));
                if (isset($decAccess->jti, $decAccess->exp)) {
                    $stmt_kill->execute([':jti' => $decAccess->jti, ':exp' => $decAccess->exp, ':exp2' => $decAccess->exp]);
                    $sessionRevoked = true;
                }
            }
            
            if ($refreshToken) {
                $decRefresh = JWT::decode($refreshToken, new Key($jwtSecretKey, 'HS256'));
                if (isset($decRefresh->jti, $decRefresh->exp)) {
                    $stmt_kill->execute([':jti' => $decRefresh->jti, ':exp' => $decRefresh->exp, ':exp2' => $decRefresh->exp]);
                    $sessionRevoked = true;
                }
            }
        } catch (Exception $revokeEx) {
            // Token parsing failed or DB error occurred during blacklist insertion
            error_log("Session Revocation Failure [Admin ID {$admin_id}]: " . $revokeEx->getMessage());
        }
    }

    // Standardized Cookie Hardening (Destroy cookies physically to clean the browser state)
    if (function_exists('revokeSecureSessionCookie')) {
        revokeSecureSessionCookie('calcpro_access_token');
        revokeSecureSessionCookie('calcpro_refresh_token');
        revokeSecureSessionCookie('calcpro_csrf_token');
        revokeSecureSessionCookie('calcpro_csrf_payload');
        revokeSecureSessionCookie('calcpro_admin_token'); // Clear legacy if exists
    }

    if (!$sessionRevoked && ($accessToken || $refreshToken)) {
        // Failsafe: The password updated, but the active token couldn't be blacklisted securely in DB.
        error_log("CRITICAL: Failed to automatically revoke JWT for Admin ID: {$admin_id} after password update.");
        sendResponse(true, null, 'Password updated! Note: We could not securely terminate your current session. Please log out manually.');
    } else {
        // Inform the client the operation succeeded, forcing a redirect via the 401 interceptor loop
        sendResponse(true, null, 'Password updated securely! Please log in again with your new password.');
    }

} catch (Exception $e) {
    error_log("Error in update_admin_password.php: " . $e->getMessage());
    sendResponse(false, null, 'An internal server error occurred.', 500, 'SERVER_ERROR');
}

$conn = null;
?>
