import React, { useState, useEffect, useCallback, useMemo, useRef } from "react";
import { Link, useNavigate } from "react-router-dom";
import { createPortal } from "react-dom";
import { motion, AnimatePresence } from "framer-motion";
import Modal from "../../components/common/Modal";
import SafeIcon from "../../../common/SafeIcon";
import * as FiIcons from "react-icons/fi";
import toast from "react-hot-toast";
import { authenticatedFetch } from "../../utils/apiService";
import LoadingSpinner from "../../components/common/LoadingSpinner";

const {
  FiPlus,
  FiGrid,
  FiSearch,
  FiTrash2,
  FiAlertTriangle,
  FiChevronDown,
  FiList,
  FiUser,
  FiSettings,
  FiRefreshCw,
  FiChevronLeft,
  FiChevronRight,
  FiSave,
  FiEdit2,
  FiFileText
} = FiIcons;

// --- Premium Viewport-Aware Headless Portal Dropdown ---
const CustomSelect = ({ value, onChange, options, className }) => {
  const [isOpen, setIsOpen] = useState(false);
  const dropdownRef = useRef(null);
  const portalRef = useRef(null);
  const [dropdownStyles, setDropdownStyles] = useState({});

  useEffect(() => {
    const handleClickOutside = (event) => {
      const clickedOutsideButton = dropdownRef.current && !dropdownRef.current.contains(event.target);
      const clickedOutsidePortal = portalRef.current && !portalRef.current.contains(event.target);
      
      if (clickedOutsideButton && clickedOutsidePortal) {
          setIsOpen(false);
      }
    };
    document.addEventListener('mousedown', handleClickOutside);
    
    const updatePosition = () => {
        if (isOpen && dropdownRef.current) {
            const rect = dropdownRef.current.getBoundingClientRect();
            const spaceBelow = window.innerHeight - rect.bottom;
            const placeAbove = spaceBelow < 250 && rect.top > spaceBelow;

            setDropdownStyles({
                top: placeAbove ? 'auto' : rect.bottom + window.scrollY + 4,
                bottom: placeAbove ? (window.innerHeight - rect.top - window.scrollY + 4) : 'auto',
                left: rect.left + window.scrollX,
                width: rect.width
            });
        }
    };
    
    window.addEventListener('scroll', updatePosition, true);
    window.addEventListener('resize', updatePosition);
    
    return () => {
        document.removeEventListener('mousedown', handleClickOutside);
        window.removeEventListener('scroll', updatePosition, true);
        window.removeEventListener('resize', updatePosition);
    };
  }, [isOpen]);

  const handleToggle = () => {
     if (!isOpen && dropdownRef.current) {
         const rect = dropdownRef.current.getBoundingClientRect();
         const spaceBelow = window.innerHeight - rect.bottom;
         const placeAbove = spaceBelow < 250 && rect.top > spaceBelow;

         setDropdownStyles({
            top: placeAbove ? 'auto' : rect.bottom + window.scrollY + 4,
            bottom: placeAbove ? (window.innerHeight - rect.top - window.scrollY + 4) : 'auto',
            left: rect.left + window.scrollX,
            width: rect.width
         });
     }
     setIsOpen(!isOpen);
  };

  const selectedLabel = options.find(opt => String(opt.value) === String(value))?.label || '';

  const dropdownContent = (
      <AnimatePresence>
        {isOpen && (
          <motion.div
            ref={portalRef}
            initial={{ opacity: 0, y: dropdownStyles.bottom !== 'auto' ? 5 : -5 }} 
            animate={{ opacity: 1, y: 0 }} 
            exit={{ opacity: 0, y: dropdownStyles.bottom !== 'auto' ? 5 : -5 }} 
            transition={{ duration: 0.15 }}
            style={dropdownStyles}
            className="absolute z-[99999] bg-white dark:bg-gray-800 border border-gray-200 dark:border-gray-700 rounded-xl shadow-2xl overflow-hidden py-1"
          >
            <div className="max-h-60 overflow-y-auto custom-scrollbar">
                {options.map((opt) => (
                  <button
                    key={opt.value}
                    type="button"
                    onClick={(e) => { e.stopPropagation(); onChange(opt.value); setIsOpen(false); }}
                    className={`w-full text-left px-4 py-2.5 text-sm transition-colors ${
                      String(value) === String(opt.value) 
                        ? 'bg-emerald-50 dark:bg-emerald-900/30 text-emerald-700 dark:text-emerald-400 font-bold' 
                        : 'text-gray-900 dark:text-gray-300 hover:bg-gray-50 dark:hover:bg-gray-700 font-medium'
                    }`}
                  >
                    {opt.label}
                  </button>
                ))}
            </div>
          </motion.div>
        )}
      </AnimatePresence>
  );

  return (
    <div className={`relative ${className || ''}`} ref={dropdownRef}>
      <button
        type="button"
        onClick={handleToggle}
        className="flex items-center justify-between w-full bg-white dark:bg-gray-800 border border-gray-300 dark:border-gray-600 text-gray-900 dark:text-white rounded-xl px-4 py-2.5 text-sm focus:ring-2 focus:ring-emerald-500 outline-none transition-shadow font-medium cursor-pointer shadow-sm"
      >
        <span className="truncate pr-2">{selectedLabel}</span>
        <SafeIcon 
            icon={FiChevronDown} 
            className={`w-4 h-4 text-gray-500 shrink-0 transition-transform duration-200 ${isOpen ? 'rotate-180' : ''}`} 
        />
      </button>
      {createPortal(dropdownContent, document.body)}
    </div>
  );
};

// Helper component for wrapped text cells
const WrappedCell = ({ text, isSlug = false }) => (
  <div className={`text-sm text-gray-900 dark:text-white whitespace-normal ${isSlug ? "break-all font-mono text-xs text-gray-500 dark:text-gray-400" : "break-words font-semibold"} max-w-[180px] leading-snug`}>
    {text}
  </div>
);

const limitOptions = [
    { value: 10, label: "10 Rows" },
    { value: 25, label: "25 Rows" },
    { value: 50, label: "50 Rows" },
    { value: 100, label: "100 Rows" }
];

const sleep = (ms) => new Promise(resolve => setTimeout(resolve, ms));

const Authors = () => {
  const navigate = useNavigate();

  // --- Core State ---
  const [allAuthors, setAuthors] = useState([]);
  const [searchTerm, setSearchTerm] = useState("");
  const [debouncedSearch, setDebouncedSearch] = useState("");

  // UX State Architecture (FCP Setup)
  const [initialLoading, setInitialLoading] = useState(true);
  const [isUpdating, setIsUpdating] = useState(false);
  const [isManualRefresh, setIsManualRefresh] = useState(false);

  // Pagination & Display Settings
  const [isSettingsOpen, setIsSettingsOpen] = useState(false);
  const [isSettingsSubmitting, setIsSettingsSubmitting] = useState(false);
  const [displayLimit, setDisplayLimit] = useState(() => Number(localStorage.getItem('content_authors_display_limit')) || 25);
  const [tempDisplayLimit, setTempDisplayLimit] = useState(displayLimit);
  const [currentPage, setCurrentPage] = useState(1);

  // Deletion Modal
  const [isDeleteModalOpen, setIsDeleteModalOpen] = useState(false);
  const [authorToDelete, setAuthorToDelete] = useState(null);
  const [actionProcessing, setActionProcessing] = useState(false);

  const fetchAuthors = useCallback(async (forceRefresh = false) => {
    if (forceRefresh) setIsUpdating(true);
    try {
      const response = await authenticatedFetch("/api/admin/get_author_profiles.php");
      if (response.success) {
        setAuthors(response.data);
      } else {
        toast.error("Failed to load authors");
      }
    } catch (error) {
      toast.error("Error connecting to server.");
    } finally {
      setIsUpdating(false);
      setInitialLoading(false);
    }
  }, []);

  useEffect(() => {
    fetchAuthors();
  }, [fetchAuthors]);

  // Search Debounce Engine
  useEffect(() => {
    const handler = setTimeout(() => {
        setDebouncedSearch(searchTerm);
    }, 500);
    return () => clearTimeout(handler);
  }, [searchTerm]);

  // Reset pagination when dataset filters change
  useEffect(() => {
      setCurrentPage(1);
  }, [debouncedSearch]);

  const filteredAuthors = useMemo(() => {
    if (!debouncedSearch) return allAuthors;
    return allAuthors.filter(
      (author) =>
        author.name?.toLowerCase().includes(debouncedSearch.toLowerCase()) ||
        author.slug?.toLowerCase().includes(debouncedSearch.toLowerCase()) ||
        author.role?.toLowerCase().includes(debouncedSearch.toLowerCase())
    );
  }, [allAuthors, debouncedSearch]);

  // Custom Pagination Math
  const totalRecords = filteredAuthors.length;
  const totalPages = Math.max(1, Math.ceil(totalRecords / displayLimit));
  
  const paginatedAuthors = useMemo(() => {
      const start = (currentPage - 1) * displayLimit;
      return filteredAuthors.slice(start, start + displayLimit);
  }, [filteredAuthors, currentPage, displayLimit]);

  // --- UX Enhancement: Manual Refresh with Labor Gap ---
  const handleManualRefresh = async () => {
    if (isUpdating || isManualRefresh) return;
    setIsManualRefresh(true);
    await fetchAuthors(true);
    setTimeout(() => setIsManualRefresh(false), 800); 
  };

  // --- UX Enhancement: Settings Manager & Dirty State ---
  const handleOpenSettings = () => {
      setTempDisplayLimit(displayLimit);
      setIsSettingsOpen(true);
  };

  const handleSaveSettings = async () => {
      setIsSettingsSubmitting(true);
      await sleep(800);
      
      setDisplayLimit(tempDisplayLimit);
      localStorage.setItem('content_authors_display_limit', tempDisplayLimit);
      setCurrentPage(1);
      
      setIsSettingsSubmitting(false);
      setIsSettingsOpen(false);
      toast.success("Display settings strictly updated.");
  };

  const isSettingsDirty = tempDisplayLimit !== displayLimit;

  // --- Actions ---
  const handleEditNavigate = (id) => {
    navigate(`/content/authors/edit/${id}`);
  };

  const handleDeleteClick = (author) => {
    if (author.is_super_admin == 1) {
      toast.error("The System Profile cannot be deleted.");
      return;
    }
    setAuthorToDelete(author);
    setIsDeleteModalOpen(true);
  };

  const confirmDelete = async () => {
    if (!authorToDelete) return;
    
    setActionProcessing(true);
    try {
      await sleep(800); // Consistent delay structure
      const response = await authenticatedFetch(`/api/admin/delete_author_profile.php?id=${authorToDelete.id}`, {
        method: "DELETE"
      });
      if (response.success) {
        toast.success("Author profile deleted.");
        setAuthors(allAuthors.filter(a => a.id !== authorToDelete.id));
      } else {
        toast.error(response.message || "Failed to delete.");
      }
    } catch (error) {
      toast.error("Error deleting author.");
    } finally {
      setActionProcessing(false);
      setIsDeleteModalOpen(false);
      setAuthorToDelete(null);
    }
  };

  return (
    <div className="space-y-6 relative">

      {/* --- DISPLAY SETTINGS MODAL --- */}
      <Modal isOpen={isSettingsOpen} onClose={() => !isSettingsSubmitting && setIsSettingsOpen(false)} title="Author View Settings" size="md" disableAnimation={true}>
        <div className="space-y-5 pt-2 pb-4 max-h-[60vh] overflow-y-auto custom-scrollbar pr-2">
            <div className="bg-emerald-50/50 dark:bg-emerald-900/10 p-5 rounded-2xl border border-emerald-100 dark:border-emerald-800/30 shadow-sm transition-colors">
                <h3 className="text-sm font-extrabold text-emerald-900 dark:text-emerald-300 uppercase tracking-wider flex items-center gap-2 mb-2">
                    <SafeIcon icon={FiList} className="w-4 h-4 text-emerald-500" />
                    Display Settings
                </h3>
                <p className="text-[11px] text-emerald-800/80 dark:text-emerald-200/70 mb-5 leading-relaxed font-medium">
                    Configure your local display preferences for the Authors module. This controls how many records are rendered per page.
                </p>
                <div>
                    <label className="block text-[11px] font-bold text-gray-500 dark:text-gray-400 uppercase tracking-wide mb-2">
                        Pagination Limit
                    </label>
                    <CustomSelect 
                        value={tempDisplayLimit} 
                        onChange={(val) => setTempDisplayLimit(parseInt(val))} 
                        options={limitOptions} 
                    />
                </div>
            </div>
        </div>
        <div className="flex justify-end gap-3 pt-4 border-t border-gray-100 dark:border-gray-700 bg-white dark:bg-gray-800">
            <button 
                onClick={() => setIsSettingsOpen(false)} 
                disabled={isSettingsSubmitting} 
                className="px-4 py-2.5 text-sm font-bold text-gray-600 dark:text-gray-300 bg-white dark:bg-gray-800 border border-gray-200 dark:border-gray-600 hover:bg-gray-50 dark:hover:bg-gray-700 rounded-lg transition-colors shadow-sm focus:outline-none disabled:opacity-50"
            >
                Cancel
            </button>
            <button 
                onClick={handleSaveSettings} 
                disabled={isSettingsSubmitting || !isSettingsDirty} 
                className="flex items-center gap-2 px-5 py-2.5 text-sm font-bold text-white bg-emerald-600 hover:bg-emerald-700 rounded-lg transition-all shadow-md active:scale-95 disabled:opacity-50 disabled:cursor-not-allowed focus:outline-none"
            >
                {isSettingsSubmitting ? (
                    <><SafeIcon icon={FiRefreshCw} className="animate-spin w-4 h-4" /> Saving...</>
                ) : (
                    <><SafeIcon icon={FiSave} className="w-4 h-4" /> Save Policies</>
                )}
            </button>
        </div>
      </Modal>

      {/* --- HEADER CARD --- */}
      <motion.div 
        initial={{ opacity: 0, y: -10 }} animate={{ opacity: 1, y: 0 }} 
        className="flex items-start gap-4 p-5 rounded-xl bg-emerald-50 dark:bg-emerald-900/10 border border-emerald-100 dark:border-emerald-800/30 shadow-sm"
      >
        <div className="p-3 bg-emerald-100 dark:bg-emerald-800/50 rounded-xl text-emerald-600 dark:text-emerald-400 shrink-0 shadow-sm border border-emerald-200 dark:border-emerald-700/50">
          <SafeIcon icon={FiUser} className="w-6 h-6" />
        </div>
        <div className="flex-1">
          <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-4">
            <div>
              <h1 className="text-xl font-bold text-gray-900 dark:text-white tracking-tight">
                Authors
              </h1>
              <p className="text-sm text-gray-600 dark:text-gray-300 mt-1 leading-relaxed">
                Manage public display profiles for your articles. The System Profile is permanent.
              </p>
            </div>
            <motion.button
              whileHover={{ scale: 1.02 }}
              whileTap={{ scale: 0.98 }}
              onClick={() => navigate("/content/authors/new")}
              className="flex items-center justify-center space-x-2 bg-emerald-600 text-white px-5 py-2.5 rounded-lg shadow-sm hover:bg-emerald-700 transition-colors font-bold text-sm whitespace-nowrap outline-none focus:outline-none"
            >
              <SafeIcon icon={FiPlus} className="w-4 h-4" />
              <span>Add Profile</span>
            </motion.button>
          </div>
        </div>
      </motion.div>

      {/* --- HTML DATA TABLE & TOOLBAR (FCP Optimized) --- */}
      <motion.div 
        initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} transition={{ delay: 0.2 }}
        className="bg-white dark:bg-gray-800 rounded-xl shadow-sm border border-gray-200 dark:border-gray-700 overflow-hidden relative flex flex-col min-h-[400px]"
      >
        <AnimatePresence>
          {(initialLoading || isUpdating || actionProcessing) && (
            <motion.div
              initial={{ opacity: 0 }}
              animate={{ opacity: 1 }}
              exit={{ opacity: 0 }}
              className="absolute inset-0 bg-white/50 dark:bg-gray-800/50 backdrop-blur-[1px] z-10 flex items-center justify-center"
            >
              <LoadingSpinner />
            </motion.div>
          )}
        </AnimatePresence>

        <div className="px-6 py-4 border-b border-gray-200 dark:border-gray-700 bg-gray-50 dark:bg-gray-700/50 flex flex-col xl:flex-row justify-between items-center gap-4 relative z-20">
          
          <div className="flex items-center gap-2">
            <div className="bg-white dark:bg-gray-800 px-3 py-1.5 rounded-md border border-gray-200 dark:border-gray-600 text-xs font-bold text-gray-600 dark:text-gray-300 shadow-sm flex items-center gap-2">
              <SafeIcon icon={FiList} className="w-3 h-3 text-emerald-500" />
              {filteredAuthors.length} Authors
            </div>
          </div>

          <div className="flex items-center gap-3 w-full sm:w-auto justify-end">
            <div className="relative w-full sm:w-64">
              <SafeIcon
                icon={FiSearch}
                className="absolute left-3 top-1/2 transform -translate-y-1/2 text-gray-400 w-4 h-4"
              />
              <input
                type="text"
                placeholder="Search authors..."
                value={searchTerm}
                onChange={(e) => setSearchTerm(e.target.value)}
                className="w-full pl-9 pr-4 py-2 text-sm border border-gray-300 dark:border-gray-600 rounded-lg bg-white text-gray-900 placeholder-gray-400 focus:ring-2 focus:ring-emerald-500 focus:border-transparent dark:bg-gray-800 dark:text-white transition-shadow outline-none shadow-sm"
              />
            </div>

            <div className="flex items-center gap-1 border-l border-gray-200 dark:border-gray-700 pl-3 ml-auto shrink-0">
                <button
                    onClick={handleOpenSettings}
                    className="p-2 text-gray-400 hover:text-gray-700 dark:hover:text-gray-300 hover:bg-gray-200 dark:hover:bg-gray-600 rounded-lg transition-colors focus:outline-none"
                    title="Display Settings"
                >
                    <SafeIcon icon={FiSettings} className="w-4 h-4" />
                </button>
                <button
                    onClick={handleManualRefresh}
                    disabled={isUpdating || isManualRefresh}
                    className={`p-2 text-emerald-600 bg-emerald-50 hover:bg-emerald-100 dark:bg-emerald-900/20 dark:text-emerald-300 dark:hover:bg-emerald-900/40 rounded-lg transition-colors border border-emerald-100 dark:border-emerald-800 focus:outline-none ${
                        isManualRefresh || isUpdating ? "opacity-50 cursor-not-allowed" : ""
                    }`}
                    title="Refresh Data"
                >
                    <SafeIcon icon={FiRefreshCw} className={`w-4 h-4 ${isManualRefresh ? "animate-spin" : ""}`} />
                </button>
            </div>
          </div>
        </div>

        <div className={`w-full overflow-x-auto flex flex-col flex-1 transition-opacity duration-200 ${(initialLoading || isUpdating || actionProcessing) ? "opacity-50 pointer-events-none" : ""}`}>
          {paginatedAuthors.length === 0 && !initialLoading ? (
            <div className="flex flex-col items-center justify-center p-16 text-center flex-1">
              <SafeIcon icon={FiGrid} className="w-12 h-12 text-gray-400 dark:text-gray-500 mb-4 opacity-50" />
              <h3 className="text-lg font-bold text-gray-900 dark:text-white mb-2">No Authors Found</h3>
              <p className="text-sm text-gray-500 dark:text-gray-400 max-w-sm">No author profiles match the current search criteria.</p>
            </div>
          ) : (
            <table className="w-full text-left border-collapse min-w-[900px]">
              <thead>
                <tr className="bg-gray-50 dark:bg-gray-700/50 text-gray-600 dark:text-gray-300 text-xs font-bold uppercase tracking-wider border-b border-gray-200 dark:border-gray-700">
                  <th className="px-6 py-4">Profile</th>
                  <th className="px-6 py-4">Role</th>
                  <th className="px-6 py-4">Published Posts</th>
                  <th className="px-6 py-4">URL Slug</th>
                  <th className="px-6 py-4">Created Date</th>
                  <th className="px-6 py-4 text-right">Actions</th>
                </tr>
              </thead>
              <tbody className="divide-y divide-gray-200 dark:divide-gray-700 text-sm text-gray-900 dark:text-gray-100">
                {paginatedAuthors.map((author) => (
                  <tr key={author.id} className="transition-colors hover:bg-gray-50 dark:hover:bg-gray-700/30">
                    <td className="px-6 py-4">
                        <div className="flex items-center space-x-3">
                            {author.profile_image_url ? (
                                <img src={author.profile_image_url} alt={author.name} className="w-10 h-10 rounded-full object-cover border border-gray-200 dark:border-gray-700 bg-white" />
                            ) : (
                                <div className="w-10 h-10 rounded-full bg-emerald-100 dark:bg-emerald-900/40 flex items-center justify-center text-emerald-600 dark:text-emerald-400 shrink-0 border border-emerald-200 dark:border-emerald-700/50">
                                    <SafeIcon icon={FiUser} className="w-5 h-5" />
                                </div>
                            )}
                            <WrappedCell text={author.name} />
                        </div>
                    </td>
                    <td className="px-6 py-4">
                        <span className="px-2.5 py-1 bg-gray-100 text-gray-700 dark:bg-gray-700 dark:text-gray-300 text-xs rounded-full font-bold uppercase tracking-wider shadow-sm">
                            {author.role || 'Author'}
                        </span>
                    </td>
                    <td className="px-6 py-4">
                        <div className="flex items-center space-x-1.5 text-sm text-gray-600 dark:text-gray-300 font-medium bg-gray-100 dark:bg-gray-800 px-2.5 py-1 rounded-md shadow-sm w-max border border-gray-200 dark:border-gray-700">
                             <SafeIcon icon={FiFileText} className="w-4 h-4 text-gray-400" />
                             <span>{author.total_posts || 0} Articles</span>
                        </div>
                    </td>
                    <td className="px-6 py-4">
                        <span className="text-sm text-emerald-600 dark:text-emerald-400 font-medium">/{author.slug || 'N/A'}</span>
                    </td>
                    <td className="px-6 py-4">
                        <span className="text-sm text-gray-500 dark:text-gray-400">{new Date(author.created_at).toLocaleDateString()}</span>
                    </td>
                    <td className="px-6 py-4 text-right">
                      <div className="flex items-center justify-end gap-1">
                         <button 
                            onClick={() => handleEditNavigate(author.id)} 
                            className="p-2 text-emerald-600 hover:bg-emerald-50 dark:hover:bg-emerald-900/30 dark:text-emerald-400 rounded-lg transition-colors focus:outline-none" 
                            title="Edit Author"
                         >
                            <SafeIcon icon={FiEdit2} className="w-4 h-4" />
                         </button>
                        {author.is_super_admin != 1 && (
                            <button 
                                onClick={() => handleDeleteClick(author)} 
                                className="p-2 text-gray-400 hover:text-red-600 hover:bg-red-50 dark:hover:bg-red-900/30 rounded-lg transition-colors focus:outline-none" 
                                title="Delete Author"
                            >
                                <SafeIcon icon={FiTrash2} className="w-4 h-4" />
                            </button>
                        )}
                      </div>
                    </td>
                  </tr>
                ))}
              </tbody>
            </table>
          )}

          {/* CUSTOM PAGINATION ENGINE */}
          {((totalRecords > 0) || (totalPages > 1)) && (
              <div className="px-6 py-4 border-t border-gray-200 dark:border-gray-700 bg-gray-50/50 dark:bg-gray-800/50 flex flex-col sm:flex-row items-center justify-between gap-4 transition-all mt-auto rounded-b-xl">
                  
                  <div className="flex items-center text-sm text-gray-500 dark:text-gray-400">
                      Showing
                      <span className="mx-1.5 px-2 py-0.5 rounded-md bg-white dark:bg-gray-700 border border-gray-200 dark:border-gray-600 font-semibold text-gray-900 dark:text-white shadow-sm">
                          {totalRecords 
                              ? Math.min(((currentPage - 1) * displayLimit) + 1, totalRecords) 
                              : currentPage}
                      </span>
                      {totalRecords ? (
                          <>
                              to
                              <span className="mx-1.5 px-2 py-0.5 rounded-md bg-white dark:bg-gray-700 border border-gray-200 dark:border-gray-600 font-semibold text-gray-900 dark:text-white shadow-sm">
                                  {Math.min((currentPage * displayLimit), totalRecords)}
                              </span>
                              of
                              <span className="mx-1.5 px-2 py-0.5 rounded-md bg-white dark:bg-gray-700 border border-gray-200 dark:border-gray-600 font-semibold text-gray-900 dark:text-white shadow-sm">
                                  {totalRecords}
                              </span>
                              entries
                          </>
                      ) : <span className="text-gray-900 dark:text-gray-100 font-medium ml-1">Page View</span>}
                  </div>

                  {(totalPages > 1) && (
                      <div className="flex items-center p-1 bg-gray-100/80 dark:bg-gray-900/50 rounded-lg border border-gray-200 dark:border-gray-700/50 shadow-inner">
                          <button
                              onClick={() => setCurrentPage(Math.max(1, currentPage - 1))}
                              disabled={currentPage === 1}
                              className="flex items-center gap-1 px-3 py-1.5 text-sm font-medium rounded-md text-gray-600 dark:text-gray-400 hover:bg-white dark:hover:bg-gray-800 hover:text-emerald-600 dark:hover:text-emerald-400 hover:shadow-sm disabled:opacity-40 disabled:hover:bg-transparent disabled:hover:shadow-none disabled:hover:text-gray-600 transition-all focus:outline-none"
                          >
                              <SafeIcon icon={FiChevronLeft} className="w-4 h-4" /> Prev
                          </button>

                          <div className="flex items-center px-4 border-x border-gray-200 dark:border-gray-700/50">
                              <span className="text-sm font-bold text-gray-900 dark:text-white">
                                  {currentPage}
                              </span>
                              <span className="text-sm text-gray-400 dark:text-gray-500 mx-1.5">/</span>
                              <span className="text-sm font-medium text-gray-500 dark:text-gray-400">
                                  {totalPages}
                              </span>
                          </div>

                          <button
                              onClick={() => setCurrentPage(Math.min(totalPages, currentPage + 1))}
                              disabled={currentPage === totalPages}
                              className="flex items-center gap-1 px-3 py-1.5 text-sm font-medium rounded-md text-gray-600 dark:text-gray-400 hover:bg-white dark:hover:bg-gray-800 hover:text-emerald-600 dark:hover:text-emerald-400 hover:shadow-sm disabled:opacity-40 disabled:hover:bg-transparent disabled:hover:shadow-none disabled:hover:text-gray-600 transition-all focus:outline-none"
                          >
                              Next <SafeIcon icon={FiChevronRight} className="w-4 h-4" />
                          </button>
                      </div>
                  )}
              </div>
          )}
        </div>
      </motion.div>

      {/* --- DELETE CONFIRMATION MODAL --- */}
      <Modal
        isOpen={isDeleteModalOpen}
        onClose={() => !actionProcessing && setIsDeleteModalOpen(false)}
        title="Delete Guest Author"
        type="danger"
        size="md"
        disableAnimation={true}
      >
        <div className="space-y-4">
          <div className="bg-red-50 dark:bg-red-900/20 p-4 rounded-lg border border-red-200 dark:border-red-800 flex items-start gap-3">
            <SafeIcon
              icon={FiAlertTriangle}
              className="w-6 h-6 text-red-600 shrink-0 mt-0.5"
            />
            <div>
              <h4 className="text-sm font-bold text-red-900 dark:text-red-100">
                Permanent Action
              </h4>
              <p className="text-xs text-red-800 dark:text-red-200/80 mt-1 leading-relaxed">
                Are you sure you want to delete the profile for <strong>{authorToDelete?.name}</strong>? Any posts currently assigned to them will revert to the internal system admin name. This action cannot be undone.
              </p>
            </div>
          </div>
          <div className="flex justify-end gap-3 pt-2 border-t border-gray-100 dark:border-gray-700 mt-4">
            <button
              onClick={() => setIsDeleteModalOpen(false)}
              disabled={actionProcessing}
              className="px-4 py-2.5 text-sm font-bold text-gray-600 dark:text-gray-300 hover:bg-gray-100 dark:hover:bg-gray-700 rounded-lg transition-colors outline-none disabled:opacity-50"
            >
              Cancel
            </button>
            <button
              onClick={confirmDelete}
              disabled={actionProcessing}
              className="px-5 py-2.5 bg-red-600 text-white rounded-lg hover:bg-red-700 disabled:opacity-50 disabled:cursor-not-allowed transition-colors text-sm font-bold shadow-sm outline-none flex items-center gap-2"
            >
              {actionProcessing ? "Deleting..." : "Delete Profile"}
            </button>
          </div>
        </div>
      </Modal>

    </div>
  );
};

export default Authors;
