import React from 'react';

class ErrorBoundary extends React.Component {
  constructor(props) {
    super(props);
    this.state = { hasError: false, error: null };
  }

  static getDerivedStateFromError(error) {
    return { hasError: true, error };
  }

  componentDidCatch(error, errorInfo) {
    const payload = {
      environment: 'cpanel',
      severity: 'critical',
      error_message: error.message || 'Unknown Admin UI Crash',
      stack_trace: errorInfo.componentStack || error.stack || '',
      url_endpoint: window.location.href,
    };

    // --- Strict Telemetry Pipeline ---
    const envUrl = import.meta.env.VITE_API_BASE_URL;
    let telemetryUrl = '/api/public/log_telemetry.php';
    
    if (envUrl) {
       const cleanBase = envUrl.replace(/\/+$/, '');
       telemetryUrl = `${cleanBase}/api/public/log_telemetry.php`;
    }

    fetch(telemetryUrl, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(payload)
    }).catch(err => console.error("Telemetry pipeline delivery failed:", err));
  }

  render() {
    if (this.state.hasError) {
      return (
        <div className="flex flex-col items-center justify-center min-h-screen bg-slate-50 dark:bg-[#020617] p-6 font-inter relative overflow-hidden">
          {/* Ambient Lighting Background */}
          <div className="absolute top-0 left-0 w-full h-full overflow-hidden z-0 pointer-events-none">
            <div className="absolute top-[-10%] left-[-10%] w-[500px] h-[500px] bg-red-500/10 rounded-full blur-[120px] mix-blend-multiply dark:mix-blend-screen animate-pulse-slow"></div>
            <div className="absolute bottom-[-10%] right-[-10%] w-[600px] h-[600px] bg-slate-600/10 rounded-full blur-[150px] mix-blend-multiply dark:mix-blend-screen"></div>
          </div>

          <div className="max-w-md w-full bg-white/80 dark:bg-[#0B1221]/80 backdrop-blur-3xl p-8 sm:p-10 rounded-[2.5rem] shadow-[0_20px_40px_rgb(0,0,0,0.1)] dark:shadow-[0_20px_40px_rgb(0,0,0,0.5)] border border-slate-200/50 dark:border-slate-800/60 text-center relative z-10">
            <div className="absolute top-0 left-0 right-0 h-[1px] bg-gradient-to-r from-transparent via-red-500/20 to-transparent"></div>
            
            <div className="w-16 h-16 bg-red-50 dark:bg-red-500/10 border border-red-100 dark:border-red-500/20 text-red-600 dark:text-red-400 rounded-2xl flex items-center justify-center mx-auto mb-6 shadow-sm">
              <svg className="w-8 h-8" fill="none" stroke="currentColor" viewBox="0 0 24 24">
                <path strokeLinecap="round" strokeLinejoin="round" strokeWidth="2" d="M12 9v2m0 4h.01m-6.938 4h13.856c1.54 0 2.502-1.667 1.732-3L13.732 4c-.77-1.333-2.694-1.333-3.464 0L3.34 16c-.77 1.333.192 3 1.732 3z" />
              </svg>
            </div>
            
            <h1 className="text-2xl font-bold text-slate-900 dark:text-white mb-3 font-montserrat tracking-tight">Interface Error</h1>
            <p className="text-slate-600 dark:text-slate-400 text-sm mb-8 leading-relaxed font-medium">
              A module within the admin panel has crashed. The telemetry system has securely logged the complete stack trace for your review.
            </p>
            
            <button 
              onClick={() => window.location.reload()} 
              className="bg-slate-900 hover:bg-slate-800 dark:bg-blue-600 dark:hover:bg-blue-500 text-white px-6 py-3.5 rounded-2xl font-semibold transition-all shadow-md hover:shadow-lg w-full tracking-wide border border-transparent dark:border-blue-500/30"
            >
              Refresh Dashboard
            </button>
          </div>
        </div>
      );
    }
    return this.props.children;
  }
}

export default ErrorBoundary;
