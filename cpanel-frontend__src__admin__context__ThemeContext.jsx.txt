import React, { createContext, useContext, useState, useEffect } from "react";
// Import flushSync to ensure React state updates perfectly align with the View Transition
import { flushSync } from "react-dom";

const ThemeContext = createContext();

export const useTheme = () => {
  const context = useContext(ThemeContext);
  if (!context) {
    throw new Error("useTheme must be used within a ThemeProvider");
  }
  return context;
};

export const ThemeProvider = ({ children }) => {
  const [isDark, setIsDark] = useState(() => {
    const saved = localStorage.getItem("calcpro-theme");
    return saved ? JSON.parse(saved) : false;
  });

  // Base initialization and fallback sync
  useEffect(() => {
    localStorage.setItem("calcpro-theme", JSON.stringify(isDark));
    if (isDark) {
      document.documentElement.classList.add("dark");
    } else {
      document.documentElement.classList.remove("dark");
    }
  }, [isDark]);

  // [ENTERPRISE UX FIX] Cinematic View Transitions
  // This takes a "snapshot" of the current screen, instantly updates the theme, 
  // and relies on the browser's native compositor to smoothly crossfade the pixels.
  const toggleTheme = () => {
    const newTheme = !isDark;

    // Fallback for older browsers that do not support View Transitions
    if (!document.startViewTransition) {
      setIsDark(newTheme);
      return;
    }

    // Modern Enterprise Cinematic Switch
    document.startViewTransition(() => {
      flushSync(() => {
        setIsDark(newTheme);
      });
    });
  };

  return (
    <ThemeContext.Provider value={{ isDark, toggleTheme }}>
      {children}
    </ThemeContext.Provider>
  );
};
