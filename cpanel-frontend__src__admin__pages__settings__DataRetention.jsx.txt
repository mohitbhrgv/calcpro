// --- Premium Viewport-Aware Headless Portal Dropdown ---
const CustomSelect = ({ value, onChange, options, className }) => {
  const [isOpen, setIsOpen] = useState(false);
  const dropdownRef = useRef(null);
  const portalRef = useRef(null);
  const [dropdownStyles, setDropdownStyles] = useState({});

  useEffect(() => {
    const handleClickOutside = (event) => {
      const clickedOutsideButton = dropdownRef.current && !dropdownRef.current.contains(event.target);
      const clickedOutsidePortal = portalRef.current && !portalRef.current.contains(event.target);
      
      if (clickedOutsideButton && clickedOutsidePortal) {
          setIsOpen(false);
      }
    };
    document.addEventListener('mousedown', handleClickOutside);
    
    const updatePosition = () => {
        if (isOpen && dropdownRef.current) {
            const rect = dropdownRef.current.getBoundingClientRect();
            const spaceBelow = window.innerHeight - rect.bottom;
            const placeAbove = spaceBelow < 250 && rect.top > spaceBelow;

            setDropdownStyles({
                top: placeAbove ? 'auto' : rect.bottom + window.scrollY + 4,
                bottom: placeAbove ? (window.innerHeight - rect.top - window.scrollY + 4) : 'auto',
                left: rect.left + window.scrollX,
                width: rect.width
            });
        }
    };
    
    window.addEventListener('scroll', updatePosition, true);
    window.addEventListener('resize', updatePosition);
    
    return () => {
        document.removeEventListener('mousedown', handleClickOutside);
        window.removeEventListener('scroll', updatePosition, true);
        window.removeEventListener('resize', updatePosition);
    };
  }, [isOpen]);

  const handleToggle = () => {
     if (!isOpen && dropdownRef.current) {
         const rect = dropdownRef.current.getBoundingClientRect();
         const spaceBelow = window.innerHeight - rect.bottom;
         const placeAbove = spaceBelow < 250 && rect.top > spaceBelow;

         setDropdownStyles({
            top: placeAbove ? 'auto' : rect.bottom + window.scrollY + 4,
            bottom: placeAbove ? (window.innerHeight - rect.top - window.scrollY + 4) : 'auto',
            left: rect.left + window.scrollX,
            width: rect.width
         });
     }
     setIsOpen(!isOpen);
  };

  const selectedLabel = options.find(opt => String(opt.value) === String(value))?.label || '';

  const dropdownContent = (
      <AnimatePresence>
        {isOpen && (
          <motion.div
            ref={portalRef}
            initial={{ opacity: 0, y: dropdownStyles.bottom !== 'auto' ? 5 : -5 }} 
            animate={{ opacity: 1, y: 0 }} 
            exit={{ opacity: 0, y: dropdownStyles.bottom !== 'auto' ? 5 : -5 }} 
            transition={{ duration: 0.15 }}
            style={dropdownStyles}
            className="absolute z-[99999] bg-white dark:bg-[#0f172a] border border-gray-200 dark:border-gray-700 rounded-xl shadow-2xl overflow-hidden py-1"
          >
            <div className="max-h-60 overflow-y-auto custom-scrollbar">
                {options.map((opt) => (
                  <button
                    key={opt.value}
                    type="button"
                    onClick={(e) => { e.stopPropagation(); onChange(opt.value); setIsOpen(false); }}
                    className={`w-full text-left px-4 py-3 text-[13px] transition-colors ${
                      String(value) === String(opt.value) 
                        ? 'bg-indigo-50 dark:bg-indigo-900/30 text-indigo-700 dark:text-indigo-400 font-bold' 
                        : 'text-gray-900 dark:text-gray-300 hover:bg-gray-50 dark:hover:bg-gray-800 font-medium'
                    }`}
                  >
                    {opt.label}
                  </button>
                ))}
            </div>
          </motion.div>
        )}
      </AnimatePresence>
  );

  return (
    <div className={`relative w-full lg:w-64 shrink-0 ${className || ''}`} ref={dropdownRef}>
      <button
        type="button"
        onClick={handleToggle}
        className="flex items-center justify-between w-full bg-white dark:bg-[#0B1221] border border-gray-300 dark:border-gray-700 text-gray-900 dark:text-white rounded-xl px-4 py-2.5 text-[14px] focus:ring-2 focus:ring-indigo-500 outline-none transition-all font-semibold cursor-pointer shadow-sm hover:border-gray-400 dark:hover:border-gray-500"
      >
        <span className="truncate pr-2">{selectedLabel}</span>
        <SafeIcon 
            icon={FiIcons.FiChevronDown} 
            className={`w-4 h-4 text-gray-400 shrink-0 transition-transform duration-200 ${isOpen ? 'rotate-180 text-indigo-500' : ''}`} 
        />
      </button>
      {createPortal(dropdownContent, document.body)}
    </div>
  );
};

// --- Standard Status Indicator ---
const ImpactBadge = ({ days, type }) => {
  let impact = "Optimal";
  let styleClass = "bg-emerald-100 text-emerald-700 border-emerald-200 dark:bg-emerald-900/30 dark:text-emerald-400 dark:border-emerald-800/50";
  
  if (type === "minutes") {
    if (days > 60) { impact = "Moderate Risk"; styleClass = "bg-amber-100 text-amber-700 border-amber-200 dark:bg-amber-900/30 dark:text-amber-400 dark:border-amber-800/50"; }
    if (days >= 1440) { impact = "High Storage"; styleClass = "bg-red-100 text-red-800 border-red-200 dark:bg-red-900/30 dark:text-red-400 dark:border-red-800/50"; }
  } else if (type === "hours") {
    if (days > 24) { impact = "Moderate Risk"; styleClass = "bg-amber-100 text-amber-700 border-amber-200 dark:bg-amber-900/30 dark:text-amber-400 dark:border-amber-800/50"; }
    if (days > 48) { impact = "Security Risk"; styleClass = "bg-red-100 text-red-800 border-red-200 dark:bg-red-900/30 dark:text-red-400 dark:border-red-800/50"; }
  } else {
    if (days > 30) { impact = "High Storage"; styleClass = "bg-amber-100 text-amber-700 border-amber-200 dark:bg-amber-900/30 dark:text-amber-400 dark:border-amber-800/50"; }
    if (days >= 180) { impact = "Data Bloat"; styleClass = "bg-red-100 text-red-800 border-red-200 dark:bg-red-900/30 dark:text-red-400 dark:border-red-800/50"; }
  }

  return (
    <span className={`px-2 py-0.5 text-[10px] font-extrabold tracking-wider uppercase rounded-md border shadow-sm whitespace-nowrap ${styleClass}`}>
      {impact}
    </span>
  );
};
