/**
 * Custom Error class to carry HTTP status codes and backend error codes.
 */
export class ApiError extends Error {
  constructor(message, status, errorCode = null) {
    super(message);
    this.name = "ApiError";
    this.status = status;
    this.errorCode = errorCode;
    // ENTERPRISE FIX: Use a hard boolean flag to bypass Babel/Vite transpilation issues
    this.isBusinessLogicError = true;
  }
}

// [NEW] THE EXIT SILENCER
let isIntentionalLogout = false;

export const setIntentionalLogout = (state) => {
    isIntentionalLogout = state;
};

const handleUnauthorized = () => {
  // Shield check: If we are already logging out gracefully, ignore this 401 panic.
  if (isIntentionalLogout) return;

  // Break the Ping-Pong Routing Loop
  window.dispatchEvent(new CustomEvent("auth-unauthorized"));
  // [ENTERPRISE FIX] Modern BrowserRouter compatible redirect
  window.location.href = "/login";
};

// --- Phase 2: ENTERPRISE SEC FIX (REMOVED CLIENT CSRF GENERATION) ---
// Securely extracts the RAW payload token generated by the PHP backend.
const getCsrfPayloadToken = () => {
  const match = document.cookie.match(new RegExp('(^|;\\s*)calcpro_csrf_payload=([^;]*)'));
  return match ? decodeURIComponent(match[2]) : null;
};

// --- NEW: CENTRALIZED ENVIRONMENT ROUTING ("THE PHONEBOOK") ---
// Securely resolves endpoints against the environment's base URL.
// Handles trailing/leading slashes and supports incremental migration.
const getFullUrl = (endpoint) => {
  // Incremental Migration Safe-Guard: If the endpoint is already a full URL, use it directly.
  if (endpoint.startsWith('http://') || endpoint.startsWith('https://')) {
    return endpoint;
  }
  
  const baseUrl = import.meta.env.VITE_API_BASE_URL || '';
  if (!baseUrl) {
    console.warn("[API Security] VITE_API_BASE_URL is not defined in the environment. Routing may fail.");
    return endpoint;
  }

  // Enterprise Path Sanitization: Safely merge paths avoiding double-slashes
  const cleanBase = baseUrl.replace(/\/+$/, '');
  const cleanEndpoint = endpoint.replace(/^\/+/, '');
  return `${cleanBase}/${cleanEndpoint}`;
};

/**
 * A centralized function for making authenticated API calls.
 */
export const authenticatedFetch = async (url, options = {}) => {
  const method = (options.method || 'GET').toUpperCase();
  const isStateChanging = ['POST', 'PUT', 'DELETE', 'PATCH'].includes(method);
  
  // Automatically grab the CSRF token only for state-changing operations
  const csrfToken = isStateChanging ? getCsrfPayloadToken() : null;

  const fetchOptions = {
    mode: "cors", // [ENTERPRISE SEC FIX] Enforce strict Cross-Origin bounds for production
    ...options,
    credentials: "include", 
    headers: {
      "Accept": "application/json", // [ENTERPRISE SEC FIX] Force backend to respond accurately
      "Content-Type": "application/json",
      ...(csrfToken ? { "X-CSRF-Token": csrfToken } : {}),
      ...options.headers,
    },
  };

  // Resolve the final secure URL via the Phonebook logic
  const secureUrl = getFullUrl(url);

  try {
    const response = await fetch(secureUrl, fetchOptions);

    // --- Phase 2 (Admin Network): Global Watchdog (Fail-Safe Interceptor) ---
    if (response.status === 503) {
      let incidentCode = 'CRITICAL_SYSTEM_FAILURE';
      try {
          const errorData = await response.clone().json();
          if (errorData && errorData.error_code) incidentCode = errorData.error_code;
      } catch (e) {
          // [ENTERPRISE FIX] Satisfy ESLint no-empty & ensure SOC 2 Observability
          console.debug("[API] 503 payload is not valid JSON, using fallback incident code.");
      }
      // Broadcast the silent offline alarm to the root component
      window.dispatchEvent(new CustomEvent("SYSTEM_OFFLINE", { detail: incidentCode }));
      // Freeze this thread to prevent localized toast errors while the app unmounts
      return new Promise(() => {});
    }

    if (!response.ok) {
      let errorMessage = `HTTP error! status: ${response.status}`;
      let parsedErrorCode = null;

      try {
         // [ENTERPRISE FIX] Prevent SyntaxError crashes when Nginx returns raw HTML for 500 errors
         const contentType = response.headers.get("content-type");
         if (contentType && contentType.includes("application/json")) {
             const errorResult = await response.json();
             errorMessage = errorResult.message || errorMessage;
             // [SECURITY FIX] Capture the backend error code so the UI can respond accordingly
             parsedErrorCode = errorResult.error_code || null;
         } else {
             const textResult = await response.text();
             console.debug("[API] Non-JSON error payload received. Preserving stability.", textResult.substring(0, 150));
         }
      } catch (parseError) {
         // [ENTERPRISE FIX] Satisfy ESLint no-empty & ensure SOC 2 Observability
         console.debug("[API] Failed to parse error payload:", parseError.message);
      }

      // --- PHASE 3: THE CONCURRENCY LISTENER ---
      if (response.status === 401) {
          // --- ENTERPRISE FIX: UNFREEZE EXECUTION ---
          // Throw the error normally even during logout, so the AuthContext's 
          // try/catch block can handle it and proceed to clear the UI securely.
          if (isIntentionalLogout) {
              throw new ApiError(errorMessage, 401, parsedErrorCode);
          }
          
          if (parsedErrorCode === 'AUTH_CONCURRENCY_VIOLATION') {
              if (!options.skipConcurrencyHandler) {
                  window.dispatchEvent(new CustomEvent("auth-concurrency-violation", { detail: errorMessage }));
              }
          } else if (!options.skipAuthHandler) {
              handleUnauthorized();
          }
          throw new ApiError(errorMessage, 401, parsedErrorCode);
      }

      // --- PHASE 2: THE WALKIE-TALKIE BROADCASTER ---
      if (response.status === 403 && (parsedErrorCode === 'PASSWORD_EXPIRED_FORCED_UPDATE' || errorMessage.toLowerCase().includes("password has expired"))) {
          window.dispatchEvent(new CustomEvent("password-expired-halt"));
          throw new ApiError(errorMessage, response.status, parsedErrorCode);
      }
      
      throw new ApiError(errorMessage, response.status, parsedErrorCode);
    }

    // Signal user activity to reset inactivity timer
    window.dispatchEvent(new CustomEvent("user-activity"));

    if (response.status === 204) {
        return null;
    }

    return await response.json();
  } catch (error) {
    // Professional Error Logic: Suppress noise during intentional logouts
    if (!error.isBusinessLogicError || error.status >= 500) {
      if (!isIntentionalLogout) {
          console.error(`[Admin API System Fault] ${fetchOptions.method || 'GET'} ${secureUrl} :`, error);
      }
    }
    throw error;
  }
};
