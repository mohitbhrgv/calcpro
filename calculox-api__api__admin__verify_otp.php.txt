<?php
// File: calcpro-api/api/admin/verify_otp.php

use Firebase\JWT\JWT;

require '../../core/config.php';
require_once '../../core/response.php';
require_once '../../core/SecurityGuard.php'; 
require_once '../../vendor/autoload.php';

// Proxy-aware IP detection
$ip_address = getSecureRealIp();
$user_agent = $_SERVER['HTTP_USER_AGENT'] ?? 'unknown';
$ua_hash = hash('sha256', $user_agent);

$b_index = SecurityGuard::generateBlindIndex($ip_address);
$enc_ip = SecurityGuard::encryptIp($ip_address);

$json = file_get_contents('php://input');
$data = json_decode($json);

// Prevent unhandled exceptions and warnings from malformed payloads
if (json_last_error() !== JSON_ERROR_NONE || !is_object($data)) {
    sendResponse(false, null, 'Invalid request payload format.', 400, 'INVALID_PAYLOAD');
}

if (!isset($data->email) || !filter_var(trim($data->email), FILTER_VALIDATE_EMAIL)) {
    sendResponse(false, null, 'A valid email address is required.', 400, 'INVALID_INPUT');
}

// Cast to string safely using null coalescing to prevent PHP warnings
$otp_code = (string)($data->otp_code ?? '');

if ($otp_code === '' || !preg_match('/^\d{6}$/', $otp_code)) {
    sendResponse(false, null, 'A valid 6-digit OTP code is required.', 400, 'INVALID_INPUT');
}

$email = trim($data->email);

$guard = new SecurityGuard($conn, $ip_address);
$securityStatus = $guard->checkIpStatus();

if (!$securityStatus['allowed']) {
    sendResponse(false, null, $securityStatus['message'], 403, $securityStatus['code']);
}

try {
    // Enforce Strict UTC Timezone comparison
    $current_utc = gmdate('Y-m-d H:i:s');

    $sql = "
        SELECT 
            a.id, a.name, a.email, a.role, 
            otp.otp_code_hash
        FROM admins a
        JOIN admin_otp_verifications otp ON a.id = otp.admin_id
        WHERE a.email = :email AND otp.ip_blind_index = :b_index AND otp.expires_at > :current_utc
        ORDER BY otp.created_at DESC
        LIMIT 1
    ";
    
    $stmt = $conn->prepare($sql);
    $stmt->execute([':email' => $email, ':b_index' => $b_index, ':current_utc' => $current_utc]);
    $admin_data = $stmt->fetch(PDO::FETCH_ASSOC);

    // Timing Attack Mitigation
    $dummy_hash = '$2y$10$abcdefghijklmnopqrstuv';
    $isValidOtp = false;

    if (!$admin_data) {
        password_verify($otp_code, $dummy_hash); // Equalize time
        $guard->logFailure($email, null, 'email');
        
        $newStatus = $guard->checkIpStatus();
        if (!$newStatus['allowed']) {
             sendResponse(false, null, $newStatus['message'], 429, $newStatus['code']);
        }
        sendResponse(false, null, 'Invalid OTP, or it has expired. Please try logging in again.', 401, 'OTP_INVALID_OR_EXPIRED');
    }
    
    // --- Verify OTP Hash ---
    if (!password_verify($otp_code, $admin_data['otp_code_hash'])) {
        $guard->logFailure($email, $admin_data['id'], 'otp');
        $newStatus = $guard->checkIpStatus();
        
        if (!$newStatus['allowed']) {
            $unblock_token = bin2hex(random_bytes(32));
            $unblock_token_hash = hash('sha256', $unblock_token);
            $unblock_expiry = gmdate('Y-m-d H:i:s', time() + 3600);
            $current_utc = gmdate('Y-m-d H:i:s');

            $block_level = ($newStatus['code'] === 'IP_PERMABANNED') ? 'permanent' : 'temporary';
            
            $insert_token_stmt = $conn->prepare("
                INSERT INTO admin_unblock_tokens 
                (admin_id, block_level, ip_address_blocked, ip_blind_index, token_hash, expires_at, created_at) 
                VALUES (:admin_id, :block_level, :ip_enc, :b_index, :token_hash, :expires_at, :created_at)
                ON DUPLICATE KEY UPDATE 
                block_level = VALUES(block_level),
                ip_address_blocked = VALUES(ip_address_blocked),
                token_hash = VALUES(token_hash),
                expires_at = VALUES(expires_at),
                created_at = VALUES(created_at)
            ");
            
            if ($insert_token_stmt->execute([
                ':admin_id' => $admin_data['id'],
                ':block_level' => $block_level,
                ':ip_enc' => $enc_ip,
                ':b_index' => $b_index,
                ':token_hash' => $unblock_token_hash,
                ':expires_at' => $unblock_expiry,
                ':created_at' => $current_utc
            ])) {
                $unlock_link = rtrim($_ENV['ADMIN_FRONTEND_URL'], '/') . "/#/admin/unlock-account/" . $unblock_token;
                
                $safe_ip = htmlspecialchars($ip_address, ENT_QUOTES, 'UTF-8');
                
                if ($block_level === 'permanent') {
                    $subject = 'CRITICAL: Account Permanently Blacklisted & Recovery Link';
                    $htmlBody = "<p>Hello {$admin_data['name']},</p><p>Due to repeated severe security violations (failed OTPs), your IP address (<strong>{$safe_ip}</strong>) has been <strong>permanently blacklisted</strong>.</p><p>If this was a mistake, you must verify your identity to revoke this permanent ban:</p><p><a href='{$unlock_link}'>Revoke Permanent Ban & Unblock My IP</a></p>";
                    $altBody = "Permanent Ban Unlock Link: {$unlock_link}";
                } else {
                    $subject = 'Account Access Alert & Recovery Link';
                    $htmlBody = "<p>Hello {$admin_data['name']},</p><p>Multiple failed OTP attempts were detected from your IP address: <strong>{$safe_ip}</strong>.</p><p>As a security measure, this IP has been temporarily blocked.</p><p>If this was you, click the link below to verify your identity and unblock your IP immediately:</p><p><a href='{$unlock_link}'>Unblock My IP Address</a></p>";
                    $altBody = "Unlock Link: {$unlock_link}";
                }
                sendSecurityEmail($admin_data['email'], $admin_data['name'], $subject, $htmlBody, $altBody);
            }
            sendResponse(false, null, $newStatus['message'], 429, $newStatus['code']);
        }
        sendResponse(false, null, 'Invalid OTP. Please try again.', 401, 'OTP_INVALID');
    }

    // --- OTP is VALID - Finalize Login ---
    $conn->beginTransaction();

    try {
        $settings_sql = "SELECT setting_value FROM settings WHERE setting_key = 'security_2fa_frequency'";
        $settings_stmt = $conn->query($settings_sql);
        $freq_row = $settings_stmt->fetch(PDO::FETCH_ASSOC);
        $frequency = $freq_row ? $freq_row['setting_value'] : 'daily';

        if ($frequency !== 'always') {
            $current_utc_time = gmdate('Y-m-d H:i:s');
            $insert_trust_stmt = $conn->prepare("
                INSERT INTO admin_trusted_devices (admin_id, ip_address, ip_blind_index, user_agent_hash, last_verified_at) 
                VALUES (:admin_id, :ip_enc, :b_index, :ua_hash, :verified_time) 
                ON DUPLICATE KEY UPDATE last_verified_at = :verified_time_update, ip_address = :ip_enc2
            ");
            $insert_trust_stmt->execute([
                ':admin_id' => $admin_data['id'],
                ':ip_enc' => $enc_ip,
                ':b_index' => $b_index,
                ':ua_hash' => $ua_hash,
                ':verified_time' => $current_utc_time,
                ':verified_time_update' => $current_utc_time,
                ':ip_enc2' => $enc_ip 
            ]);
        }

        $delete_otp_stmt = $conn->prepare("DELETE FROM admin_otp_verifications WHERE admin_id = :admin_id AND ip_blind_index = :b_index");
        $delete_otp_stmt->execute([':admin_id' => $admin_data['id'], ':b_index' => $b_index]);

        $jwtSecretKey = $_ENV['JWT_SECRET'];
        
        $session_jti = bin2hex(random_bytes(32));
        $access_jti = bin2hex(random_bytes(16));
        $refresh_jti = bin2hex(random_bytes(16));

        $access_payload = [
            'iat' => time(), 
            'iss' => $_SERVER['SERVER_NAME'], 
            'exp' => time() + 900, 
            'jti' => $access_jti,
            'session_jti' => $session_jti, 
            'data' => ['adminId' => $admin_data['id'], 'adminRole' => $admin_data['role']]
        ];
        $accessToken = JWT::encode($access_payload, $jwtSecretKey, 'HS256');

        $refresh_payload = [
            'iat' => time(),
            'iss' => $_SERVER['SERVER_NAME'],
            'exp' => time() + 43200, 
            'jti' => $refresh_jti,
            'session_jti' => $session_jti, 
            'data' => ['adminId' => $admin_data['id']]
        ];
        $refreshToken = JWT::encode($refresh_payload, $jwtSecretKey, 'HS256');

        // --- ENTERPRISE FIX: STANDARDIZED CROSS-ORIGIN COOKIES ---
        $cookieDomain = !empty($_ENV['COOKIE_DOMAIN']) ? trim($_ENV['COOKIE_DOMAIN']) : '';
        $isSecure = true; // Enforced for SameSite=None compatibility behind proxy

        setcookie('calcpro_access_token', $accessToken, [
            'expires' => time() + 900,
            'path' => '/',
            'domain' => $cookieDomain,
            'secure' => $isSecure,
            'httponly' => true,
            'samesite' => 'None'
        ]);

        setcookie('calcpro_refresh_token', $refreshToken, [
            'expires' => time() + 43200,
            'path' => '/',
            'domain' => $cookieDomain,
            'secure' => $isSecure,
            'httponly' => true,
            'samesite' => 'None'
        ]);

        $csrfPayload = bin2hex(random_bytes(32));
        $hmacSecret = base64_decode($_ENV['APP_ENCRYPTION_KEY'] ?? '');
        if (empty($hmacSecret)) {
            $hmacSecret = $_ENV['JWT_SECRET'] ?? 'fallback_secret';
        }
        $csrfSignature = hash_hmac('sha256', $csrfPayload, $hmacSecret);

        // Frontend readable payload 
        setcookie('calcpro_csrf_payload', $csrfPayload, [
            'expires' => time() + 43200, 
            'path' => '/',
            'domain' => $cookieDomain,
            'secure' => $isSecure,
            'httponly' => false, 
            'samesite' => 'None'
        ]);

        // Backend verified signature 
        setcookie('calcpro_csrf_token', $csrfSignature, [
            'expires' => time() + 43200, 
            'path' => '/',
            'domain' => $cookieDomain,
            'secure' => $isSecure,
            'httponly' => true, 
            'samesite' => 'None'
        ]);

        $update_login_stmt = $conn->prepare("UPDATE admins SET last_login = NOW(), current_session_jti = :session_jti WHERE id = :admin_id");
        $update_login_stmt->execute([
            ':session_jti' => $session_jti, 
            ':admin_id' => $admin_data['id']
        ]);

        $clear_specific_reset = $conn->prepare("DELETE FROM admin_password_resets WHERE admin_id = :admin_id AND ip_blind_index = :b_index");
        $clear_specific_reset->execute([':admin_id' => $admin_data['id'], ':b_index' => $b_index]);

        $clear_login_stmt = $conn->prepare("DELETE FROM login_attempts WHERE ip_blind_index = :b_index");
        $clear_login_stmt->execute([':b_index' => $b_index]);

        $clear_failures_stmt = $conn->prepare("DELETE FROM admin_login_failures WHERE ip_blind_index = :b_index");
        $clear_failures_stmt->execute([':b_index' => $b_index]);

        $clear_unblock_stmt = $conn->prepare("DELETE FROM admin_unblock_tokens WHERE admin_id = :admin_id AND ip_blind_index = :b_index");
        $clear_unblock_stmt->execute([':admin_id' => $admin_data['id'], ':b_index' => $b_index]);

        $conn->commit();
        
    } catch (Exception $e) {
        $conn->rollBack();
        throw $e;
    }

    $responseData = [
        'user' => [
            'id' => $admin_data['id'], 
            'name' => $admin_data['name'], 
            'email' => $admin_data['email'], 
            'role' => $admin_data['role']
        ]
    ];
    
    sendResponse(true, $responseData, 'Verification successful. Welcome back!', 200);

} catch (Exception $e) {
    error_log("Error in verify_otp.php: " . $e->getMessage());
    sendResponse(false, null, 'An internal server error occurred during OTP verification.', 500, 'SERVER_ERROR');
}

$conn = null;
?>
