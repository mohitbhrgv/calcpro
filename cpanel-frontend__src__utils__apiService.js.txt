/**
 * Custom Error class to carry HTTP status codes.
 * This allows components to differentiate between network failures,
 * business logic validations (4xx), and server crashes (5xx).
 */
export class ApiError extends Error {
  constructor(message, status) {
    super(message);
    this.name = "ApiError";
    this.status = status;
    // ENTERPRISE FIX: Use a hard boolean flag to bypass Babel/Vite transpilation issues
    // that sometimes break the 'instanceof' prototype chain.
    this.isBusinessLogicError = true; 
  }
}

// A function to handle the automatic logout process
const handleUnauthorized = () => {
  // ENTERPRISE FIX: Token is an HTTP-Only cookie managed by the browser.
  // We only need to clear the non-sensitive user state here.
  localStorage.removeItem("calcpro-admin-user");
  // Remove legacy token key just in case it was left over from previous versions
  localStorage.removeItem("calcpro-admin-token");
  // Redirect to the login page. The AuthContext and Router will handle the rest.
  window.location.hash = "/admin/login";
};

// --- CSRF PROTECTION ADDITION ---
// Securely extracts the RAW payload token generated by the PHP backend.
const getCsrfPayloadToken = () => {
  const match = document.cookie.match(new RegExp('(^|;\\s*)calcpro_csrf_payload=([^;]*)'));
  return match ? decodeURIComponent(match[2]) : null;
};

// --- NEW: CENTRALIZED ENVIRONMENT ROUTING ("THE PHONEBOOK") ---
const getFullUrl = (endpoint) => {
  if (endpoint.startsWith('http://') || endpoint.startsWith('https://')) {
    return endpoint;
  }
  
  const baseUrl = import.meta.env.VITE_API_BASE_URL || '';
  if (!baseUrl) {
    console.warn("[API Security] VITE_API_BASE_URL is not defined in the environment. Routing may fail.");
    return endpoint;
  }

  const cleanBase = baseUrl.replace(/\/+$/, '');
  const cleanEndpoint = endpoint.replace(/^\/+/, '');
  return `${cleanBase}/${cleanEndpoint}`;
};

/**
 * A centralized function for making authenticated API calls.
 * It automatically forces secure cookie inclusion, handles 401 errors, and dispatches an
 * activity event to reset the inactivity timer.
 *
 * @param {string} url The API endpoint URL to call.
 * @param {object} options The options for the fetch request (e.g., method, body).
 * @returns {Promise<object>} The JSON response from the API.
 */
export const authenticatedFetch = async (url, options = {}) => {
  const method = (options.method || 'GET').toUpperCase();
  const isStateChanging = ['POST', 'PUT', 'DELETE', 'PATCH'].includes(method);
  
  // Automatically grab the CSRF token only for state-changing operations
  const csrfToken = isStateChanging ? getCsrfPayloadToken() : null;

  if (isStateChanging && !csrfToken) {
      console.warn("[Security] CSRF token not found in storage.");
  }

  // Prepare the headers, merging any custom headers with our standard ones
  const headers = {
    "Content-Type": "application/json",
    ...(csrfToken ? { "X-CSRF-Token": csrfToken } : {}),
    ...options.headers, 
  };

  const secureUrl = getFullUrl(url);

  try {
    // ENTERPRISE FIX: 'credentials: "include"' strictly enforces the transmission 
    // of secure HTTP-Only cookies.
    // No manual JWT header injection required.
    const response = await fetch(secureUrl, { ...options, headers, credentials: "include" });

    // --- Phase 2: Global Watchdog (Fail-Safe Interceptor) ---
    if (response.status === 503) {
      let incidentCode = 'CRITICAL_SYSTEM_FAILURE';
      try {
          const errorData = await response.clone().json();
          if (errorData && errorData.error_code) incidentCode = errorData.error_code;
      } catch (e) {
          // [ENTERPRISE FIX] Satisfy ESLint no-empty & ensure SOC 2 Observability
          console.debug("[API] 503 payload is not valid JSON, using fallback incident code.");
      }
      // Broadcast the silent offline alarm to the root component
      window.dispatchEvent(new CustomEvent("SYSTEM_OFFLINE", { detail: incidentCode }));
      // Freeze this thread to prevent localized toast errors while the app unmounts
      return new Promise(() => {});
    }

    // If the server responds with 401 Unauthorized, the session cookie is invalid or expired.
    if (response.status === 401) {
      handleUnauthorized();
      throw new ApiError("Unauthorized Session", 401);
    }

    if (!response.ok) {
      let errorMessage = `HTTP error! status: ${response.status}`;
      try {
         const errorResult = await response.json();
         errorMessage = errorResult.message || errorMessage;
      } catch (parseError) {
         // [ENTERPRISE FIX] Satisfy ESLint no-empty & ensure SOC 2 Observability
         console.debug("[API] Non-JSON error payload received:", parseError.message);
      }
      
      throw new ApiError(errorMessage, response.status);
    }

    // Dispatch a custom event to signal user activity to AuthContext.
    window.dispatchEvent(new CustomEvent("user-activity"));

    // Handle 204 No Content responses safely
    if (response.status === 204) {
        return null;
    }

    return await response.json();
  } catch (error) {
    // Expected client business logic errors (400, 403, 404, 409, 429) will bypass the console
    if (!error.isBusinessLogicError || error.status >= 500) {
      console.error(`[API System Fault] ${options.method || 'GET'} ${secureUrl} :`, error.message);
    }
    
    throw error;
  }
};
