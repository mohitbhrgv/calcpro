import React from "react";

  FiLayers,

                  globalSettings={globalSettings}

                            Max Video Preview (sec)

                                (Forced by "No Index")

                                (Disabled by "No Snippet")
                              </span>

                            value={
                              isRobotsDefault
                                ? globalSettings?.seo_robots_global_max_video_preview ?? -1

                              <span className="ml-1 text-[10px] font-normal italic text-amber-600 dark:text-amber-400">
                                (Forced by "No Index")
                              </span>

                            disabled={formData.no_index || formData.no_snippet || formData.no_image_index}

                      {/* FIX: Explicit background and text color added for light mode */}
                      <input
                        type="text"
                        name="canonical_url"
                        value={formData.canonical_url || ""}
                        onChange={handleChange}
                        placeholder="https://example.com/original-post"
                        className="w-full border border-gray-300 dark:border-gray-600 rounded-lg pl-8 pr-3 py-2 text-xs bg-white text-gray-900 placeholder-gray-400 dark:bg-gray-700 dark:text-white focus:ring-2 focus:ring-blue-500 focus:border-transparent outline-none transition-shadow"

        <button
          onClick={() => toggleSection("tags")}
          className="w-full flex items-center justify-between p-4 bg-gray-50 dark:bg-gray-700/50 border-b border-gray-200 dark:border-gray-700 rounded-t-xl hover:bg-gray-100 dark:hover:bg-gray-600/50 transition-colors focus:outline-none"

          <h3 className="font-semibold text-sm uppercase tracking-wide text-gray-700 dark:text-gray-300">
            Tags

            icon={expandedSections.featuredImage ? FiChevronUp : FiChevronDown}
            className="w-4 h-4 text-gray-500"

                    Replace

                    Remove

            ) : (
              <button
                onClick={() => setIsFeaturedMediaLibraryOpen(true)}
                className="flex flex-col items-center justify-center w-full h-36 border-2 border-dashed border-gray-300 dark:border-gray-600 rounded-xl hover:bg-gray-50 dark:hover:bg-gray-700/50 hover:border-blue-400 transition-all group cursor-pointer focus:outline-none focus:ring-2 focus:ring-blue-500"
              >
                <div className="p-3 bg-gray-100 dark:bg-gray-700 rounded-full group-hover:bg-blue-50 dark:group-hover:bg-blue-900/20 transition-colors">
                  <SafeIcon
                    icon={FiImage}
                    className="w-6 h-6 text-gray-400 group-hover:text-blue-500"
                  />
                </div>
                <span className="text-xs font-medium text-gray-500 mt-2 group-hover:text-blue-600">
                  Set Featured Image
                </span>
              </button>
            )}
          </div>
        )}
      </div>

    </div>
  );
};

export default PostSidebar;
