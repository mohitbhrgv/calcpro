<?php
// File: calcpro-api/api/admin/request_admin_password_reset.php

use PHPMailer\PHPMailer\PHPMailer;
use PHPMailer\PHPMailer\Exception;

require_once dirname(__DIR__, 2) . '/core/config.php';
require_once dirname(__DIR__, 2) . '/core/response.php';
require_once dirname(__DIR__, 2) . '/core/SecurityGuard.php';

try {
    $start_time = microtime(true);

    $ip_address = getSecureRealIp();
    $b_index = SecurityGuard::generateBlindIndex($ip_address);

    $json = file_get_contents('php://input');
    $data = json_decode($json);

    if (json_last_error() !== JSON_ERROR_NONE || !is_object($data)) {
        sendResponse(false, null, 'Invalid request payload format.', 400, 'INVALID_PAYLOAD');
    }

    $email_input = isset($data->email) ? (string)$data->email : '';

    if (empty($email_input) || !filter_var(trim($email_input), FILTER_VALIDATE_EMAIL)) {
        sendResponse(false, null, 'A valid email address is required.', 400, 'INVALID_INPUT');
    }

    $email = trim($email_input);

    $guard = new SecurityGuard($conn, $ip_address);
    $securityStatus = $guard->checkIpStatus();
    if (!$securityStatus['allowed']) {
        sendResponse(false, null, $securityStatus['message'], 403, $securityStatus['code']);
    }

    $sql_find = "SELECT id, name, email FROM admins WHERE email = :email";
    $stmt_find = $conn->prepare($sql_find);
    $stmt_find->execute([':email' => $email]);
    $admin = $stmt_find->fetch(PDO::FETCH_ASSOC);
    
    if ($admin) {
        $admin_id = $admin['id'];
        $current_utc_time = gmdate('Y-m-d H:i:s');
        
        $check_sql = "SELECT hits, last_request_at FROM admin_password_resets WHERE admin_id = :admin_id AND ip_blind_index = :b_index";
        $check_stmt = $conn->prepare($check_sql);
        $check_stmt->execute([':admin_id' => $admin_id, ':b_index' => $b_index]);
        $existing_log = $check_stmt->fetch(PDO::FETCH_ASSOC);

        $should_process = true;
        $new_hits = 1;

        if ($existing_log) {
            $last_request = strtotime($existing_log['last_request_at'] . ' UTC');
            $time_diff_hours = (time() - $last_request) / 3600;

            if ($time_diff_hours >= 24) {
                $new_hits = 1;
            } else {
                if ($existing_log['hits'] >= 3) {
                    $should_process = false;
                } else {
                    $new_hits = $existing_log['hits'] + 1;
                }
            }
        }

        if ($should_process) {
            $token = bin2hex(random_bytes(32));
            $token_hash = hash('sha256', $token);
            $expiry_date = gmdate('Y-m-d H:i:s', time() + 600); 

            $upsert_sql = "
                INSERT INTO admin_password_resets (admin_id, ip_blind_index, token_hash, hits, expires_at, last_request_at) 
                VALUES (:admin_id, :b_index, :token_hash, :hits, :expires_at, :last_req)
                ON DUPLICATE KEY UPDATE 
                    token_hash = VALUES(token_hash), 
                    hits = VALUES(hits), 
                    expires_at = VALUES(expires_at), 
                    last_request_at = VALUES(last_request_at)
            ";
            $upsert_stmt = $conn->prepare($upsert_sql);
            $upsert_stmt->execute([
                ':admin_id' => $admin_id,
                ':b_index' => $b_index,
                ':token_hash' => $token_hash,
                ':hits' => $new_hits,
                ':expires_at' => $expiry_date,
                ':last_req' => $current_utc_time
            ]);

            $reset_link = rtrim($_ENV['ADMIN_FRONTEND_URL'], '/') . "/reset-password/" . $token;
            $safe_name = htmlspecialchars($admin['name'], ENT_QUOTES, 'UTF-8');
            $safe_ip = htmlspecialchars($ip_address, ENT_QUOTES, 'UTF-8');
            $app_env = ucfirst($_ENV['APP_ENV'] ?? 'Production');
            
            $subject = '{{SITE_NAME}} Password Reset Request';
            
            $htmlBody = [
                'preheader'     => 'Secure Password Reset Link',
                'event_type'    => 'CREDENTIAL RECOVERY',
                'greeting'      => "Hello {$safe_name},",
                'alert_type'    => 'info',
                'alert_message' => 'A password reset was requested for your administrator account.',
                'body'          => "<p>We received a request to reset the password for your administrator account. You can set a new password using the secure button below. This link is valid for exactly <strong>10 minutes</strong>.</p>
                                    <p><strong>If you didn't request this</strong>, you can safely ignore this email. Your current password will remain unchanged and your account is safe.</p>",
                'meta_data'     => [
                    'Triggered By' => $admin['email'],
                    'Environment'  => $app_env,
                    'IP Address'   => $safe_ip
                ],
                'action_url'    => $reset_link,
                'action_text'   => 'Reset Your Password'
            ];

            $altBody = "Hello {$safe_name},\n\nYou requested a password reset for your admin account. Copy and paste this URL into your browser to proceed: {$reset_link}\n\nThis link is highly secure and valid for exactly 10 minutes.";

            if (!sendSecurityEmail($admin['email'], $admin['name'], $subject, $htmlBody, $altBody)) {
                error_log("Failed to send password reset email to {$admin['email']}.");
            }
        }
    }
    
    $execution_time = microtime(true) - $start_time;
    $target_time = random_int(1500, 2500) / 1000; 

    if ($execution_time < $target_time) {
        $sleep_microseconds = (int)(($target_time - $execution_time) * 1000000);
        usleep($sleep_microseconds);
    }
    
    sendResponse(true, null, 'If an account with that email exists, a password reset link has been sent.');

} catch (Exception $e) {
    error_log("Error in request_admin_password_reset.php: " . $e->getMessage());
    sendResponse(false, null, 'An internal server error occurred.', 500, 'SERVER_ERROR');
}

$conn = null;
?>
