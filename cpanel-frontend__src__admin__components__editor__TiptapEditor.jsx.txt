// --- GLOBAL SMART URL FORMATTER ---
const formatSmartUrl = (url) => {
  if (!url) return '';
  const trimmed = url.trim();
  if (
    /^https?:\/\//i.test(trimmed) || 
    trimmed.startsWith('mailto:') || 
    trimmed.startsWith('tel:') || 
    trimmed.startsWith('/') || 
    trimmed.startsWith('#')
  ) {
    return trimmed;
  }
  return `https://${trimmed}`;
};

const CtaButton = Node.create({
  name: 'ctaButton',
  group: 'block',
  atom: true,
  addAttributes() {
    return {
      text: { default: 'Click Here' },
      href: { default: '' },
      bgColor: { default: '#4f46e5' },
      textColor: { default: '#ffffff' },
      hoverBgColor: { default: '#4338ca' },
      hoverTextColor: { default: '#ffffff' },
      align: { default: 'center' },
      openInNewTab: { default: true },
      nofollow: { default: false },
      sponsored: { default: false },
      paddingX: { default: 32 },
      paddingY: { default: 14 },
      hoverEffect: { default: 'none' },
      isBold: { default: 'true' },
      isItalic: { default: 'false' },
      isUnderline: { default: 'false' }
    };
  },
  parseHTML() {
    return [{
      tag: 'div[data-type="cta-wrapper"]',
      getAttrs: dom => {
        const a = dom.querySelector('a[data-type="cta-button"]');
        if (!a) return false;
        const rel = a.getAttribute('rel') || '';
        
        const pxRaw = a.getAttribute('data-px') || '32';
        const pyRaw = a.getAttribute('data-py') || '14';
        const px = pxRaw.includes('rem') ? parseFloat(pxRaw) * 16 : parseInt(pxRaw, 10);
        const py = pyRaw.includes('rem') ? parseFloat(pyRaw) * 16 : parseInt(pyRaw, 10);

        return {
          align: dom.getAttribute('data-align') || 'center',
          href: a.getAttribute('href') || '',
          text: a.innerText || 'Click Here',
          bgColor: a.getAttribute('data-bg') || '#4f46e5',
          textColor: a.getAttribute('data-text') || '#ffffff',
          hoverBgColor: a.getAttribute('data-hover-bg') || '#4338ca',
          hoverTextColor: a.getAttribute('data-hover-text') || '#ffffff',
          openInNewTab: a.getAttribute('target') === '_blank',
          nofollow: rel.includes('nofollow'),
          sponsored: rel.includes('sponsored'),
          paddingX: isNaN(px) ? 32 : px,
          paddingY: isNaN(py) ? 14 : py,
          hoverEffect: a.getAttribute('data-hover-effect') || 'none',
          isBold: a.getAttribute('data-bold') || 'true',
          isItalic: a.getAttribute('data-italic') || 'false',
          isUnderline: a.getAttribute('data-underline') || 'false'
        };
      }
    }];
  },
  renderHTML({ node }) {
    const { 
       text, href, bgColor, textColor, hoverBgColor, hoverTextColor, align, 
       openInNewTab, nofollow, sponsored, paddingX, paddingY, hoverEffect,
       isBold, isItalic, isUnderline
    } = node.attrs;

    const justifyClass = align === 'center' ? 'justify-center' : align === 'right' ? 'justify-end' : 'justify-start';

    let rels = ['noopener', 'noreferrer'];
    if (nofollow) rels.push('nofollow');
    if (sponsored) rels.push('sponsored');

    return [
      'div',
      {
        class: `flex w-full my-10 ${justifyClass}`,
        'data-type': 'cta-wrapper',
        'data-align': align
      },
      [
        'a',
        {
          'data-type': 'cta-button',
          href: href || '#',
          class: 'inline-block rounded-xl shadow-lg transition-all duration-200 cursor-pointer text-center no-underline',
          target: openInNewTab ? '_blank' : '_self',
          rel: rels.join(' '),
          'data-bg': bgColor,
          'data-text': textColor,
          'data-hover-bg': hoverBgColor,
          'data-hover-text': hoverTextColor,
          'data-px': paddingX,
          'data-py': paddingY,
          'data-hover-effect': hoverEffect,
          'data-bold': isBold,
          'data-italic': isItalic,
          'data-underline': isUnderline,
          style: `--btn-bg: ${bgColor}; --btn-color: ${textColor}; --btn-hover-bg: ${hoverBgColor}; --btn-hover-color: ${hoverTextColor}; background-color: var(--btn-bg); color: var(--btn-color); padding: ${paddingY}px ${paddingX}px; width: auto;`
        },
        text
      ]
    ];
  },
  addNodeView() {
    return ReactNodeViewRenderer(CtaButtonComponent);
  }
});

const RawHtmlComponent = ({ node, updateAttributes, selected, deleteNode }) => {
  const [isEditing, setIsEditing] = useState(!node.attrs.html);

  const stopPropagation = useCallback((e) => {
    e.stopPropagation();
    e.nativeEvent?.stopImmediatePropagation();
  }, []);

  useEffect(() => { if (!selected && node.attrs.html) setIsEditing(false); }, [selected, node.attrs.html]);

  const hasTargetBlank = node.attrs.html?.includes('target="_blank"') || false;
  const hasNofollow = node.attrs.html?.includes('nofollow') || false;
  const hasSponsored = node.attrs.html?.includes('sponsored') || false;

  const applySeoMacro = (config) => {
    const tempDiv = document.createElement('div');
    tempDiv.innerHTML = node.attrs.html || '';
    const links = tempDiv.querySelectorAll('a');
    
    links.forEach(a => {
        if (config.openInNewTab) a.setAttribute('target', '_blank');
        else a.removeAttribute('target');
        
        let rels = ['noopener', 'noreferrer'];
        if (config.nofollow) rels.push('nofollow');
        if (config.sponsored) rels.push('sponsored');
        
        if (rels.length > 2 || config.nofollow || config.sponsored) {
            a.setAttribute('rel', rels.join(' '));
        } else {
            a.setAttribute('rel', 'noopener noreferrer');
        }
    });
    
    updateAttributes({ html: tempDiv.innerHTML });
  };

  return (
    <NodeViewWrapper className={`relative w-full mt-10 mb-8 border-2 border-dashed rounded-xl transition-all bg-gray-50 dark:bg-gray-800/50 ${selected ? 'border-amber-400 ring-4 ring-amber-400/20 shadow-lg' : 'border-gray-300 dark:border-gray-600'}`}>

const CustomImage = Image.extend({
  addAttributes() {
    return {
      ...this.parent?.(),
      'data-media-id': { default: null },
      width: { default: null },
      height: { default: null },
      href: { default: '' },
      openInNewTab: { default: true },
      nofollow: { default: false },
      sponsored: { default: false }
    }
  },
  addStorage() {
    return {
      imageOptions: null
    };
  },
  parseHTML() {
    return [
      {
        tag: 'a[href] img',
        getAttrs: dom => {
          const a = dom.closest('a');
          const rel = a.getAttribute('rel') || '';
          return {
            href: a.getAttribute('href'),
            openInNewTab: a.getAttribute('target') === '_blank',
            nofollow: rel.includes('nofollow'),
            sponsored: rel.includes('sponsored')
          };
        }
      },
      {
        tag: 'img[src]'
      }
    ];
  },
  renderHTML({ HTMLAttributes }) {
    const { href, openInNewTab, nofollow, sponsored, ...imgAttrs } = HTMLAttributes;
    const img = ['img', mergeAttributes(this.options.HTMLAttributes, imgAttrs)];
    
    if (href && href.trim() !== '') {
      let rels = ['noopener', 'noreferrer'];
      if (nofollow) rels.push('nofollow');
      if (sponsored) rels.push('sponsored');
      return ['a', { href, target: openInNewTab ? '_blank' : '_self', rel: rels.join(' ') }, img];
    }
    return img;
  },
  addNodeView() {
    return ReactNodeViewRenderer(ImageNodeComponent);
  }
});

        },
       },
    };
  },

const MenuBar = ({ editor, onRequestMedia, wrapperRef }) => {
  const [activeDropdown, setActiveDropdown] = useState(null);
  const [activePopover, setActivePopover] = useState(null);
  
  const [isSticky, setIsSticky] = useState(false);
  const [stickyWidth, setStickyWidth] = useState('auto');
  const toolbarRef = useRef(null);

  const [linkInput, setLinkInput] = useState('');
  const [linkTarget, setLinkTarget] = useState(true);
  const [linkNofollow, setLinkNofollow] = useState(false);
  const [linkSponsored, setLinkSponsored] = useState(false);

  useEffect(() => {
    const scrollParent = wrapperRef.current?.closest('.overflow-y-auto') || wrapperRef.current?.closest('.custom-scrollbar') || window;
    
    const handleScroll = () => {
        if (!wrapperRef.current) return;
        
        const rect = wrapperRef.current.getBoundingClientRect();
        const headerOffset = 64; 

        if (rect.top <= headerOffset && rect.bottom > 250) {
            setIsSticky(true);
            setStickyWidth(wrapperRef.current.offsetWidth);
        } else {
            setIsSticky(false);
            setStickyWidth('auto');
        }
    };

    const target = scrollParent === document.documentElement ? window : scrollParent;
    target.addEventListener('scroll', handleScroll, { passive: true });
    window.addEventListener('resize', handleScroll);
    
    handleScroll(); 

    return () => {
        target.removeEventListener('scroll', handleScroll);
        window.removeEventListener('resize', handleScroll);
    };
  }, [wrapperRef]);

  useEffect(() => {
    const handleClickOutside = (event) => {
      if (toolbarRef.current && !toolbarRef.current.contains(event.target)) {
        setActiveDropdown(null);
        setActivePopover(null);
      }
    };
    document.addEventListener("mousedown", handleClickOutside);
    return () => document.removeEventListener("mousedown", handleClickOutside);
  }, []);

  const [, setTick] = useState(0);
  useEffect(() => {
    if (!editor) return;
    const updateRender = () => setTick(t => t + 1);
    editor.on('transaction', updateRender);
    editor.on('selectionUpdate', updateRender);
    return () => {
      editor.off('transaction', updateRender);
      editor.off('selectionUpdate', updateRender);
    };
  }, [editor]);

  const toggleDropdown = (name) => {
    setActivePopover(null);
    setActiveDropdown(prev => prev === name ? null : name);
  };

  const closeDropdowns = () => {
    setActiveDropdown(null);
    setActivePopover(null);
  };

  if (!editor) return null;

  const handleOpenLinkPopover = () => {
    const attrs = editor.getAttributes('link');
    setLinkInput(attrs.href || '');
    setLinkTarget(attrs.target !== '_self');
    setLinkNofollow(attrs.rel?.includes('nofollow') || false);
    setLinkSponsored(attrs.rel?.includes('sponsored') || false);
    setActiveDropdown(null);
    setActivePopover('link');
  };

  const handleApplyLink = () => {
    if (linkInput === '') {
      editor.chain().focus().extendMarkRange('link').unsetLink().run();
    } else {
      let rels = ['noopener', 'noreferrer'];
      if (linkNofollow) rels.push('nofollow');
      if (linkSponsored) rels.push('sponsored');

      const finalUrl = formatSmartUrl(linkInput);

      editor.chain().focus().extendMarkRange('link').setLink({ 
        href: finalUrl,
        target: linkTarget ? '_blank' : '_self',
        rel: rels.join(' ')
      }).run();
    }
    setActivePopover(null);
  };

  const insertTable = () => {
    editor.chain().focus().insertTable({ rows: 3, cols: 3, withHeaderRow: true }).run();
    closeDropdowns();
  };

  const getHeadingLabel = () => {
    if (editor.isActive('heading', { level: 1 })) return 'H1';
    if (editor.isActive('heading', { level: 2 })) return 'H2';
    if (editor.isActive('heading', { level: 3 })) return 'H3';
    if (editor.isActive('heading', { level: 4 })) return 'H4';
    if (editor.isActive('heading', { level: 5 })) return 'H5';
    if (editor.isActive('heading', { level: 6 })) return 'H6';
    return 'Text';
  };

  const getAlignmentIcon = () => {
    if (editor.isActive({ textAlign: 'center' })) return AlignCenter;
    if (editor.isActive({ textAlign: 'right' })) return AlignRight;
    if (editor.isActive({ textAlign: 'justify' })) return AlignJustify;
    return AlignLeft; 
  };

  const ActiveAlignIcon = getAlignmentIcon();

  const btnClass = "p-1.5 rounded-md text-gray-600 dark:text-gray-400 hover:bg-gray-200 dark:hover:bg-gray-700 hover:text-gray-900 dark:hover:text-white transition-colors shrink-0 flex items-center justify-center relative";
  const activeBtnClass = "p-1.5 rounded-md bg-indigo-100 dark:bg-indigo-900/60 text-indigo-700 dark:text-indigo-300 transition-colors shrink-0 flex items-center justify-center relative shadow-inner";
  const divider = <div className="w-px h-5 bg-gray-300 dark:bg-gray-600 mx-1 shrink-0" />;
  const dropdownItemClass = "w-full text-left px-3 py-1.5 text-sm text-gray-700 dark:text-gray-300 hover:bg-gray-100 dark:hover:bg-gray-700 hover:text-gray-900 dark:hover:text-white flex items-center gap-2 transition-colors";
  const inputClass = "w-full text-sm px-3 py-2 border border-gray-300 dark:border-gray-600 rounded-lg bg-white dark:bg-gray-900 text-gray-900 dark:text-gray-100 focus:ring-2 focus:ring-indigo-500 outline-none transition-all";

  const currentColor = editor.getAttributes('textStyle').color || '#000000';
  const currentHighlight = editor.getAttributes('highlight').color || '#ffff00';

  const stickyStyle = isSticky ? {
      position: 'fixed',
      top: '64px',
      width: `${stickyWidth}px`,
      zIndex: 45,
      borderTopLeftRadius: 0,
      borderTopRightRadius: 0,
      borderBottomLeftRadius: '0.5rem',
      borderBottomRightRadius: '0.5rem',
      boxShadow: '0 10px 15px -3px rgba(0, 0, 0, 0.1), 0 4px 6px -2px rgba(0, 0, 0, 0.05)'
  } : {};

  return (
    <>
      {isSticky && <div style={{ height: toolbarRef.current?.offsetHeight || 50 }} className="w-full" />}
      
      <div 
        ref={toolbarRef} 
        style={stickyStyle}
        className={`flex flex-wrap items-center gap-0.5 p-2 border-b border-gray-200 dark:border-gray-700 bg-gray-50/95 dark:bg-gray-800/95 backdrop-blur-md transition-all duration-75 ${!isSticky ? 'rounded-t-xl z-[40]' : ''}`}
      >
        <button onClick={() => editor.chain().focus().undo().run()} disabled={!editor.can().undo()} className={`${btnClass} disabled:opacity-30`} title="Undo"><Undo size={16} /></button>
        <button onClick={() => editor.chain().focus().redo().run()} disabled={!editor.can().redo()} className={`${btnClass} disabled:opacity-30`} title="Redo"><Redo size={16} /></button>

        {divider}

        <Dropdown 
          label={getHeadingLabel()} 
          isOpen={activeDropdown === 'heading'} 
          onToggle={() => toggleDropdown('heading')} 
          active={editor.isActive('heading')}
          disabled={editor.isActive('table')}
        >
          <button onClick={() => { editor.chain().focus().setParagraph().run(); closeDropdowns(); }} className={dropdownItemClass}>
            <Type size={16} className="opacity-50" /> Normal Text
          </button>

const TableToolbar = ({ editor }) => {
  const [activeDropdown, setActiveDropdown] = useState(null);
  const toolbarRef = useRef(null);

  const [, setTick] = useState(0);
  useEffect(() => {
    if (!editor) return;
    const updateRender = () => setTick(t => t + 1);
    editor.on('transaction', updateRender);
    editor.on('selectionUpdate', updateRender);
    return () => {
        editor.off('transaction', updateRender);
        editor.off('selectionUpdate', updateRender);
    };
  }, [editor]);

  useEffect(() => {
    const handleClickOutside = (event) => {
      if (toolbarRef.current && !toolbarRef.current.contains(event.target)) {
        setActiveDropdown(null);
      }
    };
    document.addEventListener("mousedown", handleClickOutside);
    return () => document.removeEventListener("mousedown", handleClickOutside);
  }, []);

  if (!editor) return null;
  
  const btnClass = "p-1.5 rounded-md text-gray-600 dark:text-gray-300 hover:bg-gray-100 dark:hover:bg-gray-700 hover:text-gray-900 dark:hover:text-white transition-colors flex items-center justify-center disabled:opacity-40 disabled:cursor-not-allowed disabled:hover:bg-transparent";
  const activeBtnClass = "p-1.5 rounded-md bg-indigo-50 dark:bg-indigo-900/40 text-indigo-700 dark:text-indigo-300 transition-colors flex items-center justify-center disabled:opacity-40 disabled:cursor-not-allowed disabled:hover:bg-transparent";
  const divider = <div className="w-px h-4 bg-gray-200 dark:bg-gray-700 mx-1 shrink-0" />;
  const dropdownItemClass = "w-full text-left px-3 py-2 text-sm text-gray-700 dark:text-gray-300 hover:bg-gray-50 dark:hover:bg-gray-700 hover:text-gray-900 dark:hover:text-white flex items-center gap-2 transition-colors";

  const getAlignmentIcon = () => {
    if (editor.isActive({ textAlign: 'center' })) return AlignCenter;
    if (editor.isActive({ textAlign: 'right' })) return AlignRight;
    if (editor.isActive({ textAlign: 'justify' })) return AlignJustify;
    return AlignLeft; 
  };

  const ActiveAlignIcon = getAlignmentIcon();
  const isHeaderCell = editor.isActive('tableHeader');

  // ACCURATE HEADER STATE DETECTION: Toggle Header Row, Toggle Header Column, and
  // Toggle Header Cell are three distinct actions and must reflect three distinct
  // active-states. A row/column is only "active" if EVERY cell within it is a
  // tableHeader node — checking just the current cell (as before) falsely marked
  // all three buttons active whenever any one of them had been toggled.
  const getTableHeaderRowColumnState = () => {
    try {
      const { $anchor } = editor.state.selection;
      let rowDepth = null;
      let tableDepth = null;

      for (let d = $anchor.depth; d > 0; d--) {
        const node = $anchor.node(d);
        if (node.type.name === 'tableRow' && rowDepth === null) {
          rowDepth = d;
        }
        if (node.type.name === 'table' && tableDepth === null) {
          tableDepth = d;
          break;
        }
      }

      if (rowDepth === null || tableDepth === null) {
        return { isHeaderRow: false, isHeaderColumn: false };
      }

      const rowNode = $anchor.node(rowDepth);
      const tableNode = $anchor.node(tableDepth);
      const cellOffsetInRow = $anchor.pos - $anchor.start(rowDepth);

      let cellIndex = -1;
      rowNode.forEach((cellNode, offset, index) => {
        if (cellIndex === -1 && offset <= cellOffsetInRow && cellOffsetInRow < offset + cellNode.nodeSize) {
          cellIndex = index;
        }
      });

      if (cellIndex === -1) {
        return { isHeaderRow: false, isHeaderColumn: false };
      }

      let isHeaderRow = rowNode.childCount > 0;
      rowNode.forEach((cellNode) => {
        if (cellNode.type.name !== 'tableHeader') isHeaderRow = false;
      });

      let isHeaderColumn = tableNode.childCount > 0;
      tableNode.forEach((currentRowNode) => {
        const columnCellIndex = cellIndex < currentRowNode.childCount ? cellIndex : currentRowNode.childCount - 1;
        const columnCellNode = currentRowNode.child(columnCellIndex);
        if (!columnCellNode || columnCellNode.type.name !== 'tableHeader') isHeaderColumn = false;
      });

      return { isHeaderRow, isHeaderColumn };
    } catch (e) {
      return { isHeaderRow: false, isHeaderColumn: false };
    }
  };

  const { isHeaderRow, isHeaderColumn } = getTableHeaderRowColumnState();

  return (
    <div ref={toolbarRef} className="flex items-center gap-0.5 bg-white dark:bg-gray-800 border border-gray-200 dark:border-gray-700 shadow-xl rounded-lg p-1 animate-in fade-in zoom-in-95 duration-100">
      <Dropdown icon={LayoutGrid} isOpen={activeDropdown === 'layout'} onToggle={() => setActiveDropdown(prev => prev === 'layout' ? null : 'layout')}>
        <div className="py-1">
            <span className="block px-3 py-1 text-[10px] font-bold text-gray-400 uppercase tracking-wider">Columns</span>
            <button onClick={() => { editor.chain().focus().addColumnBefore().run(); setActiveDropdown(null); }} className={dropdownItemClass}><Columns size={14} className="opacity-60" /> Add Column Before</button>
            <button onClick={() => { editor.chain().focus().addColumnAfter().run(); setActiveDropdown(null); }} className={dropdownItemClass}><Columns size={14} className="opacity-60" /> Add Column After</button>
            <button onClick={() => { editor.chain().focus().deleteColumn().run(); setActiveDropdown(null); }} className={`${dropdownItemClass} text-red-600 dark:text-red-400 hover:text-red-700`}><Trash2 size={14} className="opacity-60" /> Delete Column</button>

            <div className="my-1 border-t border-gray-100 dark:border-gray-700" />
            
            <span className="block px-3 py-1 text-[10px] font-bold text-gray-400 uppercase tracking-wider">Rows</span>
            <button onClick={() => { editor.chain().focus().addRowBefore().run(); setActiveDropdown(null); }} className={dropdownItemClass}><Rows size={14} className="opacity-60" /> Add Row Before</button>
            <button onClick={() => { editor.chain().focus().addRowAfter().run(); setActiveDropdown(null); }} className={dropdownItemClass}><Rows size={14} className="opacity-60" /> Add Row After</button>
            <button onClick={() => { editor.chain().focus().deleteRow().run(); setActiveDropdown(null); }} className={`${dropdownItemClass} text-red-600 dark:text-red-400 hover:text-red-700`}><Trash2 size={14} className="opacity-60" /> Delete Row</button>
        </div>
      </Dropdown>

      {divider}

      <Dropdown icon={ActiveAlignIcon} isOpen={activeDropdown === 'align'} onToggle={() => setActiveDropdown(prev => prev === 'align' ? null : 'align')} active={editor.isActive({ textAlign: 'center' }) || editor.isActive({ textAlign: 'right' }) || editor.isActive({ textAlign: 'justify' })}>
        <button onClick={() => { editor.chain().focus().setTextAlign('left').run(); setActiveDropdown(null); }} className={`${dropdownItemClass} ${editor.isActive({ textAlign: 'left' }) ? 'bg-indigo-50/50 dark:bg-indigo-900/30 text-indigo-600 dark:text-indigo-400' : ''}`}><AlignLeft size={16} className="opacity-60" /> Align Left</button>
        <button onClick={() => { editor.chain().focus().setTextAlign('center').run(); setActiveDropdown(null); }} className={`${dropdownItemClass} ${editor.isActive({ textAlign: 'center' }) ? 'bg-indigo-50/50 dark:bg-indigo-900/30 text-indigo-600 dark:text-indigo-400' : ''}`}><AlignCenter size={16} className="opacity-60" /> Align Center</button>
        <button onClick={() => { editor.chain().focus().setTextAlign('right').run(); setActiveDropdown(null); }} className={`${dropdownItemClass} ${editor.isActive({ textAlign: 'right' }) ? 'bg-indigo-50/50 dark:bg-indigo-900/30 text-indigo-600 dark:text-indigo-400' : ''}`}><AlignRight size={16} className="opacity-60" /> Align Right</button>
      </Dropdown>

      {divider}
      
      <button onClick={() => editor.chain().focus().mergeCells().run()} disabled={!editor.can().mergeCells()} className={btnClass} title="Merge Cells">
        <Combine size={14} className={editor.can().mergeCells() ? "text-blue-500 dark:text-blue-400" : ""} />
      </button>
      <button onClick={() => editor.chain().focus().splitCell().run()} disabled={!editor.can().splitCell()} className={btnClass} title="Split Cell">
        <Split size={14} className={editor.can().splitCell() ? "text-blue-500 dark:text-blue-400" : ""} />
      </button>

      {divider}

      <button onClick={() => editor.chain().focus().toggleHeaderRow().run()} className={isHeaderRow ? activeBtnClass : btnClass} title="Toggle Header Row">
        <PanelTop size={14} className={isHeaderRow ? "text-indigo-600 dark:text-indigo-400" : "opacity-60"} />
      </button>
      <button onClick={() => editor.chain().focus().toggleHeaderColumn().run()} className={isHeaderColumn ? activeBtnClass : btnClass} title="Toggle Header Column">
        <PanelLeft size={14} className={isHeaderColumn ? "text-indigo-600 dark:text-indigo-400" : "opacity-60"} />
      </button>
      <button onClick={() => editor.chain().focus().toggleHeaderCell().run()} className={isHeaderCell ? activeBtnClass : btnClass} title="Toggle Header Cell">
        <Type size={14} className={isHeaderCell ? "text-indigo-600 dark:text-indigo-400" : "opacity-60"} />
      </button>

      {divider}

      <button onClick={() => { editor.chain().focus().deleteTable().run(); setActiveDropdown(null); }} className={btnClass} title="Delete Table">
        <Trash2 size={14} className="text-red-600 dark:text-red-500" />
      </button>
    </div>
  );

const TiptapEditor = ({ value, onChange, onRequestMedia }) => {
  const wrapperRef = useRef(null);
  const [isEditorActive, setIsEditorActive] = useState(false);
  const [tableToolbarProps, setTableToolbarProps] = useState(null);

  const initialContent = (value || '').replace(/(&nbsp;|&#160;|\u00A0|\xC2\xA0)+/gi, ' ');

  // --- ENTERPRISE FIX ---
  // Memoize the extensions array to prevent Tiptap from detecting duplicate names across renders
  const extensions = useMemo(() => [
    StarterKit,
    CtaButton,
    RawHtmlBlock,
    Underline,
    TextStyle,
    Color,
    Highlight.configure({ multicolor: true }),
    TextAlign.configure({ types: ['heading', 'paragraph'] }),
    Subscript,
    Superscript,
    CustomLink,
    CustomImage,
    Table.configure({
      resizable: true,
      cellMinWidth: 48, 
      HTMLAttributes: {
        class: 'w-full border-collapse my-4 text-sm',
      },
    }),
    TableRow,
    CustomTableHeader.configure({
      HTMLAttributes: { class: 'px-4 py-2 text-left font-bold' },
    }),
    CustomTableCell.configure({
      HTMLAttributes: { class: 'px-4 py-2 text-slate-700 dark:text-slate-200' },
    }),
  ], []);

  const editor = useEditor({
    extensions: extensions,
    content: initialContent,
    editorProps: {
      attributes: { 
        class: 'tiptap-content focus:outline-none text-gray-800 dark:text-gray-200 w-full h-full block min-w-0 max-w-full',
      },
      transformPastedHTML(html) {
        return html.replace(/(&nbsp;|&#160;|\u00A0|\xC2\xA0)+/gi, ' ');
      },
      transformPastedText(text) {
        return text.replace(/[\u00A0\xC2\xA0]+/g, ' ');
      }
    },
    onUpdate: ({ editor }) => {
      onChange(editor.getHTML());
    },
  });

  const handleMediaRequest = useCallback((existingImageInfo = null) => {
    if (onRequestMedia) {
      onRequestMedia({
         ...existingImageInfo,
         insertImage: (url, altText = '', width = null, height = null) => {
            if (existingImageInfo && typeof existingImageInfo.pos === 'number') {
                editor.chain().focus().setNodeSelection(existingImageInfo.pos).updateAttributes('image', { src: url, alt: altText, width, height }).run();
            } else {
                editor.chain().focus().setImage({ src: url, alt: altText, width, height }).run();
            }
         }
      });
    } else {
       const url = window.prompt('Image URL', existingImageInfo?.src || '');
       if (url) {
          if (existingImageInfo && typeof existingImageInfo.pos === 'number') {
              editor.chain().focus().setNodeSelection(existingImageInfo.pos).updateAttributes('image', { src: url }).run();
          } else {
              editor.chain().focus().setImage({ src: url }).run();
          }
       }
    }
  }, [editor, onRequestMedia]);

  useEffect(() => {
    if (editor) {
      editor.storage.imageOptions = { onRequestMedia: handleMediaRequest };
    }
  }, [editor, handleMediaRequest]);

  useEffect(() => {
      const handleFocusIn = (e) => {
          if (wrapperRef.current && wrapperRef.current.contains(e.target)) {
              setIsEditorActive(true);
          }
      };
      const handleClick = (e) => {
          if (wrapperRef.current && wrapperRef.current.contains(e.target)) {
              setIsEditorActive(true);
          } else {
              setIsEditorActive(false); 
          }
      };

      document.addEventListener('mousedown', handleClick);
      document.addEventListener('focusin', handleFocusIn);
      return () => {
          document.removeEventListener('mousedown', handleClick);
          document.removeEventListener('focusin', handleFocusIn);
      };
  }, []);

  const updateTableToolbar = useCallback(() => {
    if (!editor || !editor.isActive('table') || !isEditorActive) {
        setTableToolbarProps(null);
        return;
    }

    const { view, state } = editor;
    const { from } = state.selection;

    try {
        let domNode = view.nodeDOM(from) || view.domAtPos(from).node;
        if (!domNode) return setTableToolbarProps(null);
        
        const element = domNode.nodeType === 3 ? domNode.parentNode : domNode;
        const tableElement = element.closest('table');
        const wrapperElement = tableElement?.closest('.tableWrapper') || tableElement;

        if (tableElement && wrapperElement && wrapperRef.current) {
            const wrapperRect = wrapperElement.getBoundingClientRect();
            // Calculate relative to the outermost wrapper
            const editorRect = wrapperRef.current.getBoundingClientRect();
            
            // Apply a 46px top offset to comfortably sit directly above the table's empty margin space
            const top = wrapperRect.top - editorRect.top - 46;
            const left = wrapperRect.right - editorRect.left;

            setTableToolbarProps({ top, left });
        } else {
            setTableToolbarProps(null);
        }
    } catch (e) {
        setTableToolbarProps(null);
    }
  }, [editor, isEditorActive]);

  useEffect(() => {
    if (!editor) return;
    
    const updateToolbars = () => {
        updateTableToolbar();
    };

    updateToolbars();

    editor.on('selectionUpdate', updateToolbars);
    editor.on('update', updateToolbars);
    window.addEventListener('resize', updateToolbars);

    return () => {
      editor.off('selectionUpdate', updateToolbars);
      editor.off('update', updateToolbars);
      window.removeEventListener('resize', updateToolbars);
    };
  }, [editor, updateTableToolbar, isEditorActive]);

  return (
    <div ref={wrapperRef} className="relative flex flex-col w-full max-w-full border border-gray-300 dark:border-gray-700 rounded-xl shadow-sm focus-within:ring-2 focus-within:ring-indigo-500 focus-within:border-transparent transition-all bg-white dark:bg-gray-800">
      
      <style>{`
        .tiptap-content h1 { font-size: 1.875rem; font-weight: 800; line-height: 1.2; margin: 2rem 0 1rem; letter-spacing: -0.025em; color: inherit; }
        .tiptap-content h2 { font-size: 1.5rem; font-weight: 700; line-height: 1.3; margin: 2rem 0 0.875rem; letter-spacing: -0.025em; color: inherit; }
        .tiptap-content h3 { font-size: 1.25rem; font-weight: 600; line-height: 1.4; margin: 1.5rem 0 0.75rem; color: inherit; }
        .tiptap-content h4 { font-size: 1.125rem; font-weight: 600; line-height: 1.5; margin: 1.5rem 0 0.75rem; color: inherit; }
        .tiptap-content h5 { font-size: 1rem; font-weight: 600; line-height: 1.5; margin: 1.5rem 0 0.75rem; color: inherit; }
        .tiptap-content h6 { font-size: 0.875rem; font-weight: 600; line-height: 1.5; margin: 1.5rem 0 0.75rem; color: inherit; text-transform: uppercase; }
