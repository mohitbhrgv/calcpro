<?php
// Core utilities
require_once '../../core/config.php';
require_once '../../core/response.php';
require_once '../../core/auth.php';

// --- SECURITY CHECK ---
$admin_data = verifyAdminToken();

try {
    // --- 1. Get Incoming Data ---
    $json = file_get_contents('php://input');
    $data = json_decode($json);

    // --- 2. Validation ---
    if (!isset($data->id) || !is_numeric($data->id)) {
        sendResponse(false, null, 'A valid Redirect ID is required.', 400, 'INVALID_INPUT');
    }

    $id = (int)$data->id;
    $updates = [];
    $executeParams = [];

    // Fetch existing redirect to ascertain the source_path in case only status is updated
    $existing_stmt = $conn->prepare("SELECT source_path FROM redirects WHERE id = :id LIMIT 1");
    $existing_stmt->execute([':id' => $id]);
    $existing_redirect = $existing_stmt->fetch(PDO::FETCH_ASSOC);
    
    if (!$existing_redirect) {
         sendResponse(false, null, 'Redirect rule not found.', 404, 'NOT_FOUND');
    }

    $final_source_path = $existing_redirect['source_path'];

    // Source Path Update
    if (isset($data->source_path)) {
        $source_path = trim($data->source_path, '/');
        
        if (empty($source_path)) {
            sendResponse(false, null, 'Source path cannot be empty.', 400, 'INVALID_INPUT');
        }
        
        // Check for duplicates (excluding self)
        $check_sql = "SELECT id FROM redirects WHERE source_path = :source_path AND id != :id";
        $check_stmt = $conn->prepare($check_sql);
        $check_stmt->execute([':source_path' => $source_path, ':id' => $id]);
        
        if ($check_stmt->fetch(PDO::FETCH_ASSOC)) {
            sendResponse(false, null, 'Duplicate source path.', 409, 'DUPLICATE_SOURCE');
        }

        $final_source_path = $source_path;
        $updates[] = "source_path = :source_path";
        $executeParams[':source_path'] = $source_path;
    }

    // Target URL Update
    if (isset($data->target_url)) {
        $target_url = trim($data->target_url);
        $updates[] = "target_url = :target_url";
        $executeParams[':target_url'] = $target_url;
    }

    // Status Code Update
    if (isset($data->status_code)) {
        $valid_codes = [301, 302, 307, 308, 410, 451];
        $status_code = in_array((int)$data->status_code, $valid_codes) ? (int)$data->status_code : 301;
        
        // Validation Logic for Target URL Requirement
        if (in_array($status_code, [301, 302, 307, 308])) {
            if (isset($data->target_url) && empty(trim($data->target_url))) {
                 sendResponse(false, null, 'Target URL is required.', 400, 'INVALID_INPUT');
            }
        }

        // --- PHASE 1 ENTERPRISE SAFEGUARD: Soft-Block & Warn ---
        // Prevent marking active URLs as 410 (Gone) or 451 (Legal) without explicit override
        if (in_array($status_code, [410, 451])) {
            $force_override = isset($data->force_override) ? (bool)$data->force_override : false;
            if (!$force_override) {
                $path_parts = explode('/', $final_source_path);
                $slug_to_check = end($path_parts);
                
                $is_live = false;
                // Strict Table Whitelist for SOC2/OWASP SQLi prevention
                $content_tables = ['pages', 'posts', 'calculators', 'categories', 'tags'];
                
                foreach ($content_tables as $table) {
                    $check_stmt = $conn->prepare("SELECT id FROM `$table` WHERE slug = :slug LIMIT 1");
                    $check_stmt->execute([':slug' => $slug_to_check]);
                    if ($check_stmt->fetch(PDO::FETCH_ASSOC)) {
                        $is_live = true;
                        break;
                    }
                }
                
                if ($is_live) {
                    sendResponse(false, null, 'Warning: This URL is currently mapped to live content.', 409, 'URL_CURRENTLY_LIVE');
                }
            }
        }

        $updates[] = "status_code = :status_code";
        $executeParams[':status_code'] = $status_code;
    }

    // Status Toggle Update ('active' / 'inactive')
    if (isset($data->status)) {
        if (in_array($data->status, ['active', 'inactive'])) {
            $updates[] = "status = :status";
            $executeParams[':status'] = $data->status;
        }
    }

    if (empty($updates)) {
        sendResponse(true, null, 'No changes to update.', 200);
    }

    // --- 3. Execute Update ---
    $sql = "UPDATE redirects SET " . implode(", ", $updates) . " WHERE id = :id";
    $executeParams[':id'] = $id;

    $stmt = $conn->prepare($sql);
    
    if ($stmt->execute($executeParams)) {
        sendResponse(true, null, 'Redirect rule updated successfully.', 200);
    } else {
        $errorInfo = $stmt->errorInfo();
        throw new Exception("Database update failed: " . $errorInfo[2]);
    }

} catch (Exception $e) {
    error_log("Error in update_redirect.php: " . $e->getMessage());
    sendResponse(false, null, 'An internal server error occurred.', 500, 'SERVER_ERROR');
}

$conn = null;
?>
