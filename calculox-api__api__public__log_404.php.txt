<?php
// File: calcpro-api/api/public/log_404.php

require_once '../../core/config.php';
require_once '../../core/response.php';

try {
    $json = file_get_contents('php://input');
    $data = json_decode($json);

    if (!isset($data->uri) || empty(trim($data->uri))) {
        exit();
    }

    // 1. FETCH SETTINGS
    $settings = [];
    $res = $conn->query("SELECT setting_key, setting_value FROM settings WHERE setting_key LIKE 'seo_404_monitor_%'");
    
    while($row = $res->fetch(PDO::FETCH_ASSOC)) {
        $settings[$row['setting_key']] = $row['setting_value'];
    }

    $mode         = $settings['seo_404_monitor_mode'] ?? 'simple';
    // Professional safeguard: Ensure limit is an integer and at least 100 if not 0
    $limit        = isset($settings['seo_404_monitor_limit']) ? (int)$settings['seo_404_monitor_limit'] : 100;
    $ignoreQuery  = ($settings['seo_404_monitor_ignore_query'] ?? 'true') === 'true';
    $excludePaths = json_decode($settings['seo_404_monitor_exclude'] ?? '[]', true);

    // 2. PREPARE URI
    $raw_uri = trim($data->uri);
    $uri = $raw_uri;

    // Implementation of 'Ignore Query Parameters'
    if ($ignoreQuery) {
        $parts = parse_url($raw_uri);
        $uri = $parts['path'] ?? $raw_uri;
    }

    // 3. CHECK EXCLUSIONS
    if (is_array($excludePaths)) {
        foreach ($excludePaths as $exclude) {
            if ($exclude['type'] === 'exact' && $uri === $exclude['value']) {
                exit();
            }
            if ($exclude['type'] === 'contains' && stripos($uri, $exclude['value']) !== false) {
                exit();
            }
        }
    }

    // 4. CAPTURE METADATA (Advanced Mode)
    // IP extraction removed for GDPR compliance
    $user_agent = null;
    $referer    = null;

    if ($mode === 'advanced') {
        $user_agent = $_SERVER['HTTP_USER_AGENT'] ?? null;
        $referer    = $_SERVER['HTTP_REFERER'] ?? null;
    }

    // Generate hash for efficient indexing
    $uri_hash = md5($uri);

    // 5. SAVE OR INCREMENT HIT
    // IP address fields removed from INSERT and UPDATE statements
    $sql = "INSERT INTO monitor_404_logs (uri, uri_hash, hits, user_agent, referer, created_at, updated_at) 
            VALUES (:uri, :uri_hash, 1, :user_agent, :referer, NOW(), NOW()) 
            ON DUPLICATE KEY UPDATE 
            hits = hits + 1, 
            user_agent = :update_user_agent, 
            referer = :update_referer,
            updated_at = NOW()";

    $stmt = $conn->prepare($sql);
    if ($stmt) {
        $stmt->execute([
            ':uri' => $uri,
            ':uri_hash' => $uri_hash,
            ':user_agent' => $user_agent,
            ':referer' => $referer,
            ':update_user_agent' => $user_agent,
            ':update_referer' => $referer
        ]);
    }

    // 6. ENFORCE LOG LIMIT (Database Rotation)
    // We enforce a hard floor of 100 to prevent accidental complete deletion if limit is malformed
    if ($limit > 0) {
        // Delete records that are NOT in the top X most recently updated
        $cleanup_sql = "DELETE FROM monitor_404_logs 
                        WHERE id NOT IN (
                            SELECT id FROM (
                                SELECT id FROM monitor_404_logs 
                                ORDER BY updated_at DESC 
                                LIMIT :limit
                            ) as tmp
                        )";
        
        $stmt_limit = $conn->prepare($cleanup_sql);
        if ($stmt_limit) {
            $stmt_limit->bindValue(':limit', $limit, PDO::PARAM_INT);
            $stmt_limit->execute();
        }
    }

} catch (Exception $e) {
    // Silent fail for public logger to prevent path disclosure
    exit();
}

$conn = null;
?>
