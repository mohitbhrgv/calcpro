<?php
// File: calcpro-api/api/admin/get_security_logs.php

// Core utilities
require_once '../../core/config.php';
require_once '../../core/response.php';
require_once '../../core/auth.php';
require_once '../../core/SecurityGuard.php'; // Required for Decryption & Blind Index

// --- SECURITY CHECK ---
$admin_data = verifyAdminToken();

function getBlockType($reason) {
    if ($reason === null) return 'other';
    $reason_lower = strtolower($reason);
    if (strpos($reason_lower, 'password') !== false) return 'failed_password';
    if (strpos($reason_lower, 'otp') !== false) return 'failed_otp';
    if (strpos($reason_lower, 'email') !== false || strpos($reason_lower, 'username') !== false) return 'failed_email';
    return 'other';
}

try {
    // --- 0. Get Configuration ---
    $limit = 15; // Default fallback
    $limit_query = $conn->query("SELECT setting_value FROM settings WHERE setting_key = 'security_log_pagination_limit'");
    
    if ($limit_query && $row = $limit_query->fetch(PDO::FETCH_ASSOC)) {
        $val = (int)$row['setting_value'];
        if ($val > 0) $limit = $val;
    }

    $search = isset($_GET['search']) ? trim($_GET['search']) : '';

    // --- Helper to get paginated data with CRYPTOGRAPHIC SEARCH support ---
    function fetchPaginatedData($conn, $table, $page, $limit, $search, $failureType = null) {
        $offset = ($page - 1) * $limit;
        
        // Build Secure Search Condition
        $whereSQL = "";
        $countParams = [];
        $isIpSearch = filter_var($search, FILTER_VALIDATE_IP);
        $searchBlindIndex = $search ? SecurityGuard::generateBlindIndex($search) : '';

        if (!empty($search)) {
            $searchTerm = "%{$search}%";
            if ($table === 'login_attempts') {
                if ($isIpSearch) {
                    $whereSQL = "WHERE (ip_blind_index = ? OR attempted_email LIKE ?)";
                    $countParams = [$searchBlindIndex, $searchTerm];
                } else {
                    $whereSQL = "WHERE (attempted_email LIKE ?)";
                    $countParams = [$searchTerm];
                }
            } elseif ($table === 'admin_login_failures') {
                // Handled in main query logic below
                $whereSQL = "";
            } else {
                // Blacklist & Temp Blocks
                if ($isIpSearch) {
                    $whereSQL = "WHERE (ip_blind_index = ? OR reason LIKE ?)";
                    $countParams = [$searchBlindIndex, $searchTerm];
                } else {
                    $whereSQL = "WHERE (reason LIKE ?)";
                    $countParams = [$searchTerm];
                }
            }
        }

        // --- 1. Get Total Count ---
        if ($table === 'admin_login_failures') {
            $count_sql = "
                SELECT COUNT(alf.id) as total 
                FROM admin_login_failures alf 
                JOIN admins a ON alf.admin_id = a.id
                WHERE 1=1
            ";
            if ($failureType) {
                $count_sql .= " AND alf.failure_type = ?";
                $countParams[] = $failureType;
            }

            if (!empty($search)) {
                if ($isIpSearch) {
                    $count_sql .= " AND (alf.ip_blind_index = ? OR a.email LIKE ?)";
                    $countParams[] = $searchBlindIndex;
                    $countParams[] = "%{$search}%";
                } else {
                    $count_sql .= " AND (a.email LIKE ?)";
                    $countParams[] = "%{$search}%";
                }
            }
        } else {
            $count_sql = "SELECT COUNT(id) as total FROM `$table` $whereSQL";
        }

        $stmt_count = $conn->prepare($count_sql);
        $stmt_count->execute($countParams);
        $totalResult = $stmt_count->fetch(PDO::FETCH_ASSOC);
        $total_records = $totalResult ? (int)$totalResult['total'] : 0;
        
        $total_pages = ceil($total_records / $limit);

        // --- 2. Get Data ---
        $sort_col = ($table === 'ip_blacklist' || $table === 'temporary_blocks') ? 'blocked_at' : 'attempted_at';
        $dataParams = [];
        
        if ($table === 'admin_login_failures') {
             $sql = "
                SELECT 
                    alf.id, alf.ip_address, alf.ip_blind_index, alf.attempted_at, alf.failure_type,
                    a.name as admin_name, a.email as admin_email
                FROM admin_login_failures alf
                JOIN admins a ON alf.admin_id = a.id
                WHERE 1=1
             ";
             if ($failureType) {
                 $sql .= " AND alf.failure_type = ?";
                 $dataParams[] = $failureType;
             }

             if (!empty($search)) {
                 if ($isIpSearch) {
                     $sql .= " AND (alf.ip_blind_index = ? OR a.email LIKE ?)";
                     $dataParams[] = $searchBlindIndex;
                     $dataParams[] = "%{$search}%";
                 } else {
                     $sql .= " AND (a.email LIKE ?)";
                     $dataParams[] = "%{$search}%";
                 }
             }

             $sql .= " ORDER BY alf.attempted_at DESC LIMIT ? OFFSET ?";
        } else {
            if (!empty($search)) {
                if ($table === 'login_attempts' && $isIpSearch) {
                    $dataParams = [$searchBlindIndex, "%{$search}%"];
                } elseif ($table === 'login_attempts' && !$isIpSearch) {
                    $dataParams = ["%{$search}%"];
                } elseif ($isIpSearch) {
                    $dataParams = [$searchBlindIndex, "%{$search}%"];
                } else {
                    $dataParams = ["%{$search}%"];
                }
            }
            $sql = "SELECT * FROM `$table` $whereSQL ORDER BY $sort_col DESC LIMIT ? OFFSET ?";
        }

        $stmt = $conn->prepare($sql);
        
        // Dynamically bind standard parameters as strings
        $paramIndex = 1;
        foreach ($dataParams as $param) {
            $stmt->bindValue($paramIndex++, $param, PDO::PARAM_STR);
        }
        
        // Strictly bind LIMIT and OFFSET as integers
        $stmt->bindValue($paramIndex++, $limit, PDO::PARAM_INT);
        $stmt->bindValue($paramIndex++, $offset, PDO::PARAM_INT);
        
        $stmt->execute();
        $data = $stmt->fetchAll(PDO::FETCH_ASSOC);

        // --- 3. ON-THE-FLY DECRYPTION ---
        $data = array_map(function($item) use ($table) {
            // Decrypt IP for UI display
            if (isset($item['ip_address'])) {
                $item['ip_address'] = SecurityGuard::decryptIp($item['ip_address']);
            }
            // Add block types
            if ($table === 'ip_blacklist' || $table === 'temporary_blocks') {
                $item['block_type'] = getBlockType($item['reason']);
            }
            return $item;
        }, $data);

        return [
            'data' => $data,
            'pagination' => [
                'currentPage' => $page,
                'totalPages' => $total_pages,
                'totalRecords' => $total_records
            ]
        ];
    }

    // --- Get Page Numbers ---
    $page_blacklist = isset($_GET['page_blacklist']) ? max(1, (int)$_GET['page_blacklist']) : 1;
    $page_temp      = isset($_GET['page_temp']) ? max(1, (int)$_GET['page_temp']) : 1;
    $page_unknown   = isset($_GET['page_unknown']) ? max(1, (int)$_GET['page_unknown']) : 1;
    $page_admin     = isset($_GET['page_admin']) ? max(1, (int)$_GET['page_admin']) : 1;
    $page_otp       = isset($_GET['page_otp']) ? max(1, (int)$_GET['page_otp']) : 1;

    // --- Fetch All Tables ---
    $ip_blacklist         = fetchPaginatedData($conn, 'ip_blacklist', $page_blacklist, $limit, $search);
    $temporary_blocks     = fetchPaginatedData($conn, 'temporary_blocks', $page_temp, $limit, $search);
    $login_attempts       = fetchPaginatedData($conn, 'login_attempts', $page_unknown, $limit, $search);
    $admin_login_failures = fetchPaginatedData($conn, 'admin_login_failures', $page_admin, $limit, $search, 'password');
    $otp_failures         = fetchPaginatedData($conn, 'admin_login_failures', $page_otp, $limit, $search, 'otp');

    // --- Response ---
    $security_logs = [
        'ipBlacklist'        => $ip_blacklist,
        'temporaryBlocks'    => $temporary_blocks,
        'loginAttempts'      => $login_attempts,
        'adminLoginFailures' => $admin_login_failures,
        'otpFailures'        => $otp_failures
    ];
    sendResponse(true, $security_logs, 'Security logs fetched successfully.');

} catch (Exception $e) {
    error_log("Error in get_security_logs.php: " . $e->getMessage());
    sendResponse(false, null, 'An internal server error occurred while fetching security logs.', 500, 'SERVER_ERROR');
}

$conn = null;
?>
