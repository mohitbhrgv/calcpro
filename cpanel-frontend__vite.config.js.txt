import { defineConfig } from 'vite';
import react from '@vitejs/plugin-react';
import tailwindcss from '@tailwindcss/vite';
import path from 'path';

// https://vitejs.dev/config/
export default defineConfig({
  plugins: [
    react(),
    tailwindcss(),
  ],
  base: '/',
  resolve: {
    alias: {
      '@': path.resolve(__dirname, './src')
    }
  },
  build: {
    outDir: 'dist',
    sourcemap: true,
    // Increased to 2500 to accommodate enterprise admin panel chunk sizes without warnings
    chunkSizeWarningLimit: 2500, 
    rollupOptions: {
      output: {
        manualChunks(id) {
          if (id.includes('node_modules')) {
            // 1. Advanced Split: Heavy SVG Icons mapped by Family
            if (id.includes('react-icons') || id.includes('lucide-react')) {
              if (id.includes('react-icons/fi')) return 'vendor-icons-fi';
              if (id.includes('react-icons/io5')) return 'vendor-icons-io5';
              if (id.includes('react-icons/hi')) return 'vendor-icons-hi';
              if (id.includes('react-icons/fa')) return 'vendor-icons-fa';
              if (id.includes('lucide-react')) return 'vendor-icons-lucide';
              return 'vendor-icons-core';
            }
            // 2. Heavy Data Visualization
            if (id.includes('echarts') || id.includes('zrender') || id.includes('chart.js') || id.includes('react-chartjs-2')) {
              return 'vendor-charts';
            }
            // 3. Heavy Animation Library
            if (id.includes('framer-motion')) {
              return 'vendor-animation';
            }
            // 4. Heavy UI Editors & Complex Inputs (Added Tiptap/ProseMirror)
            if (id.includes('react-quill') || id.includes('quill') || id.includes('react-select') || id.includes('@tiptap') || id.includes('prosemirror')) {
              return 'vendor-ui-heavy';
            }
            // 5. External APIs / SDKs / DB
            if (id.includes('@supabase') || id.includes('@questlabs')) {
              return 'vendor-sdks';
            }
            // 6. Strict React Core & Routing
            if (id.includes('/react/') || id.includes('/react-dom/') || id.includes('react-router')) {
              return 'vendor-react-core';
            }
            // 7. Standard Utilities
            return 'vendor-utils';
          }
        }
      }
    }
  },
  server: {
    allowedHosts: true 
  }
});
